export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "已复制！";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        复制提示词
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        在 Replit 中 Build ↗
      </button>
    </div>;
};

<div id="section">
  ## 您将实现的功能
</div>

完成本指南后，您将拥有：

* 一个由真实 Shopify 商店提供支持、采用自定义设计的蜡烛店面。
* 在 Shopify 中创建并展示在店面中的商品和价格。
* 可将买家引导至 Shopify 结账页面的购物车功能。
* 清晰的上线流程：申请商店、启用支付功能并发布。

<div id="section-2">
  ## 您将学到什么
</div>

您将了解如何：

* 描述您的商店，让 Agent 推荐并配置 Shopify。
* 无需处理 API 密钥即可连接 Shopify。
* 通过与 Agent 对话添加和发布商品。
* 优化商店设计和买家流程。
* 将商店从开发阶段转变为已发布、可销售的店面。

<div id="shopify">
  ## 连接 Shopify
</div>

所有构建者都能在任意应用中使用 Shopify，并直接在对话中完成连接。首先，描述一下你想构建的商店。

<Steps>
  <Step title="描述你的商店">
    复制提示词并粘贴到 [replit.com](https://replit.com) 的提示框中，或点击 **Build on Replit**，直接打开已预填该提示词的 Replit。

    <Frame>
      <img src="/assets/mirror/describe-your-store-b5d724daee99.png" alt="Replit 首页的提示框中已输入 Ember & Oak 蜡烛商店的提示词，随时可以开始构建" width="1440" height="900" data-path="images/replitai/shopify/describe-your-store.png" />
    </Frame>

    <PromptActions prompt={CANDLE_STORE_PROMPT} campaign="docs-ecommerce-store" />

    Agent 会识别出你要销售实体商品，并推荐使用 Shopify 作为店面的后端。

    <Frame>
      <img src="/assets/mirror/connect-shopify-store-7f36389f976f.png" alt="Replit Agent 为 Ember & Oak 蜡烛商店移交至 Shopify 技能，显示带有 Connect Shopify 按钮的「Connect to Shopify」卡片" width="3452" height="1984" data-path="images/replitai/shopify/connect-shopify-store.png" />
    </Frame>
  </Step>

  <Step title="批准创建 Shopify 商店">
    Agent 会请求为你的应用创建 Shopify 商店。批准后，Agent 会预配并连接一个由 Replit 拥有的开发商店，无需任何额外设置。
    你的邮箱地址会共享给 Shopify，以便你之后认领该商店。

    商店初始处于密码保护状态，在正式上线之前无法接受真实付款，因此你可以毫无风险地放心构建。
  </Step>
</Steps>

<Note>
  每个 Shopify 商店仅限一个应用使用。如果你之后要构建第二个商店，Agent 会为那个应用预配新的 Shopify 商店。有关连接器的更多信息，请参阅[连接 Shopify](/zh/ja/features/integrations/shopify)。
</Note>

<div id="section-3">
  ## 构建店面
</div>

连接 Shopify 后，Agent 会为你的商店设计并构建店面。

<Steps>
  <Step title="让 Agent 构建前端">
    Agent 会根据提示词生成店面，包括主视觉区、商品网格、购物车和支持区域。完成后，右侧面板会切换到 **Preview**，你可以滚动页面并点击设计元素。

    <Frame>
      <img src="/assets/mirror/storefront-preview-f5ecb9011a8c.png" alt="Agent 聊天旁的 Replit Preview 面板中呈现的 Ember & Oak 店面，显示蜡烛主视觉区和商品网格" width="3452" height="1978" data-path="images/replitai/shopify/storefront-preview.png" />
    </Frame>
  </Step>

  <Step title="添加第一件商品">
    新商店的商品目录一开始是空的。让 Agent 添加商品，它会在 Shopify 中创建商品并将其发布到店面。

    <AiPrompt>
      添加一款售价 $18 的薰衣草大豆蜡烛，附上一段简短温馨的描述，并将其发布到店面。
    </AiPrompt>

    商品会在 Shopify 商店中创建，并显示在店面的商品网格中。

    <Frame>
      <img src="/assets/mirror/add-product-8b098e606340.png" alt="添加薰衣草大豆蜡烛商品的 Agent 聊天。新商品显示在 Ember & Oak 店面的商品网格中" width="3452" height="1978" data-path="images/replitai/shopify/add-product.png" />
    </Frame>
  </Step>

  <Step title="丰富商品目录">
    继续描述并添加商品。Agent 会在 Shopify 中创建每件商品，包括价格、描述和图片。

    <AiPrompt>
      再添加 3 款蜡烛：Cedar &amp; Sage 售价 $20、Vanilla Bean 售价 $16、Sea Salt 售价 $22。请为每款添加简短描述并发布。
    </AiPrompt>
  </Step>
</Steps>

<Tip>
  **进阶技巧：** 让 Agent 将商品归入系列 (“使用 3 款最畅销的蜡烛创建一个节日系列”) 。你还可以添加商品变体，例如尺寸或香味；跟踪库存，并在店面显示库存状态。Shopify 是所有商品信息的唯一记录系统。
</Tip>

<div id="section-4">
  ## 改进设计与结账流程
</div>

店面的每个部分都可以通过对话来编辑。先把外观打磨好，再端到端地走一遍买家的购买流程。

<Steps>
  <Step title="改进设计">
    只需说明想要的改动，Agent 就会更新店面。

    <AiPrompt>
      请让主视觉区域更有杂志编辑的感觉 —— 更大的标题、更多留白，按钮使用柔和的琥珀色点缀。
    </AiPrompt>
  </Step>

  <Step title="测试购物车与结账">
    在 Preview 中把蜡烛加入购物车并开始结账。购物车会将买家引导至 Shopify 托管的结账页面。开发期间商店设置了密码，因此结账预览中可能会出现密码页面 —— 这是正常现象，并不是出了故障。
  </Step>
</Steps>

<Tip>
  **进阶玩法：** 让 Agent 添加商品详情页、购物车抽屉、搜索功能或精选商品板块。店面就是一个真正的 Replit 项目，只要你能描述出来，就能随意扩展 —— 博客、心愿单、自定义落地页等等 —— 结账和订单则交给 Shopify 处理。
</Tip>

<div id="section-5">
  ## 正式上线
</div>

正式上线在 Shopify 上完成。先让 Agent 启动上线流程，然后由商家完成需要自行操作的步骤。

<Steps>
  <Step title="在 Shopify 上认领店铺">
    认领在 Shopify 上进行，这是唯一一个在 Replit 之外完成的步骤。创建店铺时你的邮箱地址已同步给 Shopify，因此你可以通过 Replit 提供的链接，或 Shopify 发送的邮件来认领店铺。

    你会收到 Shopify 发来的 **店铺所有者邀请** 邮件，其中包含操作指引。打开邮件并选择 **Get started**。

    <Frame>
      <img src="/assets/mirror/claim-store-email-a0d108f2191c.png" alt="Shopify 的「Store owner invitation」邮件，邀请你成为店铺所有者，并显示「Get started」按钮" width="856" height="1024" data-path="images/replitai/shopify/claim-store-email.png" />
    </Frame>

    使用现有 Shopify 账户登录，或创建新账户以接受邀请。

    <Frame>
      <img src="/assets/mirror/claim-store-create-account-99d2e132da19.png" alt="用于接受邀请的 Shopify「Create an account」页面，显示通过邮箱、Google、Apple、Facebook 登录的选项" width="3452" height="1978" data-path="images/replitai/shopify/claim-store-create-account.png" />
    </Frame>

    选择 **Claim store**，为店铺授权 Replit 应用并开始免费试用。随后你将以主要所有者身份进入 Shopify 管理后台。

    <Frame>
      <img src="/assets/mirror/claim-store-4facb7c8b10c.png" alt="Replit 的「Claim your store」页面，带有用于开始 Shopify 免费试用的「Claim store」按钮" width="3452" height="1978" data-path="images/replitai/shopify/claim-store.png" />
    </Frame>
  </Step>

  <Step title="选择套餐并启用收款">
    选择 Shopify 套餐，并启用 Shopify Payments 或其他支付服务商。Shopify 会收集收款所需的身份验证信息和银行信息 —— 这一步不在 Replit 中进行。
  </Step>

  <Step title="设置配送、税费和域名">
    在 Shopify 中设置运费、核对税费，并可选择连接自定义域名。
  </Step>

  <Step title="发布上线">
    在 Shopify 中移除店面密码，向买家开放店铺。回到 Replit 并点击 **Publish**，将店面部署到公开 URL。
  </Step>
</Steps>

<Note>
  收款、身份验证、税费、配送和欺诈检测均在 Shopify 上处理。完整的上线检查清单和连接器详情，请参阅 [连接 Shopify](/zh/ja/features/integrations/shopify#go-live-on-shopify)。
</Note>
