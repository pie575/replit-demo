export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "Скопировано!";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        Скопировать промпт
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Создать в Replit ↗
      </button>
    </div>;
};

<div id="canvas">
  ## Доработка в Canvas и поиск альтернатив
</div>

Когда Agent завершит первую сборку, откройте Canvas, чтобы дать визуальную обратную связь и попросить Agent предложить другие варианты — без необходимости переписывать промпт.

<Steps>
  <Step title="Создание лендинга">
    Скопируйте запрос и вставьте его в [поле запроса на replit.com](https://replit.com) или нажмите **Build on Replit**, чтобы открыть Replit с уже введённым запросом.

    <PromptActions prompt={RACHEL_JONES_PROMPT} />

  <Step title="Предпросмотр приложения, созданного Agent">
    Когда Agent завершит работу, правая панель переключится на **Preview**, где вы сможете взаимодействовать с лендингом в реальном времени. Прокрутите главный экран, нажмите кнопки призыва к действию и убедитесь, что текст и изображения соответствуют вашему запросу, прежде чем приступать к доработке.

Уверенный разум. Неудержимая вы.» поверх золотистого портрета, кнопками «Заниматься с Rachel» и «Смотреть историю Rachel», сертификатами справа и полосой с пятью звёздами «Нам доверяют более 500 женщин по всему миру»" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-preview.png" />

  <Step title="Откройте Canvas">
    Выберите кнопку **Canvas** под вкладками Preview и Tools, чтобы переключить правую панель из Preview в Canvas.

    <Frame>
      <img src="/assets/mirror/landing-canvas-button-494c5b0076b1.png" alt="Рабочее пространство Replit с выделенной кнопкой Canvas под вкладками Preview и Tools" width="720" height="450" data-path="images/fitstart/launch-landing-page/landing-canvas-button.png" />

  </Step>

  <Step title="Комментирование вне режима предпросмотра">
    Используйте инструменты Canvas **Text** и **Draw**, чтобы добавлять заметки и стрелки в пустом пространстве вокруг макета, а затем указывать Agent на конкретные элементы, которые нужно изменить. За пределами области предпросмотра аннотации, размещённые на Canvas, остаются привязанными к обратной связи, чтобы Agent мог читать их вместе с дизайном.

    Посвящайте каждую аннотацию одному изменению. Короткие и конкретные заметки (&quot;Сделай эту кнопку уже&quot;) эффективнее расплывчатой обратной связи.

    </Frame>
  </Step>

  <Step title="Выберите комментарии и предпросмотр и отправьте их в Agent">
    Выделите рамкой комментарии вместе с фреймом предпросмотра, затем введите короткий промпт в **Chat about selected items** и отправьте его. Agent рассматривает комментарии и дизайн в едином контексте, поэтому понимает, к какому элементу относится каждая заметка.

    Пишите сообщение в чате в общих чертах — например: &quot;Примени изменения из комментариев.&quot; Конкретные детали указывайте в комментариях.

  </Step>

  <Step title="Поиск разных вариантов">
    Нажмите Design на Canvas и выберите предложение рядом с ним, например **Explore different approaches**, чтобы попросить Agent предложить альтернативные визуальные решения.

    </Frame>
  </Step>

  <Step title="Рассматривайте альтернативы и дорабатывайте">
    Дождитесь, пока Agent завершит генерацию. **Explore different approaches** обновляет тот же артефакт новым вариантом — переключайтесь между вариантами в выпадающем списке артефактов над макетом. **Explore different vibes** создает отдельные визуальные макеты, которые появляются на Canvas рядом с исходным.

    Когда варианты появятся, выберите направление и продолжайте итерации. Добавьте еще один комментарий к выбранному кадру, выделите рамкой его вместе с дизайном и отправьте новый запрос — например: &quot;Примени изменения, запрошенные в комментариях.&quot; Agent использует контекст варианта и новые комментарии как единый промпт.

  <Step title="Опубликовать работу">
    Когда страница будет готова в Preview, нажмите **Publish** (или **Republish** при последующих деплоях) в правом верхнем углу рабочего пространства, чтобы опубликовать последнюю версию по общедоступному URL. Перед тем как поделиться ссылкой, откройте опубликованный URL в новой вкладке и отправьте один тестовый ответ, чтобы убедиться, что опубликованная страница собирает данные.

    </Frame>
  </Step>
</Steps>
