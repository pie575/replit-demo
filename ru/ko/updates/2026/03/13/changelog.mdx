export const YouTubeEmbed = ({videoId, title = "Видео на YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

Agent 4 уже доступен. Созданный для творчества, Agent — это ваш креативный AI-партнёр, который превращает идеи в приложения, дизайны, презентации и многое другое с помощью обычного языка. Программировать не нужно.

Что можно делать с Agent 4:

* **Создавайте что угодно**: веб-приложения, мобильные приложения, дашборды с данными, инструменты на базе AI и многое другое
* **Проектируйте до разработки**: исследуйте макеты на [дизайн-холсте Canvas](/ru/learn/projects-and-artifacts/canvas) ещё до написания кода
* **Планируйте project**: используйте [режим Plan](/ru/core-concepts/agent/plan-mode), чтобы составить план разработки перед написанием кода
* **Выполняйте задачи параллельно**: Agent обрабатывает несколько задач одновременно, ускоряя работу
* **Работайте вместе с командой**: приглашайте других в project и создавайте вместе
* **Подключайте инструменты**: получайте данные из BigQuery, Linear, Slack, Notion и других сервисов прямо в чате
