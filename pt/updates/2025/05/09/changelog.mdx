export const YouTubeEmbed = ({videoId, title = "YouTube video", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

### Melhor tratamento de imagens com URLs do Unsplash quebradas

Diga adeus às imagens de banco de imagens quebradas em seus projetos!

Se você não sabia, o Agent já adiciona imagens de banco de imagens aos seus projetos do Unsplash. Mas às vezes as URLs quebram. O motivo? LLMs são ótimos em alucinar URLs de imagens de banco.

Adicionamos um recurso que substitui automaticamente imagens de banco quebradas por alternativas funcionais, garantindo que seus aplicativos sempre sejam exibidos de forma profissional.
