export const McpLinkGenerator = () => {

    if (typeof document === 'undefined') return 'https://replit.com/integrations?mcp=';
    const displayName = document.getElementById('mcp-display-name')?.value || 'My MCP Server';
    const baseUrl = document.getElementById('mcp-base-url')?.value || 'https://example.com/mcp';
    const payload = {
      displayName,
      baseUrl
    };
    const jsonString = JSON.stringify(payload);
    const encoded = btoa(unescape(encodeURIComponent(jsonString)));
    return `https://replit.com/integrations?mcp=${encoded}`;
  };
  const updateOutputs = () => {
    if (typeof document === 'undefined') return;
    const link = getLink();
    const badgeUrl = 'https://replit.com/badge?caption=Add%20to%20Replit';

    const linkEl = document.getElementById('mcp-link-output');
    const badgeMarkdownEl = document.getElementById('mcp-badge-markdown');
    if (linkEl) linkEl.textContent = link;
    if (badgeMarkdownEl) badgeMarkdownEl.textContent = badgeMarkdown;
  };
  const handleBadgeClick = e => {
    e.preventDefault();

        <label className="block text-sm font-medium mb-1">安装链接</label>
        <pre className="w-full p-3 bg-gray-100 dark:bg-gray-800 rounded-md text-sm break-all select-all cursor-text">

Replit Agent 可通过用于集成 AI 工具的开放标准 [模型上下文协议 (MCP)](https://modelcontextprotocol.io)，连接数百种外部工具和数据源。登录预注册的服务器，或添加你信任的自定义 MCP 服务器。

<div id="mcp">
  ## 连接预注册的 MCP 服务器
</div>

Replit 提供精选的 MCP 服务器，支持 Stripe、Linear、Notion 和 Sentry 等热门服务。只需点击一次即可登录并完成连接。

<Frame>
  <img src="/assets/mirror/mcp-servers-pane-8e1f431ca33c.png" alt="Replit Agent 的 MCP 服务器设置面板，列出了 Stripe、Linear、Notion、Sentry、Atlassian、Miro、PostHog、Amplitude、Mixpanel、Granola、Razorpay 等预注册服务器及各服务器的登录按钮" width="3452" height="1988" data-path="images/replitai/mcp-servers-pane.png" />
</Frame>

<Steps>
  <Step title="打开 Integrations 面板">
    前往 [replit.com/integrations](https://replit.com/integrations)，然后滚动至 **MCP Servers for Replit Agent**。
  </Step>

  <Step title="选择服务器">
    找到要连接的服务，例如 Stripe、Linear、Notion、Sentry、Atlassian、Miro、PostHog、Amplitude、Mixpanel、Granola 或 Razorpay。
  </Step>

  <Step title="登录">
    点击服务器旁的 **Sign in**。系统会引导你完成该服务的 OAuth 流程。连接完成后，服务器的连接状态将更新，Agent 即可使用该工具。
  </Step>
</Steps>

完整目录请参阅 [MCP 服务器参考](/zh/ko/features/mcp/overview)。

<div id="mcp-2">
  ## 添加自定义 MCP 服务器
</div>

如果所需的服务器不在预注册的目录中，可以通过 URL 添加。

<Frame>
  <img src="/assets/mirror/mcp-connect-dialog-5b8f9b0d828e.png" alt="MCP 服务器连接对话框，包含显示名称字段、MCP 服务器 URL 字段以及可展开的高级设置部分" width="3452" height="1988" data-path="images/replitai/mcp-connect-dialog.png" />
</Frame>

<Steps>
  <Step title="打开添加服务器对话框">
    在 MCP 服务器设置面板中，点击 **+ Add MCP server**。
  </Step>

  <Step title="设置显示名称">
    输入一个便于识别的名称。Agent 在聊天记录中引用该服务器时会使用此名称。
  </Step>

  <Step title="输入服务器 URL">
    粘贴 MCP 服务器的 HTTPS 端点。如果服务器需要身份验证，请打开 **Advanced settings**，并添加自定义请求头 (例如 `X-API-Key`) 。
  </Step>

  <Step title="测试并保存">
    点击 **Test &amp; save**。Replit 会尝试连接，并引导你完成服务器要求的 OAuth 流程。保存后，该连接会显示在 MCP 服务器列表中，并标明状态。
  </Step>
</Steps>

<Warning>
  仅连接你信任的 MCP 服务器。已连接的服务器可以向 Agent 提供工具和数据。在添加不熟悉的端点之前，请先核查其来源。
</Warning>

<div id="agent-mcp">
  ## 在 Agent 中使用 MCP 工具
</div>

服务器连接后，Agent 会自动获取工具列表，以便你在所有项目中使用这些功能。要使用服务器，只需在聊天中提及它：

> 使用 Notion MCP 服务器查找最新的会议记录。

<Frame>
  <img src="/assets/mirror/tool-call-7a53f51a11ae.png" alt="Agent 在聊天中调用自定义 MCP 服务器工具的示例" width="571" height="817" data-path="images/replitai/custom-mcp/tool-call.png" />
</Frame>

Agent 会根据你的请求，从服务器中选择合适的工具。如果工具需要确认，执行前会显示提示。
