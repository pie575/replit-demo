export const YouTubeEmbed = ({videoId, title = "Видео YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<YouTubeEmbed videoId="bAUVQfz_SIY" title="Представляем Replit Agent 4" />

Agent 4 уже здесь. Созданный для творчества, Agent — ваш ИИ-партнёр. Он превращает идеи в приложения, дизайны, слайды и многое другое. Кодирование не требуется.

Что можно делать с Agent 4:

* **Создавать что угодно**: веб-приложения, мобильные приложения, панели мониторинга данных, инструменты на базе ИИ и многое другое
* **Сначала создавать дизайн**: изучать макеты в [Canvas](/ru/learn/projects-and-artifacts/canvas), прежде чем приступать к написанию кода
* **Планировать проекты**: составлять план разработки в [режиме Plan](/ru/core-concepts/agent/plan-mode) до написания кода
* **Выполнять задачи параллельно**: Agent одновременно обрабатывает несколько задач, ускоряя работу
* **Сотрудничать с командой**: приглашать других в проект и разрабатывать вместе
* **Подключать инструменты**: получать данные из BigQuery, Linear, Slack, Notion и других сервисов прямо в чате
