export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "Скопировано!";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        Скопировать промпт
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Создать в Replit ↗
      </button>
    </div>;
};

Вы узнаете, как:

* Превратить общую идею продукта в один тестируемый сценарий.
* Использовать режим Plan, чтобы определить рамки прототипа перед его созданием.

* Параллельно проработать два направления прототипа, если вопрос о продукте остаётся неясным.

<div id="build-in-parallel">
  ### Build in parallel
</div>

Когда продуктовый вопрос касается направления, а не только реализации, проверка нескольких идей одновременно помогает сравнить варианты и не остановиться на одном слишком рано. Replit Agent поддерживает параллельные задачи: вы можете попросить Agent поработать сразу над несколькими прототипами, функциями или экспериментами и сравнить их между собой.

<Frame>
  <img src="/assets/mirror/task-plan-sidebar-ecab81b06bb0.png" alt="Представление thread с несколькими задачами Agent, выполняющимися параллельно, и индикаторами статуса" width="3430" height="1986" data-path="images/replitai/task-plan-sidebar.png" />
</Frame>

О том, как задачи ставятся в очередь, выполняются и завершаются, см. [Систему задач](/ru/core-concepts/agent/task-system).

Если вы не уверены, какое направление лучше, попросите Agent создать две параллельные задачи. Для Pace можно сравнить:

* **Прототип с упором на мотивацию**: серии, достижения, ободряющие рекомендации.
* **Прототип с упором на аналитику**: подробная статистика, динамика по неделям, личные рекорды.

<AiPrompt>
  Создай две параллельные задачи для улучшения Pace как продуктового прототипа.<br />
  Задача 1 — проработать версию с упором на мотивацию: серии, достижения и ободряющие рекомендации.<br />
  Задача 2 — проработать версию с упором на аналитику: подробная статистика, динамика по неделям и личные рекорды.<br />
  Обе версии должны быть достаточно небольшими, чтобы протестировать их уже сегодня.<br />
  Для каждой задачи определи пользовательский сценарий, что нужно создать и как мне оценивать результат.
</AiPrompt>

Когда обе задачи завершатся, сравните их, задав один и тот же вопрос:

> Какая версия с большей вероятностью заставит вас вернуться завтра и почему?

* **В прототипе слишком много функций:** попросите Agent сократить его до одного целевого пользователя и одного сценария.
* **Непонятна аудитория:** попросите Agent переписать приветственный раздел под одного целевого пользователя.
* **Фиктивные данные скрывают риски:** попросите Agent перечислить, что реально, а что сымитировано.
* **Расплывчатая обратная связь:** дайте проверяющим одну задачу и один вопрос.
* **Не удаётся опубликовать:** обратитесь к разделу [Устранение неполадок при публикации](/ru/build/troubleshooting).
