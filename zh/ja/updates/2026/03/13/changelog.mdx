export const YouTubeEmbed = ({videoId, title = "YouTube 视频", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<YouTubeEmbed videoId="bAUVQfz_SIY" title="Replit Agent 4 简介" />

Agent 4 现已推出。专为创意打造，Agent 是您的 AI 创意伙伴，可将想法变为应用、设计、幻灯片等，无需编程。

使用 Agent 4，您可以：

* **构建任何内容**：创建 Web 应用、移动应用、数据仪表板、AI 驱动的工具等
* **构建前先设计**：在开始编写代码前，于[设计 Canvas](/zh/learn/projects-and-artifacts/canvas)中探索模型稿
* **规划项目**：在编写代码前，使用 [Plan 模式](/zh/core-concepts/agent/plan-mode)规划构建流程
* **并行处理任务**：Agent 可同时处理多个任务，加快工作进度
* **与团队成员协作**：邀请他人加入项目，同时进行构建
* **连接工具**：直接在聊天中从 BigQuery、Linear、Slack、Notion 等获取数据
