export const YouTubeEmbed = ({videoId, title = "YouTube 视频", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

Agent 4 现已推出。Agent 专为创意而生，是你的 AI 创意伙伴——只需使用自然语言，就能将想法变成 App、设计稿、幻灯片等。无需编程。

使用 Agent 4，你可以：

* **构建任何内容**：创建 web 应用、移动应用、数据仪表板、AI 驱动的工具等
* **构建前先设计**：在编写代码前，先在 [Design Canvas](/zh/learn/projects-and-artifacts/canvas) 中探索原型稿
* **规划你的项目**：在编写任何代码前，使用 [Plan 模式](/zh/core-concepts/agent/plan-mode) 规划构建方案
* **并行运行 task**：Agent 可同时处理多个 task，助你更快推进
* **与团队成员协作**：邀请他人加入项目，同时开展构建
* **连接你的工具**：直接通过 Chat 从 BigQuery、Linear、Slack、Notion 等工具中获取数据
