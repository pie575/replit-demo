export const YouTubeEmbed = ({videoId, title = "YouTube 동영상", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

</div>

Replit 팀은 만드는 과정을 더 매끄럽고 직관적으로 만들기 위해 다양한 개선 사항을 적용했습니다. 더 안정적인 preview pane부터 더 일관된 모바일 동작까지, 이러한 개선으로 아이디어를 현실로 구현하기에 한층 완성도 높은 환경이 마련되었습니다.

Replit 팀은 보이지 않는 곳에서 더 빠르고 안정적인 플랫폼을 제공하기 위해 인프라를 현대화하고 있습니다. 이러한 기술적 개선은 앞으로 더 강력한 기능과 더 나은 개발자 경험을 제공하기 위한 발판이 됩니다.
