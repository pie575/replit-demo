export const YouTubeEmbed = ({videoId, title = "Видео на YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<div id="melhor-tratamento-de-imagens-com-urls-do-unsplash-quebradas">
  ### Улучшенная обработка изображений при битых ссылках Unsplash
</div>

Попрощайтесь с битыми стоковыми изображениями в ваших проектах!

Если вы не знали, Agent уже добавляет в ваши проекты стоковые изображения из Unsplash. Но иногда ссылки перестают работать. Причина? LLM отлично умеют выдумывать адреса стоковых изображений.

Мы добавили функцию, которая автоматически заменяет битые стоковые изображения рабочими аналогами, так что ваши приложения всегда выглядят профессионально.

Клиенты Enterprise теперь получают больше контроля над приватностью развертываний благодаря нескольким вариантам настройки.

Эта улучшенная возможность позволяет организациям гибко управлять доступом к развернутым приложениям, обеспечивая безопасность и гибкость, которые требуются в средах Enterprise. Пользователи команд также получают преимущество: развертывания по умолчанию являются приватными, а для публичного доступа требуется явное подтверждение.
