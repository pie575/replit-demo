export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "已复制！";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        复制提示词
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        在 Replit 上构建 ↗
      </button>
    </div>;
};

<div id="what-youll-learn">
  ## 您将学到什么
</div>

您将学习如何：

* 将目标受众、优惠和行动号召转化为页面。
* 让 Agent 生成页面文案、布局和表单行为。
* 在分享页面前检查页面是否清晰易懂。
* 通过已发布页面的 URL 测试表单。
* 根据回复质量改进页面。

<div id="refine-and-explore-alternatives-in-canvas">
  ## 在 Canvas 中优化并探索其他方案
</div>

Agent 完成首次构建后，打开 Canvas 提供视觉反馈，并让 Agent 尝试其他方向，无需重写提示词。

<Steps>
  <Step title="构建落地页">
    复制提示词并粘贴到 [replit.com](https://replit.com) 的提示框中，或点击 **在 Replit 上 Build**，在 Replit 中打开已预填提示词的内容。

    <PromptActions prompt={RACHEL_JONES_PROMPT} campaign="docs-landing-page" />
  </Step>

  <Step title="预览 Agent 构建的 App">
    Agent 完成后，右侧窗格会切换到 **Preview**，你可以实时与落地页交互。在开始优化前，滚动查看主视觉区域，点击行动号召按钮，并确认文案和图片是否符合你的要求。

    <Frame>
      <img src="/assets/mirror/landing-preview-4855c4b25270.png" alt="Replit Preview 窗格中呈现的 Rachel Jones 健身教练落地页，位于 Agent chat 旁——深色编辑风格的主视觉区域展示了一幅金色光线映照的人像，标题为“更强壮的身体。更自信的心态。势不可挡的你。”，包含“与 Rachel 合作”和“观看 Rachel 的故事”按钮，右侧显示认证资质，以及“受到全球 500 多名女性信赖”的五星评价栏" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-preview.png" />
    </Frame>
  </Step>

  <Step title="打开 Canvas">
    选择 Preview 和 Tools 标签页下的 **Canvas** 按钮，将右侧窗格从 Preview 切换为 Canvas。

    <Frame>
      <img src="/assets/mirror/landing-canvas-button-494c5b0076b1.png" alt="Replit workspace，Preview 和 Tools 标签页下的 Canvas 按钮已高亮显示" width="720" height="450" data-path="images/fitstart/launch-landing-page/landing-canvas-button.png" />
    </Frame>
  </Step>

  <Step title="在预览区域外添加批注">
    使用 Canvas 的 **文本** 和 **绘制** 工具，在模型周围的空白区域添加注释和箭头，并将箭头指向你希望 Agent 修改的具体元素。放在 Canvas 上、预览区域外的注释会附带在你的反馈中，方便 Agent 结合设计一同查看。

    每条注释只说明一项修改。简短、具体的说明 (&quot;缩小此按钮的宽度&quot;) 比开放式反馈更有效。

    <Frame>
      <img src="/assets/mirror/landing-canvas-annotate-text-8135bfe1e8e2.png" alt="Canvas 注释内容为“缩小此按钮的宽度”，位于 Rachel Jones 着陆页右侧的 Canvas 空白边距中，箭头指向页面页眉中的金色“预约免费咨询”按钮" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-annotate-text.png" />
    </Frame>
  </Step>

  <Step title="选择标注和预览，然后发送给 Agent">
    框选标注和预览 frame，在 **Chat about selected items** 中输入简短的 prompt，然后发送。Agent 会将标注和设计视为同一上下文，因此能知道每条标注对应哪个元素。

    聊天消息应保持概括性，例如“应用标注中的更改。”让标注承载具体细节。

    <Frame>
      <img src="/assets/mirror/landing-canvas-send-24b771bf29bf.png" alt="Rachel Jones landing page 与 Canvas 上的两条标注一同被选中：“缩小此按钮的宽度”指向 Book a Free Consultation 按钮，“为 Rachel 的图片添加滑入动画”指向主视觉照片；Chat about selected items 输入框中已输入聊天 prompt" width="1726" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-send.png" />
    </Frame>
  </Step>

  <Step title="探索不同方向">
    点击 Canvas 上的设计，选择其旁边的建议 (例如 **探索不同方案**) ，让 Agent 提供其他视觉设计方向。

    <Frame>
      <img src="/assets/mirror/landing-canvas-reimagine-9e60c24f342f.png" alt="在 Canvas 上原始设计旁以 Design frame 形式生成的三个备选落地页设计方案" width="1726" height="994" data-path="images/fitstart/launch-landing-page/landing-canvas-reimagine.png" />
    </Frame>
  </Step>

  <Step title="查看备选方案并反复优化">
    等待 Agent 完成生成。**探索不同方案**会以新变体更新同一 artifact——你可以通过模型图上方的 artifact 下拉菜单在不同变体之间切换。**探索不同风格**会生成独立的视觉模型图，并显示在 Canvas 上原始模型图旁边。

    变体生成后，选择一个方向并继续迭代。在选定的 frame 上添加另一条注释，将它与设计一起框选，然后发送新的请求——例如，&quot;应用注释中提出的更改。&quot;Agent 会将变体上下文和你的最新注释合并为一个 prompt。

    <Frame>
      <img src="/assets/mirror/landing-canvas-zoomed-4d1585d33fe6.png" alt="Agent 生成三种风格变体（Editorial Warmth、Quiet Clinical、Vibrant Pop）后 Canvas 上的 Rachel Jones landing page——原始 frame 被选中，并带有两条新注释（“减小此按钮的宽度”和“为 Rachel 的图片添加滑入动画”），Chat 中显示“应用注释中提出的更改”" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-zoomed.png" />
    </Frame>
  </Step>

  <Step title="发布你的作品">
    页面在 Preview 中正常运行后，点击 workspace 右上角的 **Publish** (后续部署时为 **Republish**) ，将最新版本发布到公开 URL。在新标签页中打开已发布的 URL，并提交一条测试回复，确认线上页面能捕获该回复后再分享。

    <Frame>
      <img src="/assets/mirror/landing-publish-e3b901a79440.png" alt="放大 2 倍的 Replit workspace 右上角——标题栏中的灰色 Republish 按钮以橙色边框突出显示，下方弹出菜单显示“Raouf 约 18 小时前发布”、Visibility: Public，以及 Review security 和 Republish 操作按钮" width="2142" height="1072" data-path="images/fitstart/launch-landing-page/landing-publish.png" />
    </Frame>
  </Step>
</Steps>
