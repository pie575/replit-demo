export const YouTubeEmbed = ({videoId, title = "Видео на YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<div id="unsplash-url">
  ### Улучшенная обработка изображений с неработающими URL Unsplash
</div>

Попрощайтесь с неработающими стоковыми изображениями в проектах!

Как вы знаете, Agent уже добавляет в проекты стоковые изображения с Unsplash. Но URL иногда оказываются нерабочими. Почему? LLM отлично умеют галлюцинировать URL стоковых изображений.

Мы добавили функцию, которая автоматически заменяет неработающие стоковые изображения рабочими альтернативами, чтобы ваши App всегда выглядели профессионально.
