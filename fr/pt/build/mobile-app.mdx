    if (typeof window === "undefined") {
      return null;
    }
    if (!window.__replitDocsAgentInputTypewriterLoader) {
      window.__replitDocsAgentInputTypewriterLoader = {};
    }
    return window.__replitDocsAgentInputTypewriterLoader;
  };
  const ensureAgentInputTypewriterCore = () => {
    if (typeof window === "undefined" || typeof document === "undefined") {
      return Promise.resolve(null);
    }
    if (window.Typewriter) {
      return Promise.resolve(window.Typewriter);
    }
    const loader = getAgentInputTypewriterLoader();
    if (loader.promise) {
      return loader.promise;
    }
    loader.promise = new Promise((resolve, reject) => {
      const handleLoad = () => {
        if (window.Typewriter) {
          resolve(window.Typewriter);
          return;
        }
        loader.promise = null;
        reject(new Error("Typewriter core loaded without a global Typewriter export."));
      };
      const handleError = () => {
        loader.promise = null;
        reject(new Error("Failed to load typewriter core."));
      };
      const existingScript = document.getElementById(AGENT_INPUT_TYPEWRITER_SCRIPT_ID);

          once: true

          once: true

      script.addEventListener("load", handleLoad, {

        once: true

  };
  const stopAgentInputTypewriter = (textarea, {placeholder, clearDemo = false} = {}) => {

<div id="defina-o-fluxo-mobile">
  ## Définissez le parcours mobile
</div>

Avant de vous lancer, déterminez ce qu&#39;un utilisateur doit pouvoir faire depuis son téléphone.

Pour Pace Mobile :

* **Public** : les coureurs qui souhaitent enregistrer une course rapidement.
* **Parcours principal** : choisir le type de course → saisir la distance et le temps → enregistrer la course → consulter le résumé du jour.
* **Contrainte mobile** : l&#39;action principale doit rester accessible d&#39;un seul pouce.
* **Première version** : pas de comptes, pas de fil social, pas d&#39;analyses avancées.

La première version mobile reste ainsi suffisamment réduite pour être testée.

Ce processus nécessite une adhésion à l&#39;**Apple Developer Program** avec l&#39;authentification à deux facteurs. Pour en savoir plus sur l&#39;inscription, les comptes individuels ou organisationnels, TestFlight ou la validation de l&#39;App Store et les tests externes, consultez [Créez et lancez une app mobile](/fr/pt/build/mobile-app).

<Steps>
  <Step title="Ouvrez l’onglet Publication et démarrez la publication sur l’App Store">
    Une fois votre aperçu actif dans Expo Go, ouvrez l&#39;onglet **Publication**. Sous **Production**, une carte **Prêt à publier Pace sur l&#39;App Store ?** s&#39;affiche. Sélectionnez **Commencer la publication sur l&#39;App Store**.

    </Frame>
  </Step>

  <Step title="Choisissez votre projet Expo">
    Replit ouvre **Launch** — un flux Expo intégré à Replit. Saisissez ou confirmez le nom et le slug de votre projet Expo, puis sélectionnez **Continuer**. Si vous avez déjà un projet Expo, sélectionnez **Utiliser l&#39;existant**.

    <Frame>
      <img src="/assets/mirror/app-store-03-choose-expo-project-4707fafffafe.png" alt="La boîte de dialogue Launch avec l'étape Choisir un projet Expo active — champs Nom du projet et Slug du projet pré-remplis avec Pace, un bouton Continuer et une option Utiliser l'existant" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-03-choose-expo-project.png" />
    </Frame>
  </Step>

  <Step title="Se connecter avec Apple">
    Connectez le compte Apple Developer qui sera propriétaire de la fiche sur l&#39;App Store. Ce compte doit être inscrit à l&#39;Apple Developer Program et Apple impose l&#39;authentification à deux facteurs : préparez-vous à valider un code de vérification sur un appareil de confiance.

    <Frame>
      <img src="/assets/mirror/app-store-04-sign-in-apple-459dfc9b84ba.png" alt="La boîte de dialogue Launch à l'étape Se connecter avec Apple — champs d'adresse e-mail ou de numéro de téléphone et de mot de passe Apple Developer, avec une note indiquant que le mot de passe ne quitte jamais l'appareil et que le compte doit être inscrit à l'Apple Developer Program" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-04-sign-in-apple.png" />
    </Frame>
  </Step>

  <Step title="Choisissez ou créez votre app Apple">
    Saisissez le **Nom de l&#39;app** et le **Bundle ID** (par exemple, `app.replit.pace`) qui apparaîtront sur l&#39;App Store. L&#39;identifiant de bundle ne peut plus être modifié après le premier build : choisissez-le donc avec soin.

      <img src="/assets/mirror/app-store-05-choose-apple-app-3971aea62f14.png" alt="La boîte de dialogue Launch à l'étape Choisir l'app Apple — Nom de l'app défini sur Pace, Bundle ID app.replit.pace, avec une note indiquant que l'identifiant de bundle ne pourra pas être modifié par la suite, et un bouton Création de l'app…" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-05-choose-apple-app.png" />
    </Frame>
  </Step>

  <Step title="Publier sur l’App Store">
    Replit configure les certificats Apple en votre nom. Lorsque l’étape **Prêt à lancer** s’affiche, sélectionnez **Publier sur l’App Store**. Replit compile votre app native dans le cloud et envoie le build à App Store Connect pour examen bêta via TestFlight.

    <Frame>
      <img src="/assets/mirror/app-store-06-ready-for-launch-ab02659ee0bb.png" alt="La boîte de dialogue Launch à l’étape Prêt à lancer, avec Choisir une app Apple et Configurer les certificats Apple marqués comme terminés, ainsi qu’un bouton Publier sur l’App Store et des paramètres de lancement facultatifs" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-06-ready-for-launch.png" />
    </Frame>

    Lorsque Launch est terminé, vous verrez une confirmation indiquant que l’app est en cours d’envoi vers l’App Store. La boîte de dialogue se ferme ensuite et vous ramène à l’Éditeur de projet.

  </Step>
</Steps>

<Note>
  La première build d’une version nécessite généralement une validation bêta par TestFlight avant que les testeurs puissent l’installer. Après approbation, les builds suivantes sont généralement disponibles plus rapidement pour les testeurs. Pour partager l’app avec des testeurs externes ou la soumettre à l’App Store, complétez sa fiche dans **App Store Connect** — consultez [Créer et lancer une app mobile](/fr/pt/build/mobile-app) pour obtenir la liste de contrôle complète d’App Store Connect.
</Note>

<div id="voc-terminou-quando">
  ## C&#39;est terminé lorsque
</div>

* L&#39;app s&#39;ouvre dans le simulateur iOS ou l&#39;émulateur Android.
* L&#39;app s&#39;ouvre sur votre téléphone avec Expo Go.
* Vous parvenez à réaliser le parcours mobile principal.
* Le texte, les boutons et la navigation sont agréables à utiliser sur un téléphone.
* Vous avez publié ou partagé une version mobile testable.
* Au moins une personne peut l&#39;essayer et donner son avis.
* (Facultatif) Vous avez envoyé l&#39;app sur TestFlight via le flux App Store.

<div id="precisa-de-mais-ajuda">
  ## Besoin d’aide supplémentaire ?
</div>

* **L’Agent crée une application web :** recommencez et sélectionnez **App mobile** comme type d’application avant de lancer la création.
* **L’aperçu dans Expo Go semble obsolète :** secouez votre téléphone, ouvrez le menu développeur et sélectionnez **Actualiser**.
* **Les zones tactiles semblent trop petites :** demandez à l’Agent d’agrandir les zones tactiles et de garder les actions principales à portée du pouce.
* **La navigation semble encombrée :** demandez à l’Agent de réduire l’application au nombre minimal d’écrans nécessaire au parcours principal.
* **Le simulateur fonctionne, mais pas le téléphone :** testez les comportements propres au matériel sur un appareil réel et consultez [Apps mobiles natives](/fr/pt/features/artifact-types/building-mobile-apps).
* **La publication échoue :** consultez [dépannage des apps mobiles](/fr/pt/features/troubleshooting/mobile-app).
