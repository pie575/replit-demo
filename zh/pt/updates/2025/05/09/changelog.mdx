export const YouTubeEmbed = ({videoId, title = "YouTube 视频", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<div id="melhor-tratamento-de-imagens-com-urls-do-unsplash-quebradas">
  ### 更好地处理 Unsplash 失效图片链接
</div>

告别项目中加载不出来的图库图片！

或许你还不知道，Agent 早就会从 Unsplash 为你的项目添加图库图片了。但有时链接会失效。原因？大语言模型特别擅长「编造」图库图片的 URL。

我们新增了一项功能，可自动将失效的图库图片替换为可正常显示的替代图片，让你的应用始终保持专业的外观。
