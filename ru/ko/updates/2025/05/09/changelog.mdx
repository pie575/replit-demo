export const YouTubeEmbed = ({videoId, title = "Видео на YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<div id="unsplash-url">
  ### Улучшенная обработка изображений при неработающих URL Unsplash
</div>

Попрощайтесь с битыми стоковыми изображениями в проектах!

Если вы не знали, Agent уже добавляет в проекты стоковые изображения из Unsplash. Но иногда ссылка оказывается нерабочей. Почему? Потому что LLM прекрасно умеют выдумывать URL стоковых изображений.

Мы добавили автоматическую замену битых стоковых изображений на рабочие аналоги, чтобы ваши приложения всегда выглядели профессионально.
