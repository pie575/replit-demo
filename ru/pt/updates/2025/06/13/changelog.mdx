export const YouTubeEmbed = ({videoId, title = "Видео на YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<div id="melhoria-no-compartilhamento-de-aplicativos-da-organizao">
  ### Улучшение общего доступа к App организации
</div>

Поддерживайте порядок в workspace своих команд! Новые приложения, созданные в организациях, больше не будут автоматически доступны всем участникам. Это уменьшает беспорядок и даёт больше контроля над тем, что отображается в workspaces участников команды.

<div id="melhorias-na-pgina-de-membros-de-equipes">
  ### Улучшения страницы участников команд
</div>

Управлять командой стало проще благодаря улучшениям интерфейса страницы участников команд. Теперь на вкладках «Участники», «Наблюдатели» и «Приглашения» можно сортировать, фильтровать и искать. Эти возможности, о которых просили клиенты Enterprise, помогают эффективно управлять большими командами.
