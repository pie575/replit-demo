export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "Copié !";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        Copier le prompt
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Créer avec Replit ↗
      </button>
    </div>;
};

<div id="what-youll-learn">
  ## Ce que vous allez apprendre
</div>

Vous allez apprendre à :

* Transformer une audience, une offre et un appel à l&#39;action en une page.
* Demander à Agent le texte de la page, sa disposition et le comportement du formulaire.
* Vérifier que la page est claire avant de la partager.
* Tester un formulaire depuis l&#39;URL publiée.
* Améliorer la page en fonction de la qualité des réponses.

<div id="refine-and-explore-alternatives-in-canvas">
  ## Affiner et explorer des alternatives dans Canvas
</div>

Une fois qu&#39;Agent a terminé la première version, ouvrez Canvas pour donner un retour visuel et demander à Agent d&#39;explorer d&#39;autres directions, sans avoir à réécrire le prompt.

<Steps>
  <Step title="Créez la page de destination">
    Copiez le prompt et collez-le dans le champ de prompt sur [replit.com](https://replit.com), ou cliquez sur **Build on Replit** pour ouvrir Replit avec le prompt prérempli.

    <PromptActions prompt={RACHEL_JONES_PROMPT} campaign="docs-landing-page" />
  </Step>

  <Step title="Prévisualisez l’app créée par Agent">
    Une fois qu&#39;Agent a terminé, le volet de droite bascule vers **Preview** pour vous permettre d&#39;interagir avec la page de destination en temps réel. Faites défiler la section héros, cliquez sur les appels à l&#39;action et vérifiez que les textes et les visuels correspondent à votre demande avant de commencer à les affiner.

    <Frame>
      <img src="/assets/mirror/landing-preview-4855c4b25270.png" alt="La page de destination de la coach sportive Rachel Jones affichée dans le volet Preview de Replit, à côté du Chat Agent — une section héros sombre au style éditorial avec le titre « Stronger body. Confident mind. Unstoppable you. » sur un portrait aux reflets dorés, les boutons Work with Rachel et Watch Rachel's Story, les certifications sur la droite et un bandeau cinq étoiles « Trusted by 500+ women worldwide »" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-preview.png" />
    </Frame>
  </Step>

  <Step title="Ouvrir Canvas">
    Sélectionnez le bouton **Canvas** sous les onglets Preview et Tools pour faire passer le volet de droite de Preview à Canvas.

    <Frame>
      <img src="/assets/mirror/landing-canvas-button-494c5b0076b1.png" alt="Le workspace Replit avec le bouton Canvas mis en évidence sous les onglets Preview et Tools" width="720" height="450" data-path="images/fitstart/launch-landing-page/landing-canvas-button.png" />
    </Frame>
  </Step>

  <Step title="Annotez en dehors de l’aperçu">
    Utilisez les outils **Texte** et **Dessin** du Canvas pour ajouter des notes et des flèches dans l&#39;espace vide autour du mockup, puis pointez-les vers l&#39;élément précis que vous souhaitez faire modifier par Agent. Les annotations placées sur le canvas — en dehors de la Preview — restent rattachées à votre retour, ce qui permet à Agent de les lire en même temps que le design.

    Limitez chaque annotation à une seule modification. Des notes courtes et précises (&quot;Réduire la largeur de ce bouton&quot;) sont plus efficaces que des retours vagues.

    <Frame>
      <img src="/assets/mirror/landing-canvas-annotate-text-8135bfe1e8e2.png" alt="Une annotation Canvas indiquant « Réduire la largeur de ce bouton », placée dans la marge vide du canvas à droite de la page de destination de Rachel Jones, avec une flèche pointant vers le bouton doré « Book a Free Consultation » dans l'en-tête de la page" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-annotate-text.png" />
    </Frame>
  </Step>

  <Step title="Sélectionnez les annotations et l’aperçu, puis envoyez le tout à Agent">
    Sélectionnez au lasso vos annotations en même temps que le frame de Preview, saisissez un court prompt dans **Chat about selected items**, puis envoyez-le. Agent lit les annotations et le design comme un seul et même contexte : il sait donc à quel élément chaque note se rapporte.

    Restez général dans le message de chat, par exemple : &quot;Applique les changements indiqués dans les annotations.&quot; Ce sont les annotations qui portent les détails.

    <Frame>
      <img src="/assets/mirror/landing-canvas-send-24b771bf29bf.png" alt="La page de destination de Rachel Jones sélectionnée avec deux annotations sur Canvas — « Réduire la largeur de ce bouton » pointant vers le bouton Book a Free Consultation et « Ajouter une animation d'entrée glissée pour l'image de Rachel » pointant vers la photo principale — avec un prompt saisi dans le champ Chat about selected items" width="1726" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-send.png" />
    </Frame>
  </Step>

  <Step title="Explorez différentes pistes">
    Cliquez sur le design dans Canvas et choisissez une suggestion affichée à côté, comme **Explore different approaches**, pour demander à Agent d&#39;explorer d&#39;autres directions visuelles.

    <Frame>
      <img src="/assets/mirror/landing-canvas-reimagine-9e60c24f342f.png" alt="Trois directions alternatives de page de destination générées sous forme de Design frames sur le canvas, à côté de l'original" width="1726" height="994" data-path="images/fitstart/launch-landing-page/landing-canvas-reimagine.png" />
    </Frame>
  </Step>

  <Step title="Examinez les variantes et affinez-les">
    Attendez qu&#39;Agent termine la génération. **Explorer différentes approches** met à jour le même artifact avec de nouvelles variantes — passez de l&#39;une à l&#39;autre depuis le menu déroulant de l&#39;artifact situé au-dessus du mockup. **Explorer différentes ambiances** génère des mockups visuels distincts qui apparaissent sur le canvas, à côté de l&#39;original.

    Une fois les variantes générées, choisissez une direction et poursuivez les itérations. Ajoutez une nouvelle annotation sur le frame retenu, sélectionnez-le avec le design à l&#39;aide du rectangle de sélection, puis envoyez une nouvelle demande — par exemple : &quot;Applique les modifications demandées dans les annotations.&quot; Agent traite le contexte de la variante et vos nouvelles annotations comme un seul prompt.

    <Frame>
      <img src="/assets/mirror/landing-canvas-zoomed-4d1585d33fe6.png" alt="La page de destination de Rachel Jones sur Canvas après qu'Agent a généré trois variantes d'ambiance (Editorial Warmth, Quiet Clinical, Vibrant Pop) — le frame original est sélectionné avec deux nouvelles annotations (« Réduis la largeur de ce bouton » et « Ajoute une animation de glissement pour l'image de Rachel ») et le chat indique « Applique les modifications demandées dans les annotations »" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-zoomed.png" />
    </Frame>
  </Step>

  <Step title="Publiez votre travail">
    Lorsque la page fonctionne dans Preview, cliquez sur **Publier** (ou sur **Republish** pour les déploiements suivants) en haut à droite du workspace afin de mettre la dernière version en ligne à votre URL publique. Ouvrez l’URL publiée dans un nouvel onglet et envoyez une réponse de test pour vérifier que la page en ligne l’enregistre avant de la partager.

    <Frame>
      <img src="/assets/mirror/landing-publish-e3b901a79440.png" alt="Coin supérieur droit du workspace Replit agrandi 2× : le bouton gris Republish dans l’en-tête est encadré en orange et son menu contextuel est ouvert dessous. Il affiche « Raouf a publié il y a environ 18 heures », Visibility: Public, ainsi que les boutons Review security et Republish." width="2142" height="1072" data-path="images/fitstart/launch-landing-page/landing-publish.png" />
    </Frame>
  </Step>
</Steps>
