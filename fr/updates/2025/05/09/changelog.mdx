export const YouTubeEmbed = ({videoId, title = "Vidéo YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<div id="improved-image-handling-for-broken-unsplash-urls">
  ### Gestion améliorée des images pour les URL Unsplash cassées
</div>

Dites adieu aux images libres de droits qui ne s’affichent pas dans vos projets !

Vous ne le saviez peut-être pas, mais Agent ajoute déjà des images libres de droits provenant d’Unsplash à vos projets. Il arrive toutefois que les URL ne fonctionnent plus. La raison ? Les LLM excellent dans l’invention d’URL d’images libres de droits.

Nous avons ajouté une fonctionnalité qui remplace automatiquement les images libres de droits invalides par des alternatives fonctionnelles, pour que vos applications conservent toujours une apparence professionnelle.
