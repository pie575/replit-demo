export const YouTubeEmbed = ({videoId, title = "Видео YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

  ## Команды и Enterprise
</div>

<div id="section-6">
  ### Улучшение общего доступа к приложениям организации
</div>

Порядка в командном Workspace стало больше! Новые приложения, созданные внутри организации, больше не открываются автоматически всем участникам. Так в Workspace участников команды меньше лишнего, а контроль — точнее.

<div id="section-7">
  ### Улучшения страницы участников команды
</div>

Управлять командой стало проще благодаря новым улучшениям удобства работы на странице Teams Members. На вкладках участников, наблюдателей и приглашений теперь доступны сортировка, фильтрация и поиск. Именно эту возможность запрашивали клиенты Enterprise для эффективного управления большими командами.
