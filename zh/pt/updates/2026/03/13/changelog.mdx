export const YouTubeEmbed = ({videoId, title = "YouTube 视频", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

Agent 4 来了。Agent 为创意而生，是你的 AI 创意伙伴——只需自然语言，就能把你的想法变成应用、设计、幻灯片等等。无需编程。

借助 Agent 4，你可以：

* **想造什么就造什么**：创建 Web 应用、移动应用、数据看板、AI 工具等等
* **先设计，再构建**：在着手写代码之前，先在 [Design Canvas](/zh/pt/learn/projects-and-artifacts/canvas) 中探索原型
* **规划你的项目**：使用[规划模式](/zh/pt/core-concepts/agent/plan-mode)，在写下一行代码之前就理清思路
* **并行执行任务**：Agent 可同时处理多个任务，让你推进得更快
* **与同事协作**：邀请他人加入你的项目，一起同步构建
* **连接你的工具**：直接在聊天中从 BigQuery、Linear、Slack、Notion 等提取数据
