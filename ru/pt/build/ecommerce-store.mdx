export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "Скопировано!";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        Скопировать промпт
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Создать в Replit ↗
      </button>
    </div>;
};

Вы можете создать и запустить собственную витрину Shopify, просто описав магазин, который хотите открыть. Agent подготовит магазин Shopify, создаст фронтенд и добавит ваши товары — всё в рамках одной Conversation. Когда будете готовы продавать, достаточно один раз зайти в Shopify, чтобы заявить права на магазин и активировать платежи.

В этом руководстве вы создадите **Ember &amp; Oak**, магазин свечей ручной работы, — от промпта до опубликованной витрины.

  ## Чего вы достигнете
</div>

К концу этого руководства у вас будет:

* Витрина свечей с индивидуальным дизайном, связанная с реальным магазином Shopify.
* Товары и цены, созданные в Shopify и отображаемые на витрине.
* Работающая корзина, направляющая покупателей к оформлению заказа в Shopify.
* Чёткий путь к запуску: заявить права на магазин, активировать платежи и запустить его.

  ## Что вы узнаете
</div>

Вы научитесь:

* Описывать магазин, чтобы Agent рекомендовал Shopify и настроил его для вас.
* Подключать Shopify без необходимости управлять ключами API.
* Добавлять и публиковать товары в диалоге с Agent.
* Улучшать дизайн и процесс оформления заказа.
* Превращать магазин из проекта в разработке в действующую витрину, готовую к продажам.

<div id="conectar-o-shopify">
  ## Подключение Shopify
</div>

Shopify доступен всем разработчикам во всех приложениях. Подключить его можно прямо в Conversation. Начните с описания магазина, который хотите создать.

<Steps>
  <Step title="Опишите свой магазин">
    Скопируйте промпт и вставьте его в поле промпта на [replit.com](https://replit.com) или нажмите **Создать в Replit**, чтобы открыть Replit с уже заполненным промптом.

    <Frame>
      <img src="/assets/mirror/describe-your-store-b5d724daee99.png" alt="Поле промпта Replit с введённым промптом для магазина свечей Ember & Oak, готовое к началу разработки" width="1440" height="900" data-path="images/replitai/shopify/describe-your-store.png" />
    </Frame>

    <PromptActions prompt={CANDLE_STORE_PROMPT} campaign="docs-ecommerce-store" />

    Agent распознаёт, что вы продаёте физический товар, и рекомендует Shopify в качестве бэкенда для вашей витрины.

    <Frame>
      <img src="/assets/mirror/connect-shopify-store-7f36389f976f.png" alt="Agent Replit переключается на навык Shopify для магазина свечей Ember & Oak и показывает карточку «Connect to Shopify» с кнопкой Connect Shopify" width="3452" height="1984" data-path="images/replitai/shopify/connect-shopify-store.png" />
    </Frame>
  </Step>

  <Step title="Подтвердите создание магазина Shopify">
    Agent предложит создать магазин Shopify для вашего приложения. Подтвердите создание, и Agent подготовит и подключит магазин для разработки Replit. Дополнительная настройка не требуется.
    Ваш адрес электронной почты будет передан Shopify, чтобы позже вы могли заявить права на магазин.

    Изначально магазин защищён паролем и не может принимать реальные платежи, пока вы не запустите его, поэтому вы можете свободно разрабатывать без риска.
  </Step>
</Steps>

<Note>
  Каждый магазин Shopify связан только с одним приложением. Если позже вы создадите второй
  магазин, Agent подготовит новый магазин Shopify для этого приложения. Подробнее о
  коннекторе см. в разделе [Подключение Shopify](/ru/pt/features/integrations/shopify).
</Note>

<div id="construir-a-vitrine">
  ## Создание витрины
</div>

Когда Shopify подключён, Agent спроектирует и создаст витрину, интегрированную с вашим магазином.

<Steps>
  <Step title="Доверьте Agent создание фронтенда">
    Agent создаёт витрину по вашему промпт: hero-секцию, сетку товаров, корзину и разделы поддержки. По завершении правая панель переключается на **Preview**, где вы можете прокручивать страницу и просматривать дизайн.

    <Frame>
      <img src="/assets/mirror/storefront-preview-f5ecb9011a8c.png" alt="Витрина Ember & Oak на панели Preview в Replit рядом с чатом Agent: hero-секция со свечами и сетка товаров" width="3452" height="1978" data-path="images/replitai/shopify/storefront-preview.png" />
    </Frame>
  </Step>

  <Step title="Добавьте первые товары">
    У нового магазина пустой каталог. Попросите Agent добавить товар — он создаст его в Shopify и опубликует на витрине.

    <AiPrompt>
      Добавь свечу из лавандового соевого воска за $18 с коротким уютным описанием, затем опубликуй её на витрине.
    </AiPrompt>

    Товар будет создан в вашем магазине Shopify и появится в сетке товаров витрины.

    <Frame>
      <img src="/assets/mirror/add-product-8b098e606340.png" alt="Чат Agent добавляет свечу из лавандового соевого воска, а новый товар появляется в сетке товаров витрины Ember & Oak" width="3452" height="1978" data-path="images/replitai/shopify/add-product.png" />
    </Frame>
  </Step>

  <Step title="Заполните каталог">
    Продолжайте добавлять товары, описывая их. Agent создаёт каждый товар в Shopify с указанием цены, описанием и изображением.

    <AiPrompt>
      Добавь ещё три свечи: Cedar &amp; Sage за $20, Vanilla Bean за $16 и Sea Salt за $22. Добавь каждой короткое описание и опубликуй их.
    </AiPrompt>
  </Step>
</Steps>

<Tip>
  **Развивайте магазин:** попросите Agent объединить товары в коллекции («Создай
  праздничную коллекцию из трёх моих самых продаваемых свечей»), добавить варианты
  товаров, например размер или аромат, или отслеживать запасы, чтобы
  витрина показывала товары в наличии. Shopify остаётся системой учёта для
  всего этого.
</Tip>

<div id="iterar-no-design-e-no-checkout">
  ## Доработка дизайна и оформления заказа
</div>

Любую часть витрины можно редактировать через Conversation. Доработайте внешний вид, а затем проверьте весь путь покупки от начала до конца.

<Steps>
  <Step title="Доработайте дизайн">
    Опишите желаемые изменения, и Agent обновит витрину.

    <AiPrompt>
      Сделайте hero-секцию более редакционной: увеличьте заголовок, добавьте больше свободного пространства и используйте более мягкий янтарный акцент для кнопок.
    </AiPrompt>
  </Step>

  <Step title="Проверьте корзину и оформление заказа">
    В Preview добавьте свечу в корзину и перейдите к оформлению заказа. Корзина перенаправит покупателей на страницу оформления заказа, размещённую в Shopify. Во время разработки ваш магазин защищён паролем, поэтому в Preview при оформлении заказа может отображаться страница ввода пароля — это нормально и не означает, что что-то сломано.
  </Step>
</Steps>

<Tip>
  **Развивайте дальше:** попросите Agent добавить страницу товара,
  drawer корзины, поиск или раздел рекомендуемых товаров. Поскольку
  витрина — это полноценный проект Replit, вы можете расширять её всем, что
  сможете описать: блогом, списком желаний или персонализированными
  страницами кампаний. Shopify при этом будет отвечать за оформление и обработку заказов.
</Tip>

<div id="publicar">
  ## Публикация
</div>

Запуск магазина выполняется в Shopify. Попросите Agent запустить процесс выхода в эфир, а затем выполните действия, за которые отвечает владелец магазина.

<Steps>
  <Step title="Оформите права на магазин в Shopify">
    Оформление прав выполняется в Shopify — это единственный шаг за пределами
    Replit. Поскольку при создании магазина ваш адрес электронной почты был передан Shopify,
    вы можете оформить права на магазин по ссылке от Replit или из письма,
    отправленного Shopify.

    Shopify отправляет письмо с **Приглашением владельца магазина**,
    в котором приведены дальнейшие действия. Откройте его и выберите **Начать**.

    <Frame>
      <img src="/assets/mirror/claim-store-email-a0d108f2191c.png" alt="Письмо Shopify «Приглашение владельца магазина» с предложением стать владельцем магазина и кнопкой «Начать»" width="856" height="1024" data-path="images/replitai/shopify/claim-store-email.png" />
    </Frame>

    Войдите в существующую учётную запись Shopify или создайте новую, чтобы принять
    приглашение.

    <Frame>
      <img src="/assets/mirror/claim-store-create-account-99d2e132da19.png" alt="Экран Shopify «Создать учётную запись» для принятия приглашения с вариантами входа по электронной почте, через Google, Apple и Facebook" width="3452" height="1978" data-path="images/replitai/shopify/claim-store-create-account.png" />
    </Frame>

    Выберите **Оформить права на магазин**, чтобы авторизовать приложение Replit в своём магазине и
    начать бесплатный пробный период. После этого вы получите доступ к панели управления
    Shopify в качестве основного владельца.

    <Frame>
      <img src="/assets/mirror/claim-store-4facb7c8b10c.png" alt="Экран Replit «Оформите права на магазин» с кнопкой «Оформить права на магазин» для начала бесплатного пробного периода Shopify" width="3452" height="1978" data-path="images/replitai/shopify/claim-store.png" />
    </Frame>
  </Step>

  <Step title="Выберите тариф и активируйте платежи">
    Выберите тариф Shopify и активируйте Shopify Payments или другого провайдера.
    Shopify собирает данные для подтверждения личности и банковские реквизиты, необходимые для выплат
    вам, — всё это происходит в Shopify, а не в Replit.
  </Step>

  <Step title="Настройте доставку, налоги и домен">
    Установите тарифы на доставку и проверьте налоги в Shopify, а при необходимости
    подключите пользовательский домен.
  </Step>

  <Step title="Запустите и опубликуйте">
    Удалите пароль витрины в Shopify, чтобы открыть магазин для покупателей. Вернувшись в
    Replit, нажмите **Опубликовать**, чтобы развернуть витрину по общедоступному URL-адресу.
  </Step>
</Steps>

<Note>
  Платежи, проверка личности, налоги, доставка и проверки на мошенничество
  остаются в Shopify. Полный список проверок перед запуском и сведения о коннекторе
  см. в разделе [Подключение
  Shopify](/ru/pt/features/integrations/shopify#go-live-on-shopify).
</Note>
