export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "已复制！";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        复制提示词
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        在 Replit 中构建 ↗
      </button>
    </div>;
};

页面从上到下应包含以下内容：

一个页眉栏，左侧是小型金色“RJ”标志，旁边是“RACHEL JONES — FITNESS COACH”字样，中间有菜单链接（关于、教练服务、课程、成功故事、资源），右侧有一个金色“预约免费咨询”按钮。

一个大型 hero 区域，展示一位强壮健美的女性特写照片，在黑色背景下由暖金色戏剧化光线照亮。左侧深色区域放置一个三行巨型标题：

一个最终行动号召面板——一张宽大的深色卡片，分为三部分。左侧：Rachel 的一张亲切肖像。中间：标题“你不需要更多动力。你需要正确的计划。”并配有金色“预约免费咨询”按钮。右侧：三个带图标的小型权益——“1:1 策略通话”、“个性化计划”、“开启你的蜕变”。

照片请直接生成——我想要一位健美的女性，在黑色背景下采用暖金色戏剧化灯光，营造浓厚的电影感。蜕变前/后的照片应为全身照，采用相同的暗调摄影棚风格。

<div id="refine-e-explore-alternativas-no-canvas">
  ## 在 Canvas 中优化并探索替代方案
</div>

Agent 完成初次构建后，打开 Canvas 提供视觉反馈，并让 Agent 探索不同方案，无需重写提示词。

<Steps>
  <Step title="构建首页">
    复制提示词并粘贴到 [replit.com](https://replit.com) 的提示词输入框中，或点击 **在 Replit 中构建**，在 Replit 中打开已预填提示词的项目。

    <PromptActions prompt={RACHEL_JONES_PROMPT} campaign="docs-landing-page" />
  </Step>

  <Step title="预览 Agent 构建的应用">
    Agent 完成后，右侧面板会切换到 **Preview**，您可以实时与首页交互。在开始优化前，滚动浏览主视觉区域，点击行动号召按钮，并检查文字和图片是否符合您的要求。

    <Frame>
      <img src="/assets/mirror/landing-preview-4855c4b25270.png" alt="健身教练 Rachel Jones 的首页显示在 Replit 的 Preview 面板中，旁边是 Agent 聊天窗口——深色编辑风格的主视觉区域，金色光线照亮的肖像上叠加着标题“更强壮的身体。自信的心态。势不可挡的你。”，并配有“与 Rachel 合作”和“观看 Rachel 的故事”按钮，右侧展示认证资质，以及写有“全球超过 500 位女性信赖”的五星横幅" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-preview.png" />
    </Frame>
  </Step>

  <Step title="打开 Canvas">
    选择 Preview 和 Tools 标签下方的 **Canvas** 按钮，将右侧面板从 Preview 切换到 Canvas。

    <Frame>
      <img src="/assets/mirror/landing-canvas-button-494c5b0076b1.png" alt="Replit 工作区中，Preview 和 Tools 标签下方的 Canvas 按钮已高亮显示" width="720" height="450" data-path="images/fitstart/launch-landing-page/landing-canvas-button.png" />
    </Frame>
  </Step>

  <Step title="在预览区域外添加注释">
    使用 Canvas 的**文本**和**绘图**工具，在设计稿周围的空白区域添加注释和箭头，然后将它们指向你希望 Agent 修改的具体元素。放置在 Canvas 上 (预览区域外) 的注释会附加到你的反馈中，供 Agent 连同设计一起阅读。

    每条注释只说明一项修改。简短、具体的注释 (&quot;缩小此按钮的宽度&quot;) 比开放式反馈更有效。

    <Frame>
      <img src="/assets/mirror/landing-canvas-annotate-text-8135bfe1e8e2.png" alt="一条 Canvas 注释，内容为“缩小此按钮的宽度”，放置在 Rachel Jones 首页右侧的 Canvas 空白边缘处，一支箭头指向页面标题中的金色“预约免费咨询”按钮" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-annotate-text.png" />
    </Frame>
  </Step>

  <Step title="选择批注和预览，然后发送给 Agent">
    将注释与预览 frame 一起框选，在**与所选项目 Chat**中输入简短的 prompt 并发送。Agent 会将注释和设计视为同一上下文来理解，从而知道每条注释对应哪个元素。

    让 Chat 消息保持概括性——例如 &quot;应用注释中的更改。&quot; 具体细节交由注释说明。

    <Frame>
      <img src="/assets/mirror/landing-canvas-send-24b771bf29bf.png" alt="Rachel Jones 的首页与两条注释在 Canvas 中被一同选中——指向“预约免费咨询”按钮的“减小此按钮的宽度”，以及指向首屏主视觉照片的“为 Rachel 的图片添加滑动动画”——“与所选项目 Chat”输入框中已输入一条 prompt" width="1726" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-send.png" />
    </Frame>
  </Step>

  <Step title="探索不同的设计方向">
    在 Canvas 中点击设计，然后选择其旁边的建议 (例如 **探索不同方案**) ，让 Agent 生成其他视觉方向。

    <Frame>
      <img src="/assets/mirror/landing-canvas-reimagine-9e60c24f342f.png" alt="原始设计旁以 Design frame 形式生成的三个备选首页设计方向" width="1726" height="994" data-path="images/fitstart/launch-landing-page/landing-canvas-reimagine.png" />
    </Frame>
  </Step>

  <Step title="审阅备选方案并迭代优化">
    等待 Agent 完成生成。**探索不同方案**会用新变体更新同一产物——可通过模型图上方的产物下拉菜单在它们之间切换。**探索不同风格**会生成独立的视觉模型图，并显示在 Canvas 中原始模型图的旁边。

    变体生成后，选择一个方向并继续迭代。在选定的 frame 中添加另一条注释，将其与设计稿一起框选，然后发送新的请求——例如，&quot;应用注释中提出的更改。&quot; Agent 会将变体的上下文和你的新注释合并为一个 prompt。

    <Frame>
      <img src="/assets/mirror/landing-canvas-zoomed-4d1585d33fe6.png" alt="Agent 生成三个风格变体（温暖的编辑风、静谧的临床风、鲜艳的波普风）后，Canvas 中 Rachel Jones 的主页——原始 frame 已选中，并添加了两条新注释（'减小此按钮的宽度'和'为 Rachel 的图片添加滑动动画'），Chat 中显示“应用注释中提出的更改”" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-zoomed.png" />
    </Frame>
  </Step>

  <Step title="发布您的作品">
    当页面能在 Preview 中正常运行后，点击 workspace 右上角的 **Publish** (后续部署时为 **Republish**) ，将最新版本发布到公开 URL。请在新标签页中打开已发布的 URL，并发送一条测试回复，确认线上页面能够接收到该回复后再分享。

    <Frame>
      <img src="/assets/mirror/landing-publish-e3b901a79440.png" alt="放大 2 倍的 Replit workspace 右上角——标题栏中的灰色 Republish 按钮以橙色边框突出显示，下方弹出的菜单显示“Raouf 在约 18 小时前发布”、可见性：公开，以及 Review security 和 Republish 操作按钮" width="2142" height="1072" data-path="images/fitstart/launch-landing-page/landing-publish.png" />
    </Frame>
  </Step>
</Steps>
