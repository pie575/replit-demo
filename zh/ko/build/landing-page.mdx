export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "已复制！";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        复制提示词
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        在 Replit 上构建 ↗
      </button>
    </div>;
};

<div id="canvas">
  ## 在 Canvas 中优化并探索替代方案
</div>

Agent 完成首次构建后，打开 Canvas，通过视觉反馈引导 Agent 探索不同方案，无需重新编写提示词。

<Steps>
  <Step title="构建着陆页">
    复制提示词并将其粘贴到 [replit.com](https://replit.com) 的提示词输入框中，或点击 **Build on Replit**，在已预填提示词的状态下打开 Replit。

    <PromptActions prompt={RACHEL_JONES_PROMPT} />
  </Step>

  <Step title="Agent 构建的应用预览">
    Agent 完成后，右侧窗格会切换到 **Preview**，你可以实时与着陆页交互。滚动浏览主视觉区域，点击行动号召按钮，并在开始优化前确认文案和图片是否符合你的要求。

    <Frame>
      <img src="/assets/mirror/landing-preview-4855c4b25270.png" alt="Agent 聊天旁的 Replit Preview 窗口中渲染的 Rachel Jones 健身教练着陆页——深色编辑风格的主视觉区域展示一张金色调肖像，标题为“更强壮的身体。更自信的心态。势不可挡的你。”，包含“加入 Rachel”和“观看 Rachel 的故事”按钮，右侧显示认证证书，以及“深受全球 500 多名女性信赖”的五星评价条" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-preview.png" />
    </Frame>
  </Step>

  <Step title="打开 Canvas">
    选择 Preview 和 Tools 选项卡下的 **Canvas** 按钮，将右侧窗格从 Preview 切换到 Canvas。

    <Frame>
      <img src="/assets/mirror/landing-canvas-button-494c5b0076b1.png" alt="Replit 工作区中 Preview 和 Tools 选项卡下突出显示的 Canvas 按钮" width="720" height="450" data-path="images/fitstart/launch-landing-page/landing-canvas-button.png" />
    </Frame>
  </Step>

  <Step title="在预览区域外添加批注">
    使用 Canvas **Text** 和 **Draw** 工具，在模型周围的空白区域添加备注和箭头，指明你希望 Agent 修改的具体元素。预览区域之外——放置在画布上的注释会与反馈保持关联，以便 Agent 结合设计一并读取。

    每条注释应聚焦于一项修改。简短、具体的备注 (&quot;缩小此按钮的宽度&quot;) 比笼统的反馈更有效。

    <Frame>
      <img src="/assets/mirror/landing-canvas-annotate-text-8135bfe1e8e2.png" alt="Canvas 注释“缩小此按钮的宽度”，放置在 Rachel Jones 着陆页右侧的空白画布边距中，并有一个箭头指向页面标题中的金色“Book a Free Consultation”按钮" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-annotate-text.png" />
    </Frame>
  </Step>

  <Step title="选择注释和预览内容并发送给 Agent">
    框选注释和预览 frame，然后在 **Chat about selected items** 中输入并发送简短的提示词。Agent 会将注释和设计作为同一上下文理解，因此知道每条备注指向哪个元素。

    聊天消息请保持简洁通用，例如“根据注释应用更改。”具体要求请写在注释中。

    <Frame>
      <img src="/assets/mirror/landing-canvas-send-24b771bf29bf.png" alt="Canvas 中选中了 Rachel Jones 着陆页及两条注释——指向“Book a Free Consultation”按钮的“缩小此按钮的宽度”，以及指向主视觉照片的“为 Rachel 的图片添加滑入动画”——Chat about selected items 输入框中已输入聊天提示词" width="1726" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-send.png" />
    </Frame>
  </Step>

  <Step title="探索不同的方向">
    在 Canvas 中点击设计，然后选择旁边的建议 (如 **探索不同方案**) ，让 Agent 提供不同的视觉方向。

    <Frame>
      <img src="/assets/mirror/landing-canvas-reimagine-9e60c24f342f.png" alt="Canvas 上与原始设计并列的三个由 Design 帧生成的备选着陆页方案" width="1726" height="994" data-path="images/fitstart/launch-landing-page/landing-canvas-reimagine.png" />
    </Frame>
  </Step>

  <Step title="评估备选方案并迭代">
    等待 Agent 完成生成。**探索不同方案**会通过新变体更新同一工件 — 可在模型上方的工件下拉菜单中切换。**探索不同风格**会生成独立的视觉模型，并显示在 Canvas 中原始版本旁边。

    变体出现后，选择一个方向并继续迭代。为选中的画框添加另一条注释，框选设计，然后发送新的请求 — 例如：&quot;应用注释中要求的更改。&quot;Agent 会将变体上下文和新注释合并为一条提示词。

    <Frame>
      <img src="/assets/mirror/landing-canvas-zoomed-4d1585d33fe6.png" alt="Agent 生成三种风格变体（Editorial Warmth、Quiet Clinical、Vibrant Pop）后 Canvas 中的 Rachel Jones 着陆页 — 原始画框已被选中，包含两条新注释（“缩小此按钮的宽度”和“为 Rachel 的图片添加滑入动画”），聊天中已输入“应用注释中要求的更改。”" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-zoomed.png" />
    </Frame>
  </Step>

  <Step title="发布作品">
    当页面在 Preview 中正常运行后，点击工作区右上角的 **Publish** (后续部署时则点击 **重新发布**) ，将最新版本发布到公开 URL。分享前，请在新标签页中打开已发布的 URL，并提交一条测试回复，确认已上线的页面能够收集回复。

    <Frame>
      <img src="/assets/mirror/landing-publish-e3b901a79440.png" alt="Replit 工作区右上角放大 2 倍——页眉中的灰色 Republish 按钮以橙色边框突出显示，下方打开了一个弹出框，显示“Raouf 于约 18 小时前发布”、公开状态：Public、安全审核和 Republish 操作按钮" width="2142" height="1072" data-path="images/fitstart/launch-landing-page/landing-publish.png" />
    </Frame>
  </Step>
</Steps>
