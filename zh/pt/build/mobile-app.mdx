    if (typeof window === "undefined") {
      return null;
    }
    if (!window.__replitDocsAgentInputTypewriterLoader) {
      window.__replitDocsAgentInputTypewriterLoader = {};
    }
    return window.__replitDocsAgentInputTypewriterLoader;
  };
  const ensureAgentInputTypewriterCore = () => {
    if (typeof window === "undefined" || typeof document === "undefined") {
      return Promise.resolve(null);
    }
    if (window.Typewriter) {
      return Promise.resolve(window.Typewriter);
    }
    const loader = getAgentInputTypewriterLoader();
    if (loader.promise) {
      return loader.promise;
    }
    loader.promise = new Promise((resolve, reject) => {
      const handleLoad = () => {
        if (window.Typewriter) {
          resolve(window.Typewriter);
          return;
        }
        loader.promise = null;
        reject(new Error("Typewriter core loaded without a global Typewriter export."));
      };
      const handleError = () => {
        loader.promise = null;
        reject(new Error("Failed to load typewriter core."));
      };
      const existingScript = document.getElementById(AGENT_INPUT_TYPEWRITER_SCRIPT_ID);
      if (existingScript) {
        existingScript.addEventListener("load", handleLoad, {
          once: true
        });
        existingScript.addEventListener("error", handleError, {
          once: true

        return;

      script.addEventListener("load", handleLoad, {
        once: true

      script.addEventListener("error", handleError, {
        once: true
      });
      document.head.appendChild(script);
    });
    return loader.promise;

    startButton.style.cursor = hasText ? "pointer" : "not-allowed";
  };
  const stopAgentInputTypewriter = (textarea, {placeholder, clearDemo = false} = {}) => {

      return;

    if (!record) {
      if (placeholder) {
        textarea.setAttribute("placeholder", placeholder);
      }
      return;

    if (record.startTimer) {
      window.clearTimeout(record.startTimer);
    }
    if (record.writeFrame) {
      window.cancelAnimationFrame(record.writeFrame);
    }
    if (record.typewriter?.stop) {
      record.typewriter.stop();
    }

    textarea.removeEventListener("input", record.handleInput);

<div id="defina-o-fluxo-mobile">
  ## 定义移动端流程
</div>

在开始构建之前，先确定用户需要在手机上完成哪些操作。

对于 Pace Mobile：

* **目标用户**：希望快速记录跑步的跑者。
* **核心流程**：选择跑步类型 → 输入距离和时间 → 保存跑步记录 → 查看今日摘要。
* **移动端限制**：主要操作应便于用拇指触及。
* **首个版本**：不包含账户、社交动态或高级分析功能。

这样能让首个移动端功能范围足够小，便于测试。

有针对性的反馈比笼统的意见更便于改进。如果反馈指出了明确问题，请 Agent 进行最小范围的修复，然后再次在手机上测试 App。

<div id="publique-na-app-store">
  ## 发布到 App Store
</div>

在模拟器和手机上的 Expo Go 中测试 Pace Mobile 后，您可以将同一款 App 提交至 TestFlight 和 App Store。此流程要求您加入 **Apple Developer Program** 并启用双重认证。有关注册、个人账户与组织账户、TestFlight 与 App Store 审核以及外部测试的信息，请参阅[构建并发布移动 App](/zh/pt/build/mobile-app)。

<Steps>
  <Step title="打开“部署”选项卡，然后开始将应用部署到 App Store">

    </Frame>
  </Step>

  <Step title="选择您的 Expo 项目">

    <Frame>
      <img src="/assets/mirror/app-store-03-choose-expo-project-4707fafffafe.png" alt="Launch 对话框中“选择 Expo 项目”步骤处于激活状态——项目名称和项目 slug 字段已预填为 Pace，包含“继续”按钮和“使用现有项目”选项" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-03-choose-expo-project.png" />

</Note>

<div id="voc-terminou-quando">
  ## 完成标准
</div>

* 应用能在 iOS 模拟器或 Android 仿真器中打开。
* 应用能通过 Expo Go 在你的手机上打开。
* 你能完整走通主要的移动端流程。
* 文字、按钮和导航在手机上用起来体验舒适。
* 你已发布或分享了可供测试的移动版本。
* 至少有一个人能试用并提供反馈。
* (可选) 你已通过 App Store 流程将应用提交到 TestFlight。

<div id="precisa-de-mais-ajuda">
  ## 需要更多帮助？
</div>

* **Agent 构建的是 Web App：**重新开始，并在构建前选择 **移动 App** 作为 App 类型。
* **Expo Go 预览似乎不是最新版本：**摇晃手机，打开开发者菜单，然后选择 **刷新**。
* **触控目标似乎太小：**请 Agent 增大触控目标，并将主要操作置于拇指可及范围内。
* **导航似乎过于拥挤：**请 Agent 将 App 精简为主流程所需的最少页面。
* **模拟器可以运行，但手机无法运行：**请在真实设备上测试硬件特有的行为，并参阅[原生移动 App](/zh/pt/features/artifact-types/building-mobile-apps)。
* **部署失败：**请参阅[移动 App 故障排除](/zh/pt/features/troubleshooting/mobile-app)。
