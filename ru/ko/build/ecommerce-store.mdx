export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "Скопировано!";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        Копировать запрос
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Создать в Replit ↗
      </button>
    </div>;
};

Просто опишите нужный вам магазин — и вы сможете разработать и запустить собственную витрину Shopify. Agent в одном диалоге подготовит магазин Shopify, создаст на его основе фронтенд и добавит товары. Когда будете готовы к продажам, просто перейдите в Shopify, передайте права собственности на магазин и активируйте платежи.

В этом руководстве вы создадите небольшой магазин свечей **Ember &amp; Oak** — от промпта до опубликованной витрины.

<div id="shopify">
  ## Подключение Shopify
</div>

Shopify доступен всем разработчикам приложений и подключается прямо в чате. Начните с описания магазина, который хотите создать.

<Steps>
  <Step title="Опишите магазин">
    Скопируйте промпт и вставьте его в поле ввода на [replit.com](https://replit.com) или нажмите **Build on Replit**, чтобы открыть Replit с уже введённым промптом.

    <Frame>
      <img src="/assets/mirror/describe-your-store-b5d724daee99.png" alt="Поле ввода на главной странице Replit с промптом для магазина свечей Ember & Oak, готовое к началу сборки" width="1440" height="900" data-path="images/replitai/shopify/describe-your-store.png" />
    </Frame>

    <PromptActions prompt={CANDLE_STORE_PROMPT} campaign="docs-ecommerce-store" />

    Agent распознает, что вы продаёте физические товары, и рекомендует Shopify в качестве бэкенда для витрины магазина.

    <Frame>
      <img src="/assets/mirror/connect-shopify-store-7f36389f976f.png" alt="Replit Agent переходит к навыку Shopify для магазина свечей Ember & Oak; отображается карточка «Подключение к Shopify» с кнопкой Connect Shopify" width="3452" height="1984" data-path="images/replitai/shopify/connect-shopify-store.png" />
    </Frame>
  </Step>

  <Step title="Подтвердите создание магазина Shopify">
    Agent предложит создать магазин Shopify для вашего приложения. После подтверждения Agent подготовит и подключит магазин для разработки, принадлежащий Replit. Дополнительная настройка не требуется.
    Ваш адрес электронной почты будет передан Shopify, чтобы позже вы могли передать себе права собственности на магазин.

    Изначально магазин защищён паролем и не может принимать реальные платежи до запуска, поэтому вы можете свободно разрабатывать без риска.
  </Step>
</Steps>

<Note>
  Каждый магазин Shopify подключается к одному приложению. Если позже вы создадите второй магазин, Agent подготовит новый магазин Shopify для этого приложения. Подробнее о коннекторе см. в разделе [Подключение Shopify](/ru/ko/features/integrations/shopify).
</Note>

<div id="section-3">
  ## Создание витрины
</div>

После подключения Shopify Agent спроектирует и создаст витрину на основе вашего магазина.

<Steps>
  <Step title="Попросите Agent создать фронтенд">
    Agent создаст витрину по вашему промпту: главный блок, сетку товаров, корзину и вспомогательные разделы. После завершения правая панель переключится на **Preview**, где вы сможете прокручивать страницу и выбирать элементы дизайна.

    <Frame>
      <img src="/assets/mirror/storefront-preview-f5ecb9011a8c.png" alt="Витрина Ember & Oak в окне Replit Preview рядом с чатом Agent: главный блок со свечами и сетка товаров" width="3452" height="1978" data-path="images/replitai/shopify/storefront-preview.png" />
    </Frame>
  </Step>

  <Step title="Добавьте первый товар">
    Новый магазин начинает работу с пустым каталогом. Попросите Agent добавить товар — он создаст его в Shopify и опубликует на витрине.

    <AiPrompt>
      Добавь свечу из соевого воска с лавандой за $18 с коротким уютным описанием, затем опубликуй её на витрине.
    </AiPrompt>

    Товар будет создан в магазине Shopify и появится в сетке товаров на витрине.

    <Frame>
      <img src="/assets/mirror/add-product-8b098e606340.png" alt="Чат Agent добавляет товар — свечу из соевого воска с лавандой; новый товар отображается в сетке товаров витрины Ember & Oak" width="3452" height="1978" data-path="images/replitai/shopify/add-product.png" />
    </Frame>
  </Step>

  <Step title="Заполните каталог">
    Продолжайте добавлять товары, описывая их. Agent создаст каждый товар в Shopify, указав цену, описание и изображение.

    <AiPrompt>
      Добавь ещё три свечи: Cedar &amp; Sage за $20, Vanilla Bean за $16 и Sea Salt за $22. Добавь каждой короткое описание и опубликуй их.
    </AiPrompt>
  </Step>
</Steps>

<Tip>
  **Подробнее:** Попросите Agent сгруппировать товары в коллекции («Создай коллекцию Holiday из трёх бестселлеров»), добавить варианты товара, например размер или аромат, или включить отслеживание запасов, чтобы на витрине отображалось наличие товаров. Shopify остаётся системой учёта для всего этого.
</Tip>

<div id="section-4">
  ## Доработка дизайна и оформления платежа
</div>

Любую часть витрины можно редактировать в диалоге. Доработав внешний вид, проверьте весь путь покупателя от начала до конца.

<Steps>
  <Step title="Доработайте дизайн">
    Опишите, что хотите изменить, и Agent обновит витрину.

    <AiPrompt>
      Сделай главный раздел более редакционным: увеличь заголовок, добавь больше свободного пространства и используй более мягкий янтарный акцент на кнопках.
    </AiPrompt>
  </Step>

  <Step title="Протестируйте корзину и оформление платежа">
    В Preview добавьте свечу в корзину и начните оформление платежа. Корзина перенаправит покупателя на размещённую Shopify страницу оформления платежа. Во время разработки магазин защищён паролем, поэтому в предпросмотре оформления платежа может отображаться страница ввода пароля — это нормально и не означает, что возникла проблема.
  </Step>
</Steps>

<Tip>
  **Подробнее:** Попросите Agent добавить страницу товара, выдвижную корзину, поиск или раздел рекомендаций. Поскольку витрина — это полноценный проект Replit, её можно расширить чем угодно, что вы сможете описать: блогом, списком желаний или пользовательской лендинг-страницей. Shopify обрабатывает платежи и заказы.
</Tip>

<div id="section-5">
  ## Запуск
</div>

Переход к работе в режиме реального времени выполняется в Shopify. Попросите Agent запустить процесс Go Live, затем выполните действия, необходимые со стороны продавца.

<Steps>
  <Step title="Передайте права собственности на магазин в Shopify">
    Передача прав собственности выполняется в Shopify — это единственный шаг за пределами Replit. При создании магазина ваш адрес электронной почты был передан Shopify, поэтому передать права собственности можно по ссылке от Replit или из письма Shopify.

    Shopify отправит письмо **Store owner invitation** с инструкциями. Откройте его и выберите **Get started**.

    <Frame>
      <img src="/assets/mirror/claim-store-email-a0d108f2191c.png" alt="Письмо Shopify «Store owner invitation» с приглашением стать владельцем магазина и кнопкой «Get started»" width="856" height="1024" data-path="images/replitai/shopify/claim-store-email.png" />
    </Frame>

    Войдите в существующую учётную запись Shopify или создайте новую, чтобы принять приглашение.

    <Frame>
      <img src="/assets/mirror/claim-store-create-account-99d2e132da19.png" alt="Экран Shopify «Создать учётную запись» для принятия приглашения с вариантами входа по электронной почте, через Google, Apple и Facebook" width="3452" height="1978" data-path="images/replitai/shopify/claim-store-create-account.png" />
    </Frame>

    Выберите **Claim store**, чтобы разрешить работу Replit App с магазином и начать бесплатный пробный период. После этого вы получите доступ к панели администратора Shopify в качестве основного владельца.

    <Frame>
      <img src="/assets/mirror/claim-store-4facb7c8b10c.png" alt="Экран Replit «Передача прав собственности на магазин» с кнопкой «Claim store» для начала бесплатного пробного периода Shopify" width="3452" height="1978" data-path="images/replitai/shopify/claim-store.png" />
    </Frame>
  </Step>

  <Step title="Выберите тариф и активируйте платежи">
    Выберите тариф Shopify и активируйте Shopify Payments или другого провайдера.
    Shopify соберёт сведения о личности и банковские данные, необходимые для платежей, — в Replit ничего делать не нужно.
  </Step>

  <Step title="Настройте доставку, налоги и домен">
    Настройте тарифы на доставку в Shopify, проверьте налоги и при необходимости подключите собственный домен.
  </Step>

  <Step title="Запустите и опубликуйте">
    Удалите пароль витрины в Shopify, чтобы открыть магазин для покупателей. Вернитесь в Replit и нажмите **Publish**, чтобы задеплоить витрину по общедоступному URL.
  </Step>
</Steps>

<Note>
  Платежи, проверка личности, налоги, доставка и проверка на мошенничество обрабатываются в Shopify. Полный контрольный список запуска и сведения о коннекторе см. в разделе [Подключение Shopify](/ru/ko/features/integrations/shopify#go-live-on-shopify).
</Note>
