export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "已复制！";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        复制提示词
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        在 Replit 上构建 ↗
      </button>
    </div>;
};

只需描述你想要的店铺，即可设计并上线自定义 Shopify 店面。Agent 会配置 Shopify 商店、为其构建前端，并在同一 Conversation 中添加你的产品。准备销售时，你只需前往 Shopify 一次，认领店铺并启用付款功能。

<div id="what-youll-achieve">
  ## 您将实现的目标
</div>

完成本指南后，您将拥有：

* 一个由真实 Shopify 商店支持的定制蜡烛店面。
* 在 Shopify 中创建并展示在店面上的产品和价格。
* 一个可将买家引导至 Shopify 结账流程的购物车。
* 一条清晰的上线路径：认领商店、启用付款并上线。

<div id="what-youll-learn">
  ## 你将学到什么
</div>

你将学习如何：

* 描述你的店铺，让 Agent 推荐 Shopify 并为你完成配置。
* 无需处理任何 API key 即可连接 Shopify。
* 通过与 Agent 对话添加并发布产品。
* 优化设计和买家购买流程。
* 将店铺从开发阶段上线为可销售的店面。

<div id="connect-shopify">
  ## 连接 Shopify
</div>

所有构建者都可以在任何 App 中使用 Shopify，并可直接在 Conversation 中完成连接。先描述你想要构建的商店。

<Steps>
  <Step title="描述你的商店">
    复制提示词，并将其粘贴到 [replit.com](https://replit.com) 的提示框中，或点击 **在 Replit 上构建**，在已预填该提示词的 Replit 中打开。

    <Frame>
      <img src="/assets/mirror/describe-your-store-b5d724daee99.png" alt="Replit 首页提示框中已输入 Ember & Oak 蜡烛商店提示词，准备开始构建" width="1440" height="900" data-path="images/replitai/shopify/describe-your-store.png" />
    </Frame>

    <PromptActions prompt={CANDLE_STORE_PROMPT} campaign="docs-ecommerce-store" />

    Agent 会识别出你销售的是实体商品，并推荐使用 Shopify 作为店面的后端。

    <Frame>
      <img src="/assets/mirror/connect-shopify-store-7f36389f976f.png" alt="Replit Agent 将 Ember & Oak 蜡烛商店交接给 Shopify skill，显示“连接到 Shopify”卡片及其“连接 Shopify”按钮" width="3452" height="1984" data-path="images/replitai/shopify/connect-shopify-store.png" />
    </Frame>
  </Step>

  <Step title="批准创建 Shopify 商店">
    Agent 会请求为你的 App 创建 Shopify 商店。批准后，Agent 会创建并连接一个由 Replit 拥有的开发商店，无需额外配置。
    系统会将你的电子邮箱分享给 Shopify，以便你日后认领该商店。

    商店初始受密码保护，在正式上线前无法接收真实付款，因此你可以放心自由构建。
  </Step>
</Steps>

<Note>
  每个 Shopify 商店仅限用于一个 App。如果你之后构建第二个商店，
  Agent 会为该 App 创建一个新的 Shopify 商店。有关完整的
  connector 详情，请参阅[连接 Shopify](/zh/features/integrations/shopify)。
</Note>

<div id="build-the-storefront">
  ## 构建店面
</div>

连接 Shopify 后，Agent 会根据你的商店设计并构建店面。

<Steps>
  <Step title="让 Agent 构建前端">
    Agent 会根据你的提示词生成店面，包括主视觉区、产品网格、购物车和其他内容区块。完成后，右侧窗格会切换到 **Preview**，你可以滚动页面并点击查看设计。

    <Frame>
      <img src="/assets/mirror/storefront-preview-f5ecb9011a8c.png" alt="Replit Preview 窗格中渲染的 Ember & Oak 店面位于 Agent chat 旁边，显示蜡烛主视觉区和产品网格" width="3452" height="1978" data-path="images/replitai/shopify/storefront-preview.png" />
    </Frame>
  </Step>

  <Step title="添加首批产品">
    新商店的产品目录一开始是空的。让 Agent 添加产品，它会在 Shopify 中创建该产品并将其发布到店面。

    <AiPrompt>
      添加一款售价 $18 的薰衣草大豆蜡烛，并附上简短温馨的描述，然后将其发布到店面。
    </AiPrompt>

    该产品会在你的 Shopify 商店中创建，并显示在店面的产品网格中。

    <Frame>
      <img src="/assets/mirror/add-product-8b098e606340.png" alt="Agent chat 正在添加薰衣草大豆蜡烛产品，新产品出现在 Ember & Oak 店面的产品网格中" width="3452" height="1978" data-path="images/replitai/shopify/add-product.png" />
    </Frame>
  </Step>

  <Step title="完善产品目录">
    继续描述产品来添加它们。Agent 会在 Shopify 中创建每款产品，并设置价格、描述和图片。

    <AiPrompt>
      再添加三款蜡烛：Cedar &amp; Sage 售价 $20、Vanilla Bean 售价 $16，以及 Sea Salt 售价 $22。为每款添加简短描述并发布。
    </AiPrompt>
  </Step>
</Steps>

<Tip>
  **进阶技巧：**让 Agent 将产品归入系列 (“为我最畅销的三款蜡烛创建一个
  节日系列”) 、添加尺寸或香味等产品变体，或跟踪库存，让店面显示哪些产品有
  现货。Shopify 始终是所有这些信息的权威来源。
</Tip>

<div id="iterate-on-the-design-and-checkout">
  ## 优化设计并测试结账流程
</div>

你可以通过对话编辑店面的每个部分。优化外观后，再端到端检查买家的购买流程。

<Steps>
  <Step title="优化设计">
    描述你想要修改的内容，Agent 会更新店面。

    <AiPrompt>
      让主视觉区域更具杂志感——放大标题、增加留白，并为按钮采用更柔和的琥珀色点缀。
    </AiPrompt>
  </Step>

  <Step title="测试购物车和结账流程">
    在 Preview 中，将一支蜡烛加入购物车，然后开始结账流程。购物车会将买家带到 Shopify 托管的结账流程。开发期间，你的店面受密码保护，因此结账流程预览可能会显示密码页面——这是正常现象，并不表示出现了问题。
  </Step>
</Steps>

<Tip>
  **进阶：**请 Agent 添加商品详情页、购物车抽屉、
  搜索功能或精选商品专区。由于店面是一个真正的
  Replit project，你可以按自己的描述随意扩展它——例如博客、
  愿望清单或自定义落地页——而 Shopify 则负责处理结账流程和订单。
</Tip>

<div id="ship-it">
  ## 上线
</div>

上线操作在 Shopify 中完成。请让 Agent 启动上线流程，然后完成需要由商家操作的步骤。

<Steps>
  <Step title="在 Shopify 中认领商店">
    认领操作在 Shopify 中完成，也是唯一需要在
    Replit 之外完成的步骤。由于创建商店时已将你的电子邮箱地址提供给 Shopify，
    你可以通过 Replit 提供的链接或 Shopify
    发送的电子邮件认领商店。

    Shopify 会向你发送一封 **商店所有者邀请** 电子邮件，引导你完成
    后续步骤。打开邮件并选择 **开始使用**。

    <Frame>
      <img src="/assets/mirror/claim-store-email-a0d108f2191c.png" alt="Shopify 的“商店所有者邀请”电子邮件，邀请你成为商店所有者，并附有“开始使用”按钮" width="856" height="1024" data-path="images/replitai/shopify/claim-store-email.png" />
    </Frame>

    使用现有的 Shopify 账户登录，或创建账户以接受
    邀请。

    <Frame>
      <img src="/assets/mirror/claim-store-create-account-99d2e132da19.png" alt="用于接受邀请的 Shopify“创建账户”页面，提供电子邮箱、Google、Apple 和 Facebook 登录选项" width="3452" height="1978" data-path="images/replitai/shopify/claim-store-create-account.png" />
    </Frame>

    选择 **认领商店**，以授权 Replit App 访问你的商店并开始
    免费试用。随后，你将以主要所有者身份进入 Shopify 管理后台。

    <Frame>
      <img src="/assets/mirror/claim-store-4facb7c8b10c.png" alt="Replit 的“认领你的商店”页面，包含用于开始免费 Shopify 试用的“认领商店”按钮" width="3452" height="1978" data-path="images/replitai/shopify/claim-store.png" />
    </Frame>
  </Step>

  <Step title="选择套餐并启用收款">
    选择 Shopify 套餐，并启用 Shopify Payments 或其他收款服务提供商。
    Shopify 会收集向你付款所需的身份和银行信息——这些操作均不会在 Replit
    中进行。
  </Step>

  <Step title="配置配送、税费和域名">
    在 Shopify 中设置配送费率并检查税费，还可以选择连接
    自定义域名。
  </Step>

  <Step title="上线并发布">
    在 Shopify 中移除店面密码，向买家开放商店。返回
    Replit 后，点击 **Publish**，将店面部署到公开 URL。
  </Step>
</Steps>

<Note>
  付款、身份验证、税费、配送和欺诈检查均由
  Shopify 处理。有关完整的上线清单和 connector 详情，请参阅 [连接
  Shopify](/zh/features/integrations/shopify#go-live-on-shopify)。
</Note>
