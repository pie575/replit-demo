export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "Copié !";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        Copier le prompt
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Créer sur Replit ↗
      </button>
    </div>;
};

Vous pouvez concevoir et lancer une boutique en ligne Shopify personnalisée en décrivant simplement celle que vous souhaitez. Agent crée une boutique Shopify, développe son interface et ajoute vos produits, le tout depuis une seule conversation. Lorsque vous êtes prêt à vendre, il vous suffit de vous rendre une seule fois sur Shopify pour revendiquer la boutique et activer les paiements.

Dans ce guide, vous créerez **Ember &amp; Oak**, une boutique de bougies artisanales, d’un prompt à la publication de votre boutique en ligne.

<div id="what-youll-achieve">
  ## Ce que vous allez réaliser
</div>

À la fin de ce guide, vous disposerez de :

* Une boutique de bougies au design personnalisé, reliée à une véritable boutique Shopify.
* Des produits et des prix créés dans Shopify et affichés sur la boutique en ligne.
* Un panier fonctionnel qui redirige les acheteurs vers le checkout Shopify.
* Une marche à suivre claire pour mettre votre boutique en ligne : la revendiquer, activer les paiements et la lancer.

<div id="what-youll-learn">
  ## Ce que vous apprendrez
</div>

Vous apprendrez à :

* Décrire une boutique afin qu’Agent vous recommande Shopify et le configure pour vous.
* Connecter Shopify sans avoir à gérer de clés API.
* Ajouter et publier des produits en échangeant avec Agent.
* Affiner la conception et le parcours d’achat.
* Faire passer la boutique du développement à une boutique en ligne opérationnelle.

<div id="connect-shopify">
  ## Connecter Shopify
</div>

Shopify est disponible pour tous les créateurs et toutes les apps, et vous pouvez le connecter directement depuis la conversation. Commencez par décrire la boutique que vous souhaitez créer.

<Steps>
  <Step title="Décrivez votre boutique">
    Copiez le prompt et collez-le dans le champ de prompt sur [replit.com](https://replit.com), ou cliquez sur **Créer sur Replit** pour ouvrir Replit avec le prompt prérempli.

    <Frame>
      <img src="/assets/mirror/describe-your-store-b5d724daee99.png" alt="Le champ de prompt de la page d’accueil de Replit avec le prompt de la boutique de bougies Ember & Oak saisi, prêt à lancer la création" width="1440" height="900" data-path="images/replitai/shopify/describe-your-store.png" />
    </Frame>

    <PromptActions prompt={CANDLE_STORE_PROMPT} campaign="docs-ecommerce-store" />

    Agent reconnaît que vous vendez un produit physique et recommande Shopify comme backend de votre boutique en ligne.

    <Frame>
      <img src="/assets/mirror/connect-shopify-store-7f36389f976f.png" alt="Replit Agent passe à la compétence Shopify pour la boutique de bougies Ember & Oak, avec une carte « Connect to Shopify » et un bouton Connect Shopify" width="3452" height="1984" data-path="images/replitai/shopify/connect-shopify-store.png" />
    </Frame>
  </Step>

  <Step title="Approuvez la création d’une boutique Shopify">
    Agent vous demande de créer une boutique Shopify pour votre app. Approuvez la demande, et Agent provisionne une boutique de développement appartenant à Replit et la connecte. Aucune configuration supplémentaire n’est nécessaire.
    Votre adresse e-mail est partagée avec Shopify afin que vous puissiez récupérer la boutique ultérieurement.

    La boutique est initialement protégée par mot de passe et ne peut pas accepter de paiements réels tant que vous ne l’avez pas mise en ligne. Vous pouvez donc développer librement, sans risque.
  </Step>
</Steps>

<Note>
  Chaque boutique Shopify est associée à une seule app. Si vous créez une deuxième boutique
  ultérieurement, Agent provisionne une nouvelle boutique Shopify pour cette app. Pour en savoir plus sur le
  connecteur, consultez [Connect Shopify](/fr/features/integrations/shopify).
</Note>

<div id="build-the-storefront">
  ## Créez votre boutique en ligne
</div>

Une fois Shopify connecté, Agent conçoit et crée votre boutique en ligne à partir de votre boutique Shopify.

<Steps>
  <Step title="Laissez Agent créer l’interface">
    Agent génère la boutique en ligne à partir de votre prompt — la section héro, la grille de produits, le panier et les sections complémentaires. Une fois cette étape terminée, le volet droit affiche **Preview**, ce qui vous permet de faire défiler la page et d’explorer le design.

    <Frame>
      <img src="/assets/mirror/storefront-preview-f5ecb9011a8c.png" alt="La boutique en ligne Ember & Oak affichée dans le volet Preview de Replit à côté du Chat Agent, montrant la section principale des bougies et la grille de produits" width="3452" height="1978" data-path="images/replitai/shopify/storefront-preview.png" />
    </Frame>
  </Step>

  <Step title="Ajoutez vos premiers produits">
    Une nouvelle boutique démarre avec un catalogue vide. Demandez à Agent d’ajouter un produit : il le crée dans Shopify et le publie dans la boutique en ligne.

    <AiPrompt>
      Ajoute une bougie à la cire de soja parfumée à la lavande à $18, avec une courte description chaleureuse, puis publie-la dans la boutique en ligne.
    </AiPrompt>

    Le produit est créé dans votre boutique Shopify et apparaît dans la grille de produits de la boutique en ligne.

    <Frame>
      <img src="/assets/mirror/add-product-8b098e606340.png" alt="Chat Agent ajoutant une bougie à la cire de soja parfumée à la lavande, le nouveau produit apparaissant dans la grille de produits de la boutique en ligne Ember & Oak" width="3452" height="1978" data-path="images/replitai/shopify/add-product.png" />
    </Frame>
  </Step>

  <Step title="Complétez le catalogue">
    Continuez à ajouter des produits en les décrivant. Agent crée chacun dans Shopify avec son prix, sa description et son image.

    <AiPrompt>
      Ajoute trois autres bougies : Cedar &amp; Sage à $20, Vanilla Bean à $16 et Sea Salt à $22. Donne à chacune une courte description et publie-les.
    </AiPrompt>
  </Step>
</Steps>

<Tip>
  **Pour aller plus loin :** demandez à Agent de regrouper les produits en collections (« Crée une
  collection Holiday avec mes trois bougies les plus vendues »), d’ajouter des variantes de produits,
  comme la taille ou le parfum, ou de suivre les stocks afin que la boutique en ligne indique les produits
  disponibles. Shopify reste la source de référence pour tout cela.
</Tip>

<div id="iterate-on-the-design-and-checkout">
  ## Peaufiner le design et le checkout
</div>

Chaque élément de la boutique en ligne peut être modifié dans la Conversation. Peaufinez l’apparence, puis vérifiez l’ensemble du parcours d’achat.

<Steps>
  <Step title="Peaufiner le design">
    Décrivez ce que vous souhaitez modifier, et Agent met à jour la boutique en ligne.

    <AiPrompt>
      Donne à la section principale un style plus éditorial — un titre plus grand, davantage d’espace blanc et une touche d’ambre plus douce sur les boutons.
    </AiPrompt>
  </Step>

  <Step title="Tester le panier et le checkout">
    Dans Preview, ajoutez une bougie au panier et commencez le checkout. Le panier redirige les acheteurs vers le checkout hébergé par Shopify. Pendant le développement, votre boutique est protégée par mot de passe. L’aperçu du checkout peut donc afficher une page de mot de passe — c’est normal et cela ne signifie pas qu’un problème est survenu.
  </Step>
</Steps>

<Tip>
  **Pour aller plus loin :** demandez à Agent d’ajouter une page de détail de produit, un panier latéral,
  une fonction de recherche ou une section de produits mis en avant. Comme la boutique en ligne est un véritable
  projet Replit, vous pouvez l’enrichir de tout ce que vous pouvez décrire — un blog, une
  liste de souhaits ou des pages de destination personnalisées — tandis que Shopify gère le checkout et les commandes.
</Tip>

<div id="ship-it">
  ## Lancez votre boutique
</div>

La mise en ligne se fait sur Shopify. Demandez à Agent de lancer le processus de mise en ligne, puis effectuez les étapes à la charge du marchand.

<Steps>
  <Step title="Revendiquer votre boutique sur Shopify">
    La revendication se fait sur Shopify et constitue la seule étape qui se déroule en dehors de
    Replit. Votre adresse e-mail ayant été partagée avec Shopify lors de la
    création de la boutique, vous pouvez la revendiquer via le lien fourni par Replit ou l’e-mail envoyé par Shopify.

    Shopify vous envoie un e-mail **Store owner invitation** qui vous guide tout au long des
    étapes. Ouvrez-le et sélectionnez **Get started**.

    <Frame>
      <img src="/assets/mirror/claim-store-email-a0d108f2191c.png" alt="E-mail Shopify « Store owner invitation » vous invitant à devenir propriétaire de la boutique, avec un bouton « Get started »" width="856" height="1024" data-path="images/replitai/shopify/claim-store-email.png" />
    </Frame>

    Connectez-vous à votre compte Shopify existant ou créez-en un pour accepter l’invitation.

    <Frame>
      <img src="/assets/mirror/claim-store-create-account-99d2e132da19.png" alt="Écran Shopify « Create an account » pour accepter l’invitation, avec des options de connexion par e-mail, Google, Apple et Facebook" width="3452" height="1978" data-path="images/replitai/shopify/claim-store-create-account.png" />
    </Frame>

    Sélectionnez **Claim store** pour autoriser l’application Replit sur votre boutique et démarrer
    votre essai gratuit. Vous accéderez ensuite à l’interface d’administration Shopify en tant que propriétaire principal.

    <Frame>
      <img src="/assets/mirror/claim-store-4facb7c8b10c.png" alt="Écran Replit « Claim your store » avec un bouton « Claim store » pour démarrer un essai gratuit de Shopify" width="3452" height="1978" data-path="images/replitai/shopify/claim-store.png" />
    </Frame>
  </Step>

  <Step title="Choisir un plan et activer les paiements">
    Choisissez un plan Shopify et activez Shopify Payments ou un autre prestataire de paiement.
    Shopify collecte les informations d’identité et bancaires nécessaires pour vous verser vos paiements — rien
    de cela ne se fait dans Replit.
  </Step>

  <Step title="Configurer la livraison, les taxes et un domaine">
    Définissez vos tarifs de livraison et vérifiez les taxes dans Shopify, puis connectez éventuellement
    un nom de domaine personnalisé.
  </Step>

  <Step title="Lancer et publier">
    Supprimez le mot de passe de la boutique en ligne dans Shopify pour ouvrir la boutique aux acheteurs. De retour
    dans Replit, cliquez sur **Publier** pour déployer votre boutique en ligne à son URL publique.
  </Step>
</Steps>

<Note>
  Les paiements, la vérification d’identité, les taxes, la livraison et les contrôles antifraude restent sur
  Shopify. Pour consulter la liste complète de mise en ligne et les détails du connecteur, consultez [Connecter
  Shopify](/fr/features/integrations/shopify#go-live-on-shopify).
</Note>
