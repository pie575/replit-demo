export const YouTubeEmbed = ({videoId, title = "Видео на YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<YouTubeEmbed videoId="bAUVQfz_SIY" title="Introducing Replit Agent 4" />

Agent 4 уже здесь. Созданный для творчества, Agent — ваш креативный ИИ-партнёр: он превращает идеи в приложения, дизайны, презентации и многое другое, и всё это — с помощью обычного текста. Программировать не нужно.

С Agent 4 вы можете:

* **Создавать что угодно**: веб-приложения, мобильные приложения, дашборды с данными, инструменты с ИИ и многое другое
* **Проектировать до разработки**: изучать макеты в [Design Canvas](/ru/pt/learn/projects-and-artifacts/canvas), прежде чем переходить к коду
* **Планировать свой project**: использовать [режим планирования](/ru/pt/core-concepts/agent/plan-mode), чтобы продумать будущий продукт ещё до написания кода
* **Выполнять задачи параллельно**: Agent берёт на себя несколько задач одновременно, чтобы вы двигались быстрее
* **Работать вместе с коллегами**: приглашайте других в свой project и создавайте вместе
* **Подключать свои инструменты**: получайте данные из BigQuery, Linear, Slack, Notion и других сервисов прямо в чате
