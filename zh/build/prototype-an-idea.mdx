export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "已复制！";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        复制提示词
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        在 Replit 上构建 ↗
      </button>
    </div>;
};

<div id="build-in-parallel">
  ### 并行构建
</div>

当产品问题涉及方向，而不只是实现时，同时探索多个想法能帮助你比较不同方案，避免过早定夺。Replit Agent 支持并行任务——你可以让 Agent 同时处理多个原型、功能或实验，并排审阅结果。

<Frame>
  <img src="/assets/mirror/task-plan-sidebar-ecab81b06bb0.png" alt="线程视图，显示多个并行运行且带有状态指示器的 Agent 任务" width="3430" height="1986" data-path="images/replitai/task-plan-sidebar.png" />
</Frame>

有关任务如何排队、运行和完成，请参阅[任务系统](/zh/core-concepts/agent/task-system)。

当你不确定哪个方向更好时，可以让 Agent 创建两个并行任务。对于 Pace，你可以比较：

* **激励优先的原型**：连续打卡、成就和鼓励性建议。
* **数据分析优先的原型**：更深入的统计数据、周环比趋势和个人纪录。

<AiPrompt>
  为改进 Pace 产品原型创建两个并行任务。<br />
  任务 1 应探索一个激励优先的版本，包含连续打卡、成就和鼓励性建议。<br />
  任务 2 应探索一个数据分析优先的版本，包含更深入的统计数据、周环比趋势和个人纪录。<br />
  两个版本都要足够精简，以便今天就能测试。<br />
  对于每项任务，请定义用户流程、要构建的内容，以及我应如何评估结果。
</AiPrompt>

两个任务都完成后，使用同一个反馈问题进行比较：

> 哪个版本更能让你明天愿意再回来使用？为什么？

<div id="need-further-help">
  ## 还需要帮助？
</div>

* **原型功能太多：**请 Agent 将其精简为面向一个目标用户的单一流程。
* **目标受众不明确：**请 Agent 为一个目标用户重写欢迎部分。
* **模拟数据掩盖了风险：**请 Agent 列出哪些内容是真实的，哪些是模拟的。
* **反馈过于笼统：**给审阅者一项任务和一个问题。
* **Publishing 失败：**请参阅[排查 Publishing 问题](/zh/build/troubleshooting)。
