export const YouTubeEmbed = ({videoId, title = "Vidéo YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

Agent 4 est arrivé. Conçu pour la créativité, Agent est votre partenaire créatif basé sur l’IA : il transforme vos idées en apps, designs, diapositives et bien plus encore, le tout à partir du langage naturel. Sans programmation.

Avec Agent 4, vous pouvez :

* **Créer n’importe quoi** : créez des apps web, des apps mobiles, des tableaux de bord de données, des outils basés sur l’IA et bien plus encore
* **Concevoir avant de créer** : explorez des maquettes dans le [Design Canvas](/fr/pt/learn/projects-and-artifacts/canvas) avant de vous lancer dans le code
* **Planifier votre projet** : utilisez le [Mode Planification](/fr/pt/core-concepts/agent/plan-mode) pour planifier votre création avant d’écrire la moindre ligne de code
* **Exécuter des tâches en parallèle** : Agent gère plusieurs tâches simultanément pour vous permettre d’avancer plus vite
* **Collaborer avec des collègues** : invitez d’autres personnes à votre projet et créez ensemble en même temps
* **Connecter vos outils** : récupérez des données de BigQuery, Linear, Slack, Notion et bien plus encore, directement depuis le Chat
