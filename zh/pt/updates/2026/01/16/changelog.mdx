export const YouTubeEmbed = ({videoId, title = "YouTube 视频", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

  ### Replit 上的移动 App
</div>

<YouTubeEmbed videoId="YSeDWL4cDLs" title="Replit 上的移动 App" />

只需一句简单的 prompt，就能创建原生 iOS 应用。Agent 会构建完整的 React Native 应用，你还可以在过程中随时加入数据库、Stripe 支付或 OpenAI 集成等功能。用 Expo Go 扫描二维码即可预览应用，然后通过 Replit 的引导式流程提交到 App Store。审核与分发由 Apple 通过 App Store 完成。
