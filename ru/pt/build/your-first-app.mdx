export const FreeModePlanAvailability = `Free Mode доступен на тарифах Starter, Core и Pro. Лимиты Free Mode зависят от вашего тарифа, а в Core и Pro предусмотрен лимит, который обновляется каждые ${FreeModeAllowanceRefreshPeriod}.`;

export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "Скопировано!";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        Скопировать промпт
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Создать в Replit ↗
      </button>
    </div>;
};

export const GATHER_PROMPT = `Создайте адаптивное приложение для поиска мастер-классов под названием Gather.

Посетители должны иметь возможность просматривать предстоящие мастер-классы, открывать подробности мероприятий и отправлять RSVP, используя реалистичные демонстрационные данные.

Приложение считается готовым, если навигация работает, после отправки RSVP отображается понятное подтверждение, а раскладка корректно работает на мобильных устройствах и компьютерах.

Пока не добавляйте вход в систему, базу данных, платежи или внешние интеграции.`;

⏰ *Примерное время: 10 минут*

<div id="before-you-start">
  ## Перед началом
</div>

Вам понадобится [учётная запись Replit](https://replit.com). В этом руководстве используется **Free Mode**, который автоматически выбирает модель с помощью интеллектуальной маршрутизации. {FreeModePlanAvailability}

Цель — отработать повторяемый цикл составления промптов:

1. Опишите один полезный результат.
2. Добавьте наблюдаемые критерии успеха.
3. Укажите, что пока не нужно создавать.
4. Протестируйте результат в Preview.
5. Попросите об одном точечном улучшении.

<div id="build-the-first-version">
  ## Создайте первую версию
</div>

<Steps>
  <Step title="Начните работу над новым Website">
    Начните новый диалог и выберите **Website** в качестве артефакта. Его можно выбрать в меню **New** или через **Add attachment → Create something new → Website** в поле ввода промпта.

    Откройте переключатель режимов и выберите **Free**.

    <Tip>
      Вы также можете попросить Replit создать веб-сайт, просто описав его на естественном языке. Если сначала выбрать Website, будет явно указан нужный артефакт, а к промпту добавится метка Website.
    </Tip>
  </Step>

  <Step title="Опишите результат и критерии успеха">
    Используйте этот промпт:

    <PromptActions prompt={GATHER_PROMPT} campaign="docs-your-first-app" />

    <Frame>
      <img src="/assets/mirror/01-prompt-free-mode-f5c56ce766c8.jpg" alt="Промпт Gather готов к отправке: выбран артефакт Website и включён Free Mode" width="1280" height="720" data-path="images/first-app-gather/01-prompt-free-mode.jpg" />
    </Frame>

    Промпт начинается с одной цели: создать приложение для поиска мастер-классов. Затем в нём указано, что должны уметь посетители, определены проверяемые критерии успеха и отложены вход в систему, работа с данными, платежи и интеграции.

    Эти ограничения помогают сосредоточиться на первой версии. Когда основной сценарий заработает, вы сможете добавить возможности, необходимые для продакшена.
  </Step>

  <Step title="Позвольте Agent создать приложение в Free Mode">
    Отправьте промпт. Agent планирует артефакт, создаёт приложение, запускает его и проверяет результат. Следите за ходом работы в чате и дождитесь, пока Agent представит артефакт Website и запустит Preview.

    Создание полноценной первой версии в Free Mode может занять несколько минут. Если Replit порекомендует более высокий режим для сложного запроса, вы сможете решить, стоит ли перейти на него; эта версия Gather создаётся в Free Mode.

    <Frame>
      <img src="/assets/mirror/02-first-version-163a5fb49aac.jpg" alt="Первый рабочий сайт Gather в стандартной раскладке Build с чатом с Agent и Preview" width="1393" height="868" data-path="images/first-app-gather/02-first-version.jpg" />
    </Frame>
  </Step>

  <Step title="Проверьте результат в Preview">
    Протестируйте приложение как посетитель. Убедитесь, что вы можете:

    * Просматривать список мастер-классов.
    * Открывать страницу мастер-класса с подробностями.
    * Отправлять RSVP.
    * Видеть понятное подтверждение после отправки.
    * Пользоваться основной навигацией.
    * Уменьшать ширину Preview до размера экрана телефона.

    Сравните результат с критериями успеха в промпте. Если что-то не работает, опишите проблему Agent, а не пытайтесь самостоятельно диагностировать код.
  </Step>
</Steps>

<div id="improve-the-app-with-a-focused-prompt">
  ## Улучшите приложение с помощью конкретного промпта
</div>

Когда основной сценарий работает, запрашивайте по одному чёткому улучшению за раз. Для Gather добавьте представление отправленных RSVP только для текущей сессии:

> Добавь представление «Мои RSVP» и сделай так, чтобы отправленные мастер-классы отображались в нём в течение текущей сессии. Сохрани существующий визуальный стиль и не добавляй вход в систему или базу данных.

<Frame>
  <img src="/assets/mirror/03-refinement-prompt-94b96ae38acb.jpg" alt="Конкретный последующий промпт с просьбой к Agent добавить представление «Мои RSVP» только для текущей сессии в Free Mode" width="1393" height="868" data-path="images/first-app-gather/03-refinement-prompt.jpg" />
</Frame>

Снова протестируйте сценарий RSVP, затем откройте **Мои RSVP**. Отправленный мастер-класс должен отображаться в течение текущей сессии браузера.

Теперь вы завершили основной цикл Replit: **промпт, Build, тестирование и доработка**.
