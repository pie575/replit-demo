export const YouTubeEmbed = ({videoId, title = "YouTube video", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<YouTubeEmbed videoId="bAUVQfz_SIY" title="Introducing Replit Agent 4" />

Agent 4 现已发布。Agent 为创造而生，是你的 AI 创意伙伴，只需用日常语言描述，就能把想法变成应用、设计、幻灯片等成果。无需编写代码。

使用 Agent 4 可以：

* **构建任何内容**：创建 Web 应用、移动应用、数据看板、AI 驱动的工具等
* **先设计，后构建**：在动手写代码前，先在[设计 Canvas](/zh/learn/projects-and-artifacts/canvas) 中探索原型
* **制定项目计划**：在编写代码前，使用 [Plan 模式](/zh/core-concepts/agent/plan-mode)规划构建方案
* **并行执行任务**：Agent 同时处理多项任务，推进更快
* **与团队成员协作**：邀请他人加入 project，多人同时构建
* **连接工具**：直接在聊天中从 BigQuery、Linear、Slack、Notion 等获取数据
