export const YouTubeEmbed = ({videoId, title = "Видео на YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<YouTubeEmbed videoId="YSeDWL4cDLs" title="Мобильные приложения в Replit" />

Создавайте нативные приложения для iOS по простому prompt. Agent создаёт полноценные приложения на React Native, а вы можете добавлять базы данных, платежи Stripe или интеграции с OpenAI. Предварительно просматривайте приложение, отсканировав QR-код в Expo Go, а затем отправляйте его в App Store через пошаговый процесс Replit. Apple проверяет и публикует приложения в App Store.

Готовы создать своё первое мобильное приложение? Начните с [Native Mobile Apps](/ru/pt/features/artifact-types/building-mobile-apps) или пройдите руководство [Build and launch a mobile app](/ru/pt/tutorials/build-and-launch-a-mobile-app).

  ### Обновлённый интерфейс вызовов инструментов в чате Agent
</div>

В чате Agent появился более лаконичный и компактный интерфейс для вызовов инструментов. Вызовы инструментов сворачиваются в сгруппированные строки со значками, расположенными друг над другом, чтобы вы могли с первого взгляда понять, что происходит. Изменения и создание файлов напрямую связаны с панелью различий для удобного просмотра. Этот упрощённый дизайн помогает отслеживать работу Agent с меньшим визуальным шумом и большей ясностью.
