export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "已复制！";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        复制提示词
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        在 Replit 上构建 ↗
      </button>
    </div>;
};

<div id="before-you-start">
  ## 开始之前
</div>

你需要一个 [Replit 账户](https://replit.com)。本指南使用 **Free Mode**，它通过智能模型路由自动选择模型。{FreeModePlanAvailability}

目标是练习一套可重复的提示循环：

1. 描述一个有价值的成果。
2. 补充可观察的成功标准。
3. 划定边界，明确哪些内容暂时不要构建。
4. 在 Preview 中测试结果。
5. 提出一项针对性的改进要求。

<div id="build-the-first-version">
  ## 构建第一个版本
</div>

<Steps>
  <Step title="新建一个 Website">
    开始一个新的 Conversation，并选择 **Website** 作为 artifact。你可以从 **New** 菜单中选择，也可以在提示词输入框中通过 **Add attachment → Create something new → Website** 选择。

    打开模式选择器，选择 **Free**。

    <Tip>
      你也可以直接用自然语言让 Replit 构建一个网站。先选择 Website 能明确指定要生成的 artifact，并在提示词中添加一个 Website 标签。
    </Tip>
  </Step>

  <Step title="描述目标结果与成功标准">
    使用以下提示词：

    <PromptActions prompt={GATHER_PROMPT} campaign="docs-your-first-app" />

    <Frame>
      <img src="/assets/mirror/01-prompt-free-mode-f5c56ce766c8.jpg" alt="已选择 Website artifact 并启用 Free Mode，Gather 提示词已准备好发送" width="1280" height="720" data-path="images/first-app-gather/01-prompt-free-mode.jpg" />
    </Frame>

    这个提示词只围绕一个目标展开：一个用于发现工作坊的应用。接着说明访客必须能完成哪些操作，定义可观察的成功标准，并把登录、数据、支付和集成留到之后再做。

    这些边界让第一次构建保持聚焦。等核心体验跑通之后，你可以再补充面向生产的能力。
  </Step>

  <Step title="让 Agent 在 Free Mode 下构建">
    发送提示词。Agent 会规划 artifact、创建应用、启动应用并检查结果。在 Chat 中关注进度，直到 Agent 给出 Website artifact 且 Preview 正常运行。

    在 Free Mode 下完成首次完整构建可能需要几分钟。如果 Replit 针对更复杂的请求建议使用更高级别的模式，你可以自行决定是否切换；本次 Gather 构建在 Free Mode 下即可完成。

    <Frame>
      <img src="/assets/mirror/02-first-version-163a5fb49aac.jpg" alt="默认 Build 布局中的第一个可用 Gather 网站，包含 Agent chat 和 Preview" width="1393" height="868" data-path="images/first-app-gather/02-first-version.jpg" />
    </Frame>
  </Step>

  <Step title="在 Preview 中测试结果">
    像访客一样使用这个应用，确认你可以：

    * 浏览工作坊列表。
    * 打开工作坊详情页。
    * 提交 RSVP。
    * 提交后看到清晰的确认信息。
    * 使用主导航。
    * 将 Preview 调整到手机屏幕宽度。

    将结果与提示词中的成功标准逐项对照。如果有问题，向 Agent 描述具体现象，而不是自己去排查代码。
  </Step>
</Steps>

<div id="improve-the-app-with-a-focused-prompt">
  ## 使用明确的提示词改进 App
</div>

核心体验运行正常后，每次只提出一项明确的改进。对于 Gather，添加一个仅在当前会话中显示已提交 RSVP 的视图：

> 添加“我的 RSVP”视图，让已提交的工作坊显示在当前会话中。保持现有视觉风格，不要添加登录功能或数据库。

<Frame>
  <img src="/assets/mirror/03-refinement-prompt-94b96ae38acb.jpg" alt="一条明确的后续提示词，要求 Agent 在 Free Mode 中添加仅限当前会话的“我的 RSVP”视图" width="1393" height="868" data-path="images/first-app-gather/03-refinement-prompt.jpg" />

再次测试 RSVP 流程，然后打开 **我的 RSVP**。已提交的工作坊应显示在当前浏览器会话中。
