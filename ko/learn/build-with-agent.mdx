  const AGENT_INPUT_START_DELAY = 500;
  const getAgentInputTypewriterLoader = () => {
    if (typeof window === "undefined") {
      return null;
    }
    if (!window.__replitDocsAgentInputTypewriterLoader) {
      window.__replitDocsAgentInputTypewriterLoader = {};
    }
    return window.__replitDocsAgentInputTypewriterLoader;
  };
  const ensureAgentInputTypewriterCore = () => {
    if (typeof window === "undefined" || typeof document === "undefined") {
      return Promise.resolve(null);
    }
    if (window.Typewriter) {
      return Promise.resolve(window.Typewriter);
    }
    const loader = getAgentInputTypewriterLoader();
    if (loader.promise) {
      return loader.promise;
    }
    loader.promise = new Promise((resolve, reject) => {
      const handleLoad = () => {
        if (window.Typewriter) {
          resolve(window.Typewriter);
          return;
        }
        loader.promise = null;
        reject(new Error("Typewriter core loaded without a global Typewriter export."));
      };
      const handleError = () => {
        loader.promise = null;
        reject(new Error("Failed to load typewriter core."));
      };
      const existingScript = document.getElementById(AGENT_INPUT_TYPEWRITER_SCRIPT_ID);
      if (existingScript) {
        existingScript.addEventListener("load", handleLoad, {
          once: true
        });
        existingScript.addEventListener("error", handleError, {
          once: true
        });
        return;
      }
      const script = document.createElement("script");

      script.src = AGENT_INPUT_TYPEWRITER_SRC;
      script.async = true;
      script.addEventListener("load", handleLoad, {
        once: true
      });
      script.addEventListener("error", handleError, {
        once: true
      });
      document.head.appendChild(script);
    });
    return loader.promise;
  };

    if (typeof window === "undefined") {
      return null;

    startButton: container?.querySelector?.('[data-agent-input-role="start-button"]') || null,
    pillsContainer: container?.querySelector?.('[data-agent-input-role="pills"]') || null

    return textarea?.value?.trim() || "";

  const updateAgentInputButtonState = container => {

    if (!startButton) {
      return;

    const hasText = getAgentInputPromptValue(container).length > 0;
    startButton.style.opacity = hasText ? "1" : "0.4";
    startButton.style.cursor = hasText ? "pointer" : "not-allowed";
  };
  const stopAgentInputTypewriter = (textarea, {placeholder, clearDemo = false} = {}) => {
    if (typeof window === "undefined" || !textarea) {
      return;
    }
    const registry = getAgentInputTypewriterRegistry();
    const record = registry?.get(textarea);
    if (!record) {
      if (placeholder) {
        textarea.setAttribute("placeholder", placeholder);
      }
      return;
    }
    record.stopped = true;
    if (record.startTimer) {
      window.clearTimeout(record.startTimer);
    }
    if (record.writeFrame) {
      window.cancelAnimationFrame(record.writeFrame);
    }
    if (record.typewriter?.stop) {
      record.typewriter.stop();
    }

    textarea.removeEventListener("pointerdown", record.handlePointerDown);
    textarea.removeEventListener("input", record.handleInput);
    if (clearDemo && textarea.value === record.pendingValue) {
      textarea.value = "";
    }
    textarea.setAttribute("placeholder", placeholder || "");

    registry.delete(textarea);

      return;

      return;

        return;

[Vibe coding 101](/ko/learn/foundations/vibe-coding-101)에서는 사고방식을 다루고, 이 페이지에서는 실제로 Agent와 함께 작업하는 방법을 설명합니다. [Effective prompting](/ko/learn/effective-prompting)에서는 더 명확한 지시를 작성하는 방법을 안내합니다.

Agent를 경기장의 쿼터백이라고 생각해 보세요. 상황을 읽고, 방향을 정하고, 작업을 진행해 나갑니다. 여러분은 코치입니다. 전략을 세우고, 플레이를 지시하고, 결과를 검토하고, 다음 수를 결정합니다.

Agent는 많은 일을 할 수 있지만, 여러분이 과정을 주도할 때 가장 잘 작동합니다. 구체적으로 요청하고, 작업을 계획하고, context를 추가하고, 검토와 테스트를 거치고, 복구가 필요할 때는 checkpoints를 활용하세요.

Agent는 복잡한 작업도 처리할 수 있지만, 명확한 방향 제시가 필요합니다.

이렇게 쓰는 대신:

동작, 목표, 제약 조건에 관한 변경이라면 텍스트를 사용하세요:

<AiPrompt>
  Context: 이 앱은 생일 케이크를 주문하는 부모를 위한 것입니다.

  <br />

  양식은 짧게 유지하고, 픽업 정보는 눈에 잘 띄게 하며, 딱딱한 기업 느낌이 나는 요소는
  피해 주세요.
</AiPrompt>

layout을 말로 설명하기 어려울 때는 스크린샷이나 스케치를 사용하세요:

<AiPrompt>
  이 스크린샷을 요청 양식의 layout direction으로 사용해 주세요.

  <br />

  전체 구조는 맞추되, 기존 필드와 제출
  동작은 그대로 유지해 주세요.
</AiPrompt>

시각적인 변경이라 앱의 특정 부분을 콕 집어 가리키고 싶을 때는 [Canvas](/ko/design/canvas) 주석을 사용하세요.

<Frame>
  <img src="/assets/mirror/agent-canvas-annotations-c88ab5856966.png" alt="진행 카드를 가리키는 사각형, 화살표, 텍스트 label과 양식 영역의 펜 스케치 및 label이 있는 Canvas 위의 피트니스 앱 frame" width="1440" height="900" data-path="images/fitstart/work-with-agent/agent-canvas-annotations.png" />
</Frame>

<AiPrompt>
  제 Canvas 주석을 참고해 hero section을 업데이트해 주세요.

  <br />

  메모는 헤드라인, 이미지, 기본 버튼에만 적용해 주세요.

  <br />

  케이터링 요청 양식은 변경하지 마세요.
</AiPrompt>

좋은 context에는 변경하면 안 되는 부분도 함께 담깁니다. 그래야 Agent가 다른 영역을 실수로 다시 작성하는 일 없이 원하는 부분만 개선할 수 있습니다.

<div id="review-and-test">
  ## 검토 및 테스트
</div>

Agent는 빌드하기 전에 계획을 제안하고, 완료 후에는 변경된 내용을 요약해 주기도 합니다. 두 가지 모두 읽어보세요.

task plan 카드에서 **View**를 선택해 계획을 엽니다. 확장된 계획에는 Agent가 무엇을 만들려고 하는지, 성공 기준, 범위에서 제외되는 항목, 그리고 Agent가 따를 단계가 표시됩니다.

<Frame>
  <img src="/assets/mirror/agent-plan-content-c298ea216ebc.png" alt="What and Why, Done 기준, 범위 제외 항목, 번호가 매겨진 빌드 단계를 보여주는 확장된 task plan" width="1440" height="900" data-path="images/fitstart/work-with-agent/agent-plan-content.png" />
</Frame>

Agent가 빌드를 시작하기 전에 계획을 검토하세요.

* 올바른 문제를 해결하고 있나요?
* 빠진 것은 없나요?
* 범위를 벗어난 항목은 없나요?
* 먼저 만들어볼 더 작은 버전이 있나요?
* 결과를 어떻게 테스트해야 하는지 설명하고 있나요?

계획의 범위가 너무 넓다면 Agent에게 줄여달라고 요청하세요.

<AiPrompt>
  이 계획은 범위가 너무 넓습니다.

  <br />

  오늘 바로 테스트할 수 있는 가장 작은 버전으로 줄여주세요.

  <br />

  나머지는 모두 향후 개선 목록으로 옮겨주세요.
</AiPrompt>

Agent가 빌드를 마치면 앱을 직접 테스트하세요. Preview를 열고 실제 사용자의 입장에서 써 보세요.

확인할 사항:

* 주요 동작을 끝까지 완료할 수 있나요?
* 요청한 변경 사항이 반영되었나요?
* 모바일에서도 앱이 정상적으로 작동하나요?
* 중요한 동작이 깨지지는 않았나요?
* Agent가 관련 없는 부분을 변경하지는 않았나요?
* 이어서 만들어 나가기에 충분한 결과인가요?

문제가 있다면 무엇을 보았는지 설명하세요.

이렇게 하는 대신:

<AiPrompt>양식을 고쳐주세요.</AiPrompt>

이렇게 해보세요:

<AiPrompt>
  케이터링 요청 양식을 제출하면 성공 메시지는 나타나지만, 양식에는 여전히 이전 텍스트가
  남아 있습니다.

  <br />

  제출 후 양식을 비우고 성공 메시지는 계속 표시되도록 해주세요.
</AiPrompt>

구체적인 피드백을 주면 Agent가 정확한 문제를 해결하는 데 도움이 됩니다.

<div id="use-checkpoints">
  ## checkpoint 사용하기
</div>

checkpoint는 변경이 잘못된 방향으로 흘러갈 때 원래 상태로 되돌리는 데 도움이 됩니다.

결과가 원하는 모습에 거의 가까워졌다면 구체적인 피드백을 주세요:

<AiPrompt>
  거의 다 됐지만, 양식이 이제 너무 깁니다.

  <br />

  새로운 시각적 스타일은 유지하고, 선택 항목인 메모 필드를 제거하며,
  픽업 날짜를 더 쉽게 확인할 수 있게 해주세요.
</AiPrompt>

앱이 이전보다 나빠졌거나, 중요한 동작이 망가졌거나, Agent가 예상보다 많은 것을 변경했다면 checkpoint를 사용하세요.

Agent 패널에서 **History**를 열면 checkpoint를 확인할 수 있습니다. 각 checkpoint에는 무엇이 변경되었는지가 표시되며, **Rollback here**와 **Changes** 컨트롤이 함께 제공됩니다.

<Frame>
  <img src="/assets/mirror/agent-checkpoints-history-c33b3b0dd592.png" alt="'Transitioned from Plan to Build mode'라는 이름의 checkpoint와 Rollback here 및 Changes 버튼이 표시된 Agent History 패널" width="720" height="450" data-path="images/fitstart/work-with-agent/agent-checkpoints-history.png" />
</Frame>

**Rollback here**를 선택하면 Agent가 rollback을 적용하기 전에 무엇이 복원되는지 확인시켜 줍니다.

<Frame>
  <img src="/assets/mirror/agent-checkpoints-rollback-dialog-28de278b44da.png" alt="대상 checkpoint와 영향을 받을 항목(파일, Agent memory, task, Database 체크박스)을 나열한 Rollback to checkpoint 확인 대화 상자" width="900" height="880" data-path="images/fitstart/work-with-agent/agent-checkpoints-rollback-dialog.png" />
</Frame>

이어서 수정할지 roll back할지 판단이 서지 않는다면, Agent에게 비교를 도와달라고 요청하세요.

<AiPrompt>
  마지막 변경으로 앱이 더 나빠졌습니다.

  <br />

  현재 버전과 이전에 정상 작동하던 버전을 비교해 주세요.

  <br />

  이어서 수정하는 것과 roll back하는 것 중 무엇이 더 안전한지 알려주고, 그 이유를 설명해 주세요.
</AiPrompt>

잘못된 방향으로 가는 일은 흔합니다. checkpoint가 있으면 더 안전한 버전으로 돌아갈 길을 남겨둔 채 자유롭게 시도해 볼 수 있습니다.

복구에 대한 더 자세한 안내는 [Checkpoints and rollbacks](/ko/features/version-control/checkpoints-and-rollbacks)를 참고하세요.

<div id="what-good-agent-collaboration-feels-like">
  ## 좋은 Agent 협업이란
</div>

다음과 같다면 올바른 방향으로 가고 있는 것입니다.

* 요청할 때 목표와 제약 조건을 함께 설명합니다
* Agent의 계획이 검토하기에 충분히 명확합니다
* 말로 설명하기 어려울 때는 스크린샷, 파일, Canvas 메모를 추가합니다
* 변경 후에는 중요한 흐름을 테스트합니다
* 무언가 어긋났을 때 핵심을 짚은 피드백을 제공합니다
* 잘못된 변경을 끝까지 되돌리려 애쓰는 대신 checkpoint를 활용합니다
