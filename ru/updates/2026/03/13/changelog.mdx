export const YouTubeEmbed = ({videoId, title = "Видео на YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

Agent 4 уже здесь. Agent создан для творчества — это ваш креативный ИИ-партнёр, который превращает идеи в приложения, дизайны, презентации и многое другое по описанию на обычном языке. Программировать не нужно.

С Agent 4 вы можете:

* **Создавать что угодно**: веб-приложения, мобильные приложения, дашборды с данными, инструменты на базе ИИ и не только
* **Проектировать до разработки**: изучать mockup-ы на [Design Canvas](/ru/learn/projects-and-artifacts/canvas), прежде чем переходить к коду
* **Планировать свой project**: использовать [режим Plan](/ru/core-concepts/agent/plan-mode), чтобы продумать разработку ещё до написания кода
* **Выполнять задачи параллельно**: Agent обрабатывает несколько задач одновременно, поэтому вы движетесь быстрее
* **Работать вместе с коллегами**: приглашайте других в свой project и создавайте приложения совместно
* **Подключать свои инструменты**: получайте данные из BigQuery, Linear, Slack, Notion и других сервисов прямо из чата
