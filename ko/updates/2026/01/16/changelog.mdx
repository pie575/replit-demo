export const YouTubeEmbed = ({videoId, title = "YouTube 동영상", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<div id="redesigned-tool-calls-ui-in-agent-chat">
  ### Agent chat의 도구 호출 UI 재설계
</div>

이제 Agent chat의 도구 호출 인터페이스가 더 깔끔하고 간결해졌습니다. 도구 호출은 아이콘이 겹쳐진 그룹 행으로 접혀서 표시되므로, 현재 진행 중인 작업을 한눈에 파악할 수 있습니다. 파일 편집과 생성 항목은 diff 패널로 바로 연결되어 손쉽게 검토할 수 있습니다. 이렇게 정돈된 디자인 덕분에 불필요한 정보 없이 Agent의 작업 과정을 더 명확하게 따라갈 수 있습니다.
