export const FreeModePlanAvailability = `Free Mode는 Starter, Core, Pro 요금제에서 사용할 수 있습니다. Free Mode 한도는 요금제에 따라 다르며, Core와 Pro에는 ${FreeModeAllowanceRefreshPeriod}마다 갱신되는 사용량이 포함됩니다.`;

export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "복사됨!";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        프롬프트 복사
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Build on Replit ↗
      </button>
    </div>;
};

export const GATHER_PROMPT = `Gather라는 반응형 워크숍 탐색 앱을 만드세요.

방문자는 예정된 워크숍을 찾아보고, 이벤트 세부 정보를 확인하며, 현실적인 샘플 데이터를 사용해 RSVP를 제출할 수 있어야 합니다.

성공 기준은 내비게이션이 정상적으로 작동하고, RSVP를 제출하면 명확한 확인 메시지가 표시되며, 레이아웃이 모바일과 데스크톱 모두에서 잘 작동하는 것입니다.

아직 로그인, 데이터베이스, 결제 또는 외부 통합은 추가하지 마세요.`;

<div id="before-you-start">
  ## 시작하기 전에
</div>

[Replit 계정](https://replit.com)이 필요합니다. 이 가이드는 **Free Mode**를 사용하며, Free Mode는 intelligent model routing으로 모델을 자동 선택합니다. {FreeModePlanAvailability}

목표는 반복해서 활용할 수 있는 프롬프트 loop를 익히는 것입니다.

1. 유용한 결과 하나를 설명합니다.
2. 눈으로 확인할 수 있는 성공 기준을 추가합니다.
3. 아직 만들지 않을 부분의 범위를 정합니다.
4. Preview에서 결과를 테스트합니다.
5. 개선 사항 하나만 콕 집어 요청합니다.

<div id="build-the-first-version">
  ## 첫 번째 버전 만들기
</div>

<Steps>
  <Step title="새 Website 시작하기">
    새 conversa를 시작하고 artifact로 **Website**를 선택합니다. **New** 메뉴에서 선택하거나 프롬프트 box의 **Add attachment → Create something new → Website**에서 선택할 수 있습니다.

    모드 선택기를 열고 **Free**를 선택합니다.

    <Tip>
      자연어로 Replit에 웹사이트를 만들어 달라고 바로 요청할 수도 있습니다. Website를 먼저 선택하면 의도한 artifact가 명확해지고 프롬프트에 Website 표시가 추가됩니다.
    </Tip>
  </Step>

  <Step title="결과물과 성공 기준 설명하기">
    다음 프롬프트를 사용하세요:

    <PromptActions prompt={GATHER_PROMPT} campaign="docs-your-first-app" />

    <Frame>
      <img src="/assets/mirror/01-prompt-free-mode-f5c56ce766c8.jpg" alt="Website artifact가 선택되고 Free Mode가 활성화된 상태로 전송 준비가 된 Gather 프롬프트" width="1280" height="720" data-path="images/first-app-gather/01-prompt-free-mode.jpg" />
    </Frame>

    이 프롬프트는 워크숍 탐색 앱이라는 하나의 결과물에서 출발합니다. 이어서 방문자가 할 수 있어야 하는 일을 명시하고, 관찰 가능한 성공 기준을 정의하며, 로그인, 데이터, payments, 통합은 뒤로 미룹니다.

    이렇게 범위를 정해 두면 첫 build가 핵심에만 집중할 수 있습니다. production capabilities는 핵심 경험이 제대로 작동한 뒤에 추가하면 됩니다.
  </Step>

  <Step title="Free Mode에서 Agent가 만들도록 하기">
    프롬프트를 전송합니다. Agent가 artifact를 계획하고, 앱을 만들고, 실행한 뒤 결과를 확인합니다. Chat에서 진행 상황을 지켜보며 Agent가 Website artifact를 제시하고 Preview가 실행될 때까지 기다립니다.

    Free Mode에서는 첫 build를 완성하는 데 몇 분이 걸릴 수 있습니다. 더 복잡한 요청에 대해 Replit이 더 높은 모드를 권장하더라도 상향 여부는 직접 결정하면 됩니다. 이 Gather build는 Free Mode에서 완료됩니다.

    <Frame>
      <img src="/assets/mirror/02-first-version-163a5fb49aac.jpg" alt="Agent chat과 Preview가 있는 기본 Build layout에 표시된, 처음으로 작동하는 Gather 웹사이트" width="1393" height="868" data-path="images/first-app-gather/02-first-version.jpg" />
    </Frame>
  </Step>

  <Step title="Preview에서 결과 테스트하기">
    방문자의 입장에서 앱을 사용해 보세요. 다음을 확인합니다:

    * 워크숍 목록을 둘러볼 수 있는지
    * 워크숍 상세 페이지를 열 수 있는지
    * RSVP를 제출할 수 있는지
    * 제출 후 명확한 확인 메시지가 표시되는지
    * 기본 내비게이션을 사용할 수 있는지
    * Preview를 휴대폰 크기의 너비로 조정할 수 있는지

    결과를 프롬프트에 적어 둔 성공 기준과 비교해 보세요. 문제가 있다면 코드를 직접 진단하려 하지 말고 증상을 Agent에게 설명하세요.
  </Step>
</Steps>

<div id="improve-the-app-with-a-focused-prompt">
  ## 명확한 프롬프트 하나로 앱 개선하기
</div>

핵심 기능이 정상 동작하면, 한 번에 하나씩 명확한 개선 사항을 요청하세요. Gather에서는 제출한 RSVP를 현재 세션 동안만 볼 수 있는 화면을 추가해 보겠습니다.

> My RSVPs 뷰를 추가하고, 제출한 workshops이 현재 세션에 표시되도록 해줘. 기존 시각적 스타일은 그대로 유지하고 로그인이나 데이터베이스는 추가하지 마.
