export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "복사됨!";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        프롬프트 복사
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Replit에서 Build하기 ↗
      </button>
    </div>;
};

<div id="build-in-parallel">
  ### 병렬로 빌드하기
</div>

제품에 대한 고민이 단순한 구현이 아니라 방향에 관한 것이라면, 여러 아이디어를 동시에 탐색해 보는 편이 좋습니다. 너무 일찍 하나를 확정하는 대신 선택지를 비교해 볼 수 있기 때문입니다. Replit Agent는 병렬 task를 지원하므로, Agent에게 두 개 이상의 프로토타입, 기능, 실험을 동시에 진행하게 한 뒤 나란히 놓고 검토할 수 있습니다.

<Frame>
  <img src="/assets/mirror/task-plan-sidebar-ecab81b06bb0.png" alt="여러 Agent task가 상태 표시와 함께 병렬로 실행되는 모습을 보여주는 thread 화면" width="3430" height="1986" data-path="images/replitai/task-plan-sidebar.png" />
</Frame>

task가 대기열에 쌓이고 실행되고 완료되는 방식은 [task system](/ko/core-concepts/agent/task-system)을 참고하세요.

어느 방향이 더 나은지 확신이 서지 않을 때는 Agent에게 병렬 task 두 개를 만들어 달라고 요청하세요. Pace라면 다음과 같이 비교해 볼 수 있습니다.

* **동기 부여 중심 프로토타입**: 연속 기록, 업적, 격려하는 추천.
* **분석 중심 프로토타입**: 더 깊이 있는 통계, 주간 대비 추세, 개인 기록.

<AiPrompt>
  Pace를 제품 프로토타입으로 개선하기 위한 병렬 task 두 개를 만들어 줘.<br />
  Task 1은 연속 기록, 업적, 격려하는 추천이 담긴 동기 부여 중심 버전을 탐색해 줘.<br />
  Task 2는 더 깊이 있는 통계, 주간 대비 추세, 개인 기록이 담긴 분석 중심 버전을 탐색해 줘.<br />
  두 버전 모두 오늘 안에 테스트할 수 있을 만큼 작게 유지해 줘.<br />
  각 task마다 사용자 플로우, 무엇을 빌드할지, 결과를 어떻게 평가하면 좋을지 정의해 줘.
</AiPrompt>

두 task가 모두 끝나면 동일한 feedback 질문으로 비교해 보세요.

> 어느 버전이 내일 다시 돌아오고 싶게 만드나요? 그 이유는 무엇인가요?

<div id="need-further-help">
  ## 추가 도움이 필요하신가요?
</div>

* **프로토타입에 기능이 너무 많은 경우:** Agent에게 대상 사용자 한 명과 플로우 하나로 줄여 달라고 요청하세요.
* **대상 사용자가 불분명한 경우:** Agent에게 대상 사용자 한 명에 맞춰 환영 섹션을 다시 작성해 달라고 요청하세요.
* **목업 데이터가 위험을 가리는 경우:** Agent에게 무엇이 실제 데이터이고 무엇이 목업인지 정리해 달라고 요청하세요.
* **피드백이 모호한 경우:** 검토자에게 task 하나와 질문 하나만 제시하세요.
* **Publishing이 실패하는 경우:** [Publishing 문제 해결](/ko/build/troubleshooting)을 참고하세요.
