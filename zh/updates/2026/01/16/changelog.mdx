export const YouTubeEmbed = ({videoId, title = "YouTube 视频", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

  ### Replit 上的移动应用
</div>

<YouTubeEmbed videoId="YSeDWL4cDLs" title="Replit 上的移动应用" />

只需一个简单的 prompt，即可构建原生 iOS 应用。Agent 可创建完整的 React Native 应用，还可在开发过程中添加数据库、Stripe 支付或 OpenAI 集成等功能。使用 Expo Go 扫描二维码即可预览应用，然后通过 Replit 的引导流程提交到 App Store。Apple 会通过 App Store 审核和发布应用。

准备构建你的第一个移动应用了吗？查看[原生移动应用](/zh/features/artifact-types/building-mobile-apps)开始使用，或按照[构建并发布移动应用](/zh/tutorials/build-and-launch-a-mobile-app)教程操作。

<div id="nodejs-security-patch">
  ### Node.js 安全补丁
</div>

本周，Node.js 披露了一个严重的安全漏洞。所有 Agent 开发环境和 Replit 的 Node.js 应用均已更新至安全版本。重新发布应用后，您的应用也会更新至安全版本。
