export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "已复制！";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        复制提示词
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        在 Replit 上构建 ↗
      </button>
    </div>;
};

<div id="section">
  ## 您将学到什么
</div>

您将学习如何：

* 将目标受众、优惠和行动号召转化为页面。
* 请 Agent 处理页面文案、布局和表单功能。
* 在分享前确认页面内容是否清晰。
* 通过发布后的 URL 测试表单。
* 根据回复质量改进页面。

<div id="canvas">
  ## 在 Canvas 中探索优化方案和替代方向
</div>

Agent 完成初步构建后，打开 Canvas 提供视觉反馈，并让 Agent 探索其他方向，无需重写提示词。

<Steps>
  <Step title="构建落地页">
    复制提示词并粘贴到 [replit.com](https://replit.com) 的提示词输入框中，或者点击 **Build on Replit**，在 Replit 中打开已预先填入提示词的页面。

    <PromptActions prompt={RACHEL_JONES_PROMPT} campaign="docs-landing-page" />
  </Step>

  <Step title="预览 Agent 构建的应用">
    Agent 完成后，右侧窗格会切换到 **Preview**，你可以实时与着陆页交互。开始改进前，请滚动浏览主视觉区、点击号召性用语，并确认文案和图片是否符合你的要求。

    <Frame>
      <img src="/assets/mirror/landing-preview-4855c4b25270.png" alt="Agent 聊天旁的 Replit Preview 窗格中渲染的 Rachel Jones 健身教练着陆页——深色编辑风格的主视觉区，金色光线照亮的人像上方显示“更强健的身体。更自信的心态。势不可挡的你。”标题，包含“与 Rachel 合作”和“观看 Rachel 的故事”按钮，右侧展示认证资质，以及“受到全球 500 多名女性信赖”的五星评价横幅" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-preview.png" />
    </Frame>
  </Step>

  <Step title="打开 Canvas">
    选择 Preview 和 Tools 标签下方的 **Canvas** 按钮，将右侧窗格从 Preview 切换为 Canvas。

    <Frame>
      <img src="/assets/mirror/landing-canvas-button-494c5b0076b1.png" alt="Replit 工作区中突出显示的 Preview 和 Tools 标签下方的 Canvas 按钮" width="720" height="450" data-path="images/fitstart/launch-landing-page/landing-canvas-button.png" />
    </Frame>
  </Step>

  <Step title="在预览区域外添加注释">
    使用 Canvas 的 **Text** 和 **Draw** 工具，在模型周围的空白区域添加备注和箭头，标明你希望修改的具体元素。放置在画布上 (预览区域外) 的注释会保留在反馈中，Agent 可以结合设计一并查看。

    每条注释只聚焦一项修改。简短、具体的备注 (“缩窄此按钮的宽度”) 比开放式反馈更有效。

    <Frame>
      <img src="/assets/mirror/landing-canvas-annotate-text-8135bfe1e8e2.png" alt="位于 Rachel Jones 落地页右侧画布边距中的“缩窄此按钮的宽度”画布注释，带有指向页面标题中金色“Book a Free Consultation”按钮的箭头" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-annotate-text.png" />
    </Frame>
  </Step>

  <Step title="选择标注和预览，然后发送给 Agent">
    将标注和预览框架一起框选，然后在 **Chat about selected items** 中输入简短提示词并发送。Agent 会将标注和设计视为同一组上下文，因此能理解每条备注对应哪个元素。

    聊天消息保持笼统即可，例如“应用标注中的更改”。具体要求交给标注说明。

    <Frame>
      <img src="/assets/mirror/landing-canvas-send-24b771bf29bf.png" alt="Canvas 上与两个标注一起被选中的 Rachel Jones 落地页 — 指向 Book a Free Consultation 按钮的“Reduce the width of this button”，以及指向主视觉照片的“Add a slide-in animation for Rachel's image”，Chat about selected items 输入框中输入的聊天提示词" width="1726" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-send.png" />
    </Frame>
  </Step>

  <Step title="探索不同的方向">
    点击 Canvas 上的设计，然后选择旁边显示的建议 (例如 **探索不同方案**) ，让 Agent 探索不同的视觉方向。

    <Frame>
      <img src="/assets/mirror/landing-canvas-reimagine-9e60c24f342f.png" alt="Canvas 上原始设计旁边生成的三个备选落地页 Design frame" width="1726" height="994" data-path="images/fitstart/launch-landing-page/landing-canvas-reimagine.png" />
    </Frame>
  </Step>

  <Step title="查看替代方案并迭代">
    等待 Agent 完成生成。**探索不同方案**会以新的变体更新同一产物 — 您可以通过模型图上方的产物下拉菜单切换。**探索不同风格**会在 Canvas 上原始设计旁生成不同的视觉模型图。

    变体生成后，选择一个方向并继续迭代。为所选帧添加其他注释，将其与设计一起框选，然后发送新的请求 — 例如“应用注释中要求的更改”。Agent 会将变体的上下文和新注释合并为一个提示词使用。

    <Frame>
      <img src="/assets/mirror/landing-canvas-zoomed-4d1585d33fe6.png" alt="Agent 生成三个风格变体（Editorial Warmth、Quiet Clinical、Vibrant Pop）后的 Canvas 上 Rachel Jones 落地页 — 原始帧与两条新注释（“Reduce the width of this button”和“Add a slide-in animation for Rachel's image”）一同被选中，聊天中输入了“Apply the requested changes from the annotations”" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-zoomed.png" />
    </Frame>
  </Step>

  <Step title="发布你的作品">
    当页面在 Preview 中正常运行后，点击工作区右上角的 **Publish** (后续部署时点击 **重新发布**) ，将最新版本发布到公开 URL。请在新标签页中打开已发布的 URL，并在分享前通过已上线页面发送一条测试回复，确认其已被捕获。

    <Frame>
      <img src="/assets/mirror/landing-publish-e3b901a79440.png" alt="以 2 倍缩放显示的 Replit 工作区右上角 — 标题栏中的灰色 Republish 按钮以橙色轮廓高亮显示，下方的弹出窗口显示“Raouf published about 18 hours ago”、Visibility: Public、Review security 和 Republish 操作按钮" width="2142" height="1072" data-path="images/fitstart/launch-landing-page/landing-publish.png" />
    </Frame>
  </Step>
</Steps>
