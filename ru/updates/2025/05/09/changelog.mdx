export const YouTubeEmbed = ({videoId, title = "Видео на YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<div id="improved-image-handling-for-broken-unsplash-urls">
  ### Улучшенная обработка неработающих URL-адресов Unsplash
</div>

Попрощайтесь с неработающими стоковыми изображениями в своих проектах!

Возможно, вы не знали, но Agent уже добавляет в ваши проекты стоковые изображения из Unsplash. Однако иногда URL-адреса оказываются недействительными. Причина? LLM отлично справляются с галлюцинацией URL-адресов стоковых изображений.

Мы добавили функцию, которая автоматически заменяет недоступные стоковые изображения на рабочие альтернативы, чтобы ваши приложения всегда выглядели профессионально.
