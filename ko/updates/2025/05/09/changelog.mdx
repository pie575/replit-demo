export const YouTubeEmbed = ({videoId, title = "YouTube 동영상", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<div id="notion-integration">
  ### Notion 연동
</div>

<YouTubeEmbed videoId="ENYxveSU-pM" title="Notion integration" />

Notion Agent 연동 기능을 모든 사용자에게 제공합니다. 이제 Replit projects와 Notion workspace를 매끄럽게 연결할 수 있습니다.

<Frame>
  <img src="/assets/mirror/notion-app-2632ca957cc8.png" width="5120" height="2880" data-path="images/changelog/2025-05-09/notion-app.png" />
</Frame>

Agent의 안내에 따라 설정하면서 Notion의 강력한 database 및 문서 기능을 활용하는 애플리케이션을 빠르게 만들어 보세요. 이 연동을 통해 Notion API에 연결된 애플리케이션을 구축할 수 있습니다.

즉, Notion workspace로 데이터를 push하고 pull하는 Replit apps를 만들 수 있다는 뜻입니다!

<div id="improved-image-handling-for-broken-unsplash-urls">
  ### 깨진 Unsplash URL에 대한 개선된 이미지 처리
</div>

project에서 깨진 스톡 이미지와 이제 작별하세요!

혹시 모르셨다면, Agent는 이미 Unsplash에서 스톡 이미지를 가져와 project에 추가합니다. 하지만 URL이 깨질 때가 있습니다. 이유가 뭘까요? LLM이 스톡 이미지 URL을 그럴듯하게 지어내는 데 워낙 능하기 때문입니다.

이제 깨진 스톡 이미지를 정상적으로 작동하는 대체 이미지로 자동 교체하는 기능이 추가되었습니다. 덕분에 여러분의 애플리케이션은 언제나 완성도 높은 모습을 유지합니다.

<div id="enhanced-enterprise-deployment-privacy-settings">
  ### 향상된 Enterprise 배포 개인정보 설정
</div>

이제 Enterprise 고객은 다양한 구성 옵션을 통해 배포 개인정보를 더욱 세밀하게 제어할 수 있습니다.

이 기능을 통해 조직은 배포된 애플리케이션에 대한 접근 권한을 정밀하게 관리할 수 있으며, Enterprise 환경에 필요한 보안과 유연성을 확보할 수 있습니다. 또한 팀 사용자의 배포도 이제 기본적으로 비공개로 설정되며, 공개적으로 공유하려면 직접 설정을 변경해야 합니다.
