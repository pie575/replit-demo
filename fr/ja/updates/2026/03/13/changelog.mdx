export const YouTubeEmbed = ({videoId, title = "Vidéo YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

Agent 4 est arrivé. Conçu pour stimuler votre créativité, Agent est votre partenaire créatif basé sur l&#39;IA. Transformez vos idées en applications, designs, diapositives et bien plus encore, sans coder.

Ce que vous pouvez faire avec Agent 4 :

* **Planifier vos projets** : définissez votre build dans le [mode Plan](/fr/core-concepts/agent/plan-mode) avant l&#39;écriture du code
* **Exécuter des tâches en parallèle** : Agent gère plusieurs tâches simultanément pour accélérer votre travail
* **Collaborer avec vos coéquipiers** : invitez d&#39;autres personnes à votre projet et créez ensemble en temps réel
* **Connecter vos outils** : récupérez directement depuis le chat des données de BigQuery, Linear, Slack, Notion et bien plus encore
