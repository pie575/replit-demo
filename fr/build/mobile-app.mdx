    if (typeof window === "undefined") {
      return null;
    }
    if (!window.__replitDocsAgentInputTypewriterLoader) {
      window.__replitDocsAgentInputTypewriterLoader = {};
    }
    return window.__replitDocsAgentInputTypewriterLoader;
  };
  const ensureAgentInputTypewriterCore = () => {
    if (typeof window === "undefined" || typeof document === "undefined") {
      return Promise.resolve(null);
    }
    if (window.Typewriter) {
      return Promise.resolve(window.Typewriter);
    }
    const loader = getAgentInputTypewriterLoader();
    if (loader.promise) {
      return loader.promise;
    }
    loader.promise = new Promise((resolve, reject) => {
      const handleLoad = () => {
        if (window.Typewriter) {
          resolve(window.Typewriter);
          return;
        }
        loader.promise = null;
        reject(new Error("Typewriter core loaded without a global Typewriter export."));
      };
      const handleError = () => {
        loader.promise = null;
        reject(new Error("Failed to load typewriter core."));
      };
      const existingScript = document.getElementById(AGENT_INPUT_TYPEWRITER_SCRIPT_ID);
      if (existingScript) {
        existingScript.addEventListener("load", handleLoad, {
          once: true

        existingScript.addEventListener("error", handleError, {
          once: true

        return;

      script.addEventListener("load", handleLoad, {
        once: true

      script.addEventListener("error", handleError, {
        once: true

      document.head.appendChild(script);
    });
    return loader.promise;

    pillsContainer: container?.querySelector?.('[data-agent-input-role="pills"]') || null

  const updateAgentInputButtonState = container => {

    if (!startButton) {
      return;

    const hasText = getAgentInputPromptValue(container).length > 0;
    startButton.style.opacity = hasText ? "1" : "0.4";
    startButton.style.cursor = hasText ? "pointer" : "not-allowed";
  };
  const stopAgentInputTypewriter = (textarea, {placeholder, clearDemo = false} = {}) => {
    if (typeof window === "undefined" || !textarea) {
      return;
    }
    const registry = getAgentInputTypewriterRegistry();

    if (!record) {
      if (placeholder) {
        textarea.setAttribute("placeholder", placeholder);
      }
      return;

    record.stopped = true;
    if (record.startTimer) {
      window.clearTimeout(record.startTimer);
    }
    if (record.writeFrame) {
      window.cancelAnimationFrame(record.writeFrame);
    }
    if (record.typewriter?.stop) {
      record.typewriter.stop();
    }

    textarea.removeEventListener("pointerdown", record.handlePointerDown);
    textarea.removeEventListener("input", record.handleInput);

      textarea.value = "";

<div id="define-the-mobile-flow">
  ## Définir le parcours mobile
</div>

Avant de vous lancer dans le développement, déterminez ce que l&#39;utilisateur doit pouvoir faire depuis son téléphone.

Pour Pace Mobile :

* **Public** : les coureurs qui souhaitent enregistrer une course rapidement.
* **Parcours principal** : choisir le type de course → saisir la distance et la durée → enregistrer la course → consulter le résumé du jour.
* **Contrainte mobile** : l&#39;action principale doit être accessible avec un seul pouce.
* **Première version** : pas de comptes, pas de fil social, pas d&#39;analyses avancées.

La première version mobile reste ainsi suffisamment restreinte pour être testée.

Ce processus nécessite une adhésion à l’**Apple Developer Program** avec authentification à deux facteurs. Pour en savoir plus sur l’inscription, les comptes individuels ou d’organisation, la validation par TestFlight ou l’App Store, et les tests externes, consultez [Créer et lancer une app mobile](/fr/build/mobile-app).

<Steps>
  <Step title="Ouvrir l'onglet Publishing et lancer la publication sur l'App Store">
    Une fois votre Preview Expo Go en ligne, ouvrez l&#39;onglet **Publishing**. Sous **Production**, une carte **Ready to publish Pace to the App Store ?** s&#39;affiche. Sélectionnez **Start publishing to the App Store**.

    </Frame>
  </Step>

  <Step title="Choisir votre projet Expo">
    Replit ouvre **Launch** — un flux Expo intégré à Replit. Saisissez ou confirmez le nom et le slug de votre projet Expo, puis sélectionnez **Continue**. Si vous disposez déjà d&#39;un projet Expo, sélectionnez plutôt **Use existing**.

    <Frame>
      <img src="/assets/mirror/app-store-03-choose-expo-project-4707fafffafe.png" alt="La boîte de dialogue Launch à l'étape Choose Expo project — les champs Project name et Project slug pré-remplis avec Pace, un bouton Continue et une option Use existing" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-03-choose-expo-project.png" />
    </Frame>
  </Step>

  <Step title="Se connecter avec Apple">
    Connectez le compte Apple Developer qui détiendra la fiche App Store. Ce compte doit être inscrit au Apple Developer Program, et Apple impose l&#39;authentification à deux facteurs — préparez-vous à valider un code de vérification sur un appareil de confiance.

    <Frame>
      <img src="/assets/mirror/app-store-04-sign-in-apple-459dfc9b84ba.png" alt="La boîte de dialogue Launch à l'étape Sign in with Apple — les champs adresse e-mail ou numéro de téléphone Apple Developer et mot de passe, avec une note indiquant que le mot de passe ne quitte jamais l'appareil et que le compte doit être inscrit au Apple Developer Program" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-04-sign-in-apple.png" />
    </Frame>
  </Step>

  <Step title="Choisir ou créer votre app Apple">
    Saisissez le **App name** et le **Bundle ID** (par exemple `app.replit.pace`) qui apparaîtront sur l&#39;App Store. L&#39;identifiant de bundle ne peut plus être modifié après le premier build : choisissez-le avec soin.

    <Frame>

    </Frame>
  </Step>

  <Step title="Lancer sur l'App Store">
    Replit configure les certificats Apple à votre place. Lorsque l&#39;étape **Ready for launch** s&#39;active, sélectionnez **Launch to App Store**. Replit compile votre app native dans le cloud et soumet le build à App Store Connect pour la revue bêta TestFlight.

    <Frame>
      <img src="/assets/mirror/app-store-06-ready-for-launch-ab02659ee0bb.png" alt="La boîte de dialogue Launch à l'étape Ready for launch, avec Choose Apple app et Configure Apple certificates marquées comme terminées, un bouton Launch to App Store et des Optional launch parameters" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-06-ready-for-launch.png" />

    Une fois Launch terminé, une confirmation indique que l&#39;app cible bien l&#39;App Store, puis la boîte de dialogue se ferme et vous ramène au Project Editor.

      <img src="/assets/mirror/app-store-targeting-f8797071c9e0.png" alt="La boîte de dialogue Launch avec Targeting App Store marqué comme terminé et un message Redirecting back to Replit — l'état de confirmation après avoir sélectionné Launch to App Store" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-targeting.png" />

  </Step>

<Note>
  La première build d’une version nécessite généralement une validation bêta TestFlight avant que les testeurs puissent l’installer. Une fois approuvées, les builds suivantes sont généralement disponibles plus rapidement pour les testeurs. Pour la partager avec des testeurs externes ou la soumettre à la publication sur l’App Store, complétez la fiche dans **App Store Connect** — consultez [Créer et lancer une app mobile](/fr/build/mobile-app) pour accéder à la checklist complète d’App Store Connect.
</Note>

<div id="need-further-help">
  ## Besoin d&#39;aide supplémentaire ?
</div>

* **Agent crée une application web à la place :** recommencez et sélectionnez **Mobile app** comme type d&#39;application avant de lancer la création.
* **L&#39;aperçu Expo Go n&#39;est pas à jour :** secouez votre téléphone, ouvrez le menu développeur et sélectionnez **Refresh**.
* **Les zones tactiles paraissent trop petites :** demandez à Agent d&#39;agrandir les zones tactiles et de placer les actions principales à portée de pouce.
* **La navigation paraît surchargée :** demandez à Agent de réduire l&#39;application au strict minimum d&#39;écrans nécessaires au parcours principal.
* **Le simulateur fonctionne, mais pas le téléphone :** testez les comportements propres au matériel sur un appareil réel et consultez [Applications mobiles natives](/fr/features/artifact-types/building-mobile-apps).
* **La publication échoue :** consultez [Dépannage des applications mobiles](/fr/features/troubleshooting/mobile-app).
