export const YouTubeEmbed = ({videoId, title = "Видео на YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

Replit — это платформа на базе ИИ, позволяющая создавать и публиковать приложения в одной вкладке браузера.
Вместо работы со сложными средами разработки вы получаете инструменты для программирования, публикации и совместной работы в едином интерфейсе.

Традиционная разработка приложений требует установки программ, языков и пакетов — этот процесс настройки занимает много времени.
В Replit среда настраивается мгновенно, поэтому вы можете сразу приступить к созданию приложений независимо от того, новичок вы или опытный разработчик.
