export const YouTubeEmbed = ({videoId, title = "Vidéo YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<div id="unsplash-url">
  ### Gestion améliorée des images pour les URL Unsplash cassées
</div>

Dites adieu aux images d&#39;illustration cassées dans vos projets !

Si vous ne le saviez pas, Agent ajoute déjà des images d&#39;illustration Unsplash à vos projets. Mais il arrive que les URL soient rompues. La raison ? Les LLM sont particulièrement doués pour halluciner des URL d&#39;images d&#39;illustration.

Nous avons ajouté le remplacement automatique des images cassées par des alternatives fonctionnelles, afin que vos applications conservent toujours une apparence professionnelle.
