export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "已复制！";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        复制提示词
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        在 Replit 上使用 Build ↗
      </button>
    </div>;
};

<div id="section">
  ## 开始之前
</div>

你需要一个[Replit 账户](https://replit.com)。本指南将使用**Free 模式**，并通过 Auto 选择模型。{FreeModePlanAvailability}

目标是练习一套可反复使用的提示词流程：

1. 描述一个有用的成果。
2. 添加可观察的成功标准。
3. 明确暂不构建的内容边界。
4. 在 Preview 中测试结果。
5. 请求一项重点明确的改进。

<div id="section-2">
  ## 构建第一个版本
</div>

<Steps>
  <Step title="创建新的 Website">
    新建对话，并选择 **Website** 作为产物。你可以从 **New** 菜单中选择，也可以在提示词框中依次选择 **Add attachment → Create something new → Website**。

    打开模式选择器，然后选择 **Free**。

    <Tip>
      你也可以直接用自然语言让 Replit 构建网站。先选择 Website 可明确要创建的产物，并会在提示词中添加 Website 标签。
    </Tip>
  </Step>

  <Step title="说明预期成果和成功标准">
    使用以下提示词：

    <PromptActions prompt={GATHER_PROMPT} campaign="docs-your-first-app" />

    <Frame>
      <img src="/assets/mirror/01-prompt-free-mode-f5c56ce766c8.jpg" alt="已选择 Website 产物并启用 Free 模式、准备发送的 Gather 提示词" width="1280" height="720" data-path="images/first-app-gather/01-prompt-free-mode.jpg" />
    </Frame>

    此提示词以一个成果为起点：一个用于发现工作坊的应用。接着说明访客应能完成哪些操作，定义可观察的成功标准，并暂不考虑登录、数据、支付和集成。

    这些边界能让首次构建保持专注。待核心体验正常运行后，你可以再添加面向生产环境的功能。
  </Step>

  <Step title="让 Agent 在 Free 模式下构建">
    发送提示词。Agent 会规划产物、创建应用、启动应用并检查结果。你可以在聊天中跟踪进度，等待 Agent 展示 Website 产物并运行 Preview。

    在 Free 模式下，完成首次构建可能需要几分钟。对于更复杂的请求，如果 Replit 建议使用更高级的模式，你可以自行决定是否升级。这个 Gather 构建可在 Free 模式下完成。

    <Frame>
      <img src="/assets/mirror/02-first-version-163a5fb49aac.jpg" alt="默认 Build 布局中显示的第一个可用 Gather Website，其中包含 Agent 聊天和预览" width="1393" height="868" data-path="images/first-app-gather/02-first-version.jpg" />
    </Frame>
  </Step>

  <Step title="在 Preview 中测试结果">
    像访客一样试用应用。请确认以下内容：

    * 可以浏览工作坊列表。
    * 可以打开工作坊详情页面。
    * 可以提交 RSVP。
    * 提交后会显示清晰的确认信息。
    * 可以使用主导航。
    * 可以将 Preview 调整为手机屏幕宽度。

    将结果与提示词中的成功标准进行比较。如果有任何问题，请向 Agent 描述症状，而不是尝试自行诊断代码。
  </Step>
</Steps>

<div id="section-3">
  ## 使用聚焦的提示词改进应用
</div>

当核心体验能够正常运行后，每次只提出一项明确的改进需求。对于 Gather，请添加一个仅显示当前会话中已提交 RSVP 的视图：

> 请添加一个 My RSVPs 视图，让已提交的工作坊显示在当前会话中。保留现有的视觉风格，不要添加登录功能或数据库。

<Frame>
  <img src="/assets/mirror/03-refinement-prompt-94b96ae38acb.jpg" alt="聚焦的后续提示词，要求 Agent 在 Free 模式下添加仅限当前会话的 My RSVPs 视图" width="1393" height="868" data-path="images/first-app-gather/03-refinement-prompt.jpg" />

再次测试 RSVP 流程，然后打开 **My RSVPs**。你提交的工作坊应会显示在当前浏览器会话中。
