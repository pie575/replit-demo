export const YouTubeEmbed = ({videoId, title = "Vidéo YouTube", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

<div id="melhor-tratamento-de-imagens-com-urls-do-unsplash-quebradas">
  ### Meilleure gestion des images en cas d&#39;URL Unsplash cassées
</div>

Dites adieu aux images d&#39;illustration cassées dans vos projets !

Si vous ne le saviez pas, Agent ajoute déjà des images d&#39;illustration issues d&#39;Unsplash à vos projets. Mais il arrive que les URL soient cassées. La raison ? Les LLM sont très doués pour halluciner des URL d&#39;images d&#39;illustration.

Nous avons ajouté une fonctionnalité qui remplace automatiquement les images cassées par des alternatives fonctionnelles, pour que vos applications gardent toujours un rendu professionnel.
