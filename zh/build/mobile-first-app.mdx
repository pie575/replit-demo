export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "已复制！";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        复制提示词
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        在 Replit 上构建 ↗
      </button>
    </div>;
};

<div id="before-you-start">
  ## 开始之前
</div>

你只需注册一个免费的 [Replit 账户](https://replit.com/)。

目标是完成一个完整闭环：

* 使用 [Agent](/zh/features/agent/overview) 构建应用
* 将其作为移动应用测试
* 发布并分享应用

Agent 是 Replit 的 AI 构建助手。你只需描述想要的内容，它就会创建应用、解释
所做的工作，并修复出现的问题。

<div id="1-tell-agent-to-build-the-app">
  ## 1. 让 Agent 构建应用
</div>

打开 Replit 并新建一个应用。描述你想构建的内容，然后选择 **移动应用**
作为应用类型。

    使用预览窗格工具栏中的 **Simulate on...** 下拉菜单 (位于扳手和刷新图标旁) ，即可在 **iOS**、**Android** 和 **Web** 之间切换。

<div id="check-that-it-actually-works">
  ### 检查是否正常运行
</div>

像玩家一样体验 PixelGames：

* 查看 **MENU** 屏幕，确认可以选择 **Blocks**，其他三个游戏均标记为“即将推出”。
* 打开 Blocks，使用方向键和 A/B 按钮玩几轮。
* 确认 HUD 中的分数、等级和消除行数会随游戏进程更新。

如果发现任何异常，请向 Agent 描述你看到的情况；无需自行查找 bug。

<div id="youve-got-pixelgames-running">
  ## PixelGames 已运行
</div>

只需一个 提示词，你就能制作出一款可玩的游戏，并直接在 Replit 中预览。

<div id="troubleshooting">
  ## 故障排查
</div>

<AccordionGroup>
  <Accordion title="预览窗格一直空白">
    首次构建需要一两分钟，请稍等片刻，别急着认定它卡住了。
    如果之后仍然空白，请查看控制台中的报错信息，并把它发给 Agent。
  </Accordion>

  <Accordion title="改动没有出现在预览中">
    在控制台中按 **R** 重新加载 PixelGames；如果内容看起来仍是旧的，再点击预览窗格
    工具栏中的刷新图标。
  </Accordion>

  <Accordion title="预览中出现红色错误页面">
    这通常意味着存在 JavaScript 错误或缺少某个软件包。请仔细阅读报错信息，它往往会
    直接指出出错的文件和行号，然后把它发给 Agent 并请求修复。
  </Accordion>

  <Accordion title="App 在网页端正常，但在 iOS 或 Android 上出错（或反之）">
    有些软件包在不同平台上的表现不一样：能在网页端运行的包，可能并不支持原生平台，
    或者需要另外配置。在依赖某个包之前，先问问 Agent“这个软件包支持原生平台吗？”
  </Accordion>

  <Accordion title="构建耗时很长">
    首次构建总是比较慢，因为还没有任何缓存，后续构建会快很多。
    如果始终很慢，可以让 Agent 检查 project 中是否存在多余的软件包。
  </Accordion>

  <Accordion title="出现找不到模块的错误">
    这说明某个软件包未安装，或者虽已安装但缓存是旧的。可以让 Agent
    安装它或清除缓存，同时再确认一下包名有没有拼错。
  </Accordion>
</AccordionGroup>
