export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "Скопировано!";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        Скопировать промпт
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Создать в Replit ↗
      </button>
    </div>;
};

Вы можете создать и запустить собственную витрину Shopify, просто описав нужный магазин. Agent создаст магазин Shopify, разработает для него фронтенд и добавит ваши товары — всё в рамках одного диалога. Когда будете готовы к продажам, достаточно будет один раз перейти в Shopify, чтобы получить права на магазин и включить приём платежей.

В этом руководстве вы создадите **Ember &amp; Oak** — магазин свечей небольших партий — от промпта до опубликованной витрины.

* Работающая корзина, перенаправляющая покупателей на оформление заказа в Shopify.

Вы узнаете, как:

* Описать магазин так, чтобы Agent порекомендовал Shopify и настроил его за вас.
* Подключить Shopify, не имея дела с ключами API.
* Добавлять и публиковать товары прямо в переписке с Agent.
* Дорабатывать дизайн и путь покупателя.
* Довести магазин от стадии разработки до работающей витрины, готовой к продажам.

<div id="connect-shopify">
  ## Подключение Shopify
</div>

Shopify доступен всем разработчикам во всех приложениях, и подключить его можно прямо в Conversation. Начните с описания магазина, который хотите создать.

<Steps>
  <Step title="Опишите свой магазин">
    Скопируйте промпт и вставьте его в поле ввода промпта на [replit.com](https://replit.com) или нажмите **Build on Replit**, чтобы открыть Replit с уже готовым промптом.

    <Frame>
      <img src="/assets/mirror/describe-your-store-b5d724daee99.png" alt="Поле ввода промпта на главной странице Replit с введённым промптом для магазина свечей Ember & Oak, готовое к началу сборки" width="1440" height="900" data-path="images/replitai/shopify/describe-your-store.png" />
    </Frame>

    <PromptActions prompt={CANDLE_STORE_PROMPT} campaign="docs-ecommerce-store" />

    Agent определит, что вы продаёте физический товар, и порекомендует Shopify в качестве бэкенда для вашей витрины.

    <Frame>
      <img src="/assets/mirror/connect-shopify-store-7f36389f976f.png" alt="Replit Agent передаёт работу skill Shopify для магазина свечей Ember & Oak, показывая карточку «Connect to Shopify» с кнопкой Connect Shopify" width="3452" height="1984" data-path="images/replitai/shopify/connect-shopify-store.png" />
    </Frame>
  </Step>

  <Step title="Подтвердите создание магазина Shopify">
    Agent запросит разрешение создать магазин Shopify для вашего приложения. Подтвердите — и Agent развернёт принадлежащий Replit тестовый магазин и подключит его. Никакой дополнительной настройки не потребуется.
    Ваш адрес электронной почты передаётся Shopify, чтобы позже вы могли получить права на магазин.

    Изначально магазин защищён паролем и не принимает реальные платежи до запуска, так что вы можете спокойно вести разработку без риска.
  </Step>
</Steps>

<Note>
  Каждый магазин Shopify привязан к одному приложению. Если позже вы создадите
  второй магазин, Agent развернёт для этого приложения новый магазин Shopify. Подробнее
  о коннекторе см. в разделе [Подключение Shopify](/ru/features/integrations/shopify).
</Note>

<div id="build-the-storefront">
  ## Создание витрины
</div>

После подключения Shopify Agent проектирует и создаёт витрину для вашего магазина.

<Steps>
  <Step title="Дайте Agent создать фронтенд">
    Agent генерирует витрину по вашему промпту — главный баннер, сетку товаров, корзину и вспомогательные разделы.     По завершении правая панель переключается на **Preview**, чтобы вы могли прокрутить страницу и пройтись по дизайну.

    <Frame>
      <img src="/assets/mirror/storefront-preview-f5ecb9011a8c.png" alt="Витрина Ember & Oak, отображённая в панели Preview Replit рядом с чатом Agent, с главным баннером со свечами и сеткой товаров" width="3452" height="1978" data-path="images/replitai/shopify/storefront-preview.png" />
    </Frame>
  </Step>

  <Step title="Добавьте первые товары">
    В новом магазине каталог пуст. Попросите Agent добавить товар — он создаст его в Shopify и опубликует на витрине.

    <AiPrompt>
      Добавь лавандовую соевую свечу за $18 с коротким уютным описанием, затем опубликуй её на витрине.
    </AiPrompt>

    Товар создаётся в вашем магазине Shopify и появляется в сетке товаров на витрине.

    <Frame>
      <img src="/assets/mirror/add-product-8b098e606340.png" alt="Чат Agent с добавлением лавандовой соевой свечи и новым товаром, появившимся в сетке товаров витрины Ember & Oak" width="3452" height="1978" data-path="images/replitai/shopify/add-product.png" />
    </Frame>
  </Step>

  <Step title="Заполните каталог">
    Продолжайте добавлять товары, просто описывая их. Agent создаёт каждый из них в Shopify с ценой, описанием и изображением.

    <AiPrompt>
      Добавь ещё три свечи: Cedar &amp; Sage за $20, Vanilla Bean за $16 и Sea Salt за $22. Дай каждой короткое описание и опубликуй их.
    </AiPrompt>
  </Step>
</Steps>

<Tip>
  **Что дальше:** попросите Agent сгруппировать товары в коллекции («Создай
  праздничную коллекцию из трёх моих самых продаваемых свечей»), добавить варианты товара,
  например размер или аромат, либо настроить учёт запасов, чтобы витрина показывала,
  что есть в наличии. Shopify при этом остаётся основным источником данных для всего перечисленного.
</Tip>

<div id="iterate-on-the-design-and-checkout">
  ## Доработайте дизайн и оформление заказа
</div>

Любую часть витрины можно изменить через Conversation. Отточите внешний вид, а затем проверьте путь покупателя от начала до конца.

<Steps>
  <Step title="Доработайте дизайн">
    Опишите, что хотите изменить, и Agent обновит витрину.

    <AiPrompt>
      Сделай главный экран более журнальным: крупнее заголовок, больше воздуха и мягкий янтарный акцент на кнопках.
    </AiPrompt>
  </Step>

  <Step title="Проверьте корзину и оформление заказа">
    В Preview добавьте свечу в корзину и начните оформление заказа. Из корзины покупатели попадают на страницу оформления заказа на стороне Shopify. Во время разработки ваш магазин защищён паролем, поэтому при предпросмотре оформления может появиться страница ввода пароля — это нормально и не значит, что что-то сломалось.
  </Step>
</Steps>

<Tip>
  **Идём дальше:** попросите Agent добавить страницу товара, выдвижную
  корзину, поиск или раздел с избранными товарами. Витрина — это
  полноценный project в Replit, поэтому её можно расширить чем угодно, что
  вы сумеете описать: блогом, списком желаний или собственными посадочными страницами, —
  а Shopify возьмёт на себя оформление заказов и их обработку.
</Tip>

<div id="ship-it">
  ## Запуск
</div>

Запуск происходит на стороне Shopify. Попросите Agent запустить процесс Go Live, а затем выполните шаги, которые продавец должен пройти самостоятельно.

<Steps>
  <Step title="Получите права на магазин в Shopify">
    Права на магазин оформляются в Shopify — это единственный шаг, который
    выполняется за пределами Replit. Поскольку при создании магазина ваш адрес
    электронной почты был передан Shopify, вы можете получить права по ссылке от
    Replit или из письма, которое пришлёт Shopify.

    Shopify отправит вам письмо **Store owner invitation**, в котором описаны все
    шаги. Откройте его и выберите **Get started**.

    <Frame>
      <img src="/assets/mirror/claim-store-email-a0d108f2191c.png" alt="Письмо Shopify «Store owner invitation» с приглашением стать владельцем магазина и кнопкой «Get started»" width="856" height="1024" data-path="images/replitai/shopify/claim-store-email.png" />
    </Frame>

    Войдите в существующий аккаунт Shopify или создайте новый, чтобы принять
    приглашение.

    <Frame>
      <img src="/assets/mirror/claim-store-create-account-99d2e132da19.png" alt="Экран Shopify «Create an account» для принятия приглашения с вариантами входа по email, через Google, Apple и Facebook" width="3452" height="1978" data-path="images/replitai/shopify/claim-store-create-account.png" />
    </Frame>

    Выберите **Claim store**, чтобы авторизовать приложение Replit в вашем
    магазине и начать бесплатный пробный период. После этого вы попадёте в
    админ-панель Shopify уже как основной владелец.

    <Frame>
      <img src="/assets/mirror/claim-store-4facb7c8b10c.png" alt="Экран Replit «Claim your store» с кнопкой «Claim store» для начала бесплатного пробного периода Shopify" width="3452" height="1978" data-path="images/replitai/shopify/claim-store.png" />
    </Frame>
  </Step>

  <Step title="Выберите тарифный план и включите приём платежей">
    Выберите тарифный план Shopify и активируйте Shopify Payments или другого
    провайдера. Shopify запрашивает документы, удостоверяющие личность, и
    банковские реквизиты для выплат, — ничего из этого не происходит в Replit.
  </Step>

  <Step title="Настройте доставку, налоги и домен">
    Задайте тарифы доставки и проверьте налоги в Shopify, а при желании
    подключите собственный домен.
  </Step>

  <Step title="Запустите и опубликуйте">
    Снимите пароль с витрины в Shopify, чтобы открыть магазин для покупателей.
    Вернувшись в Replit, нажмите **Publish**, чтобы развернуть витрину по её
    публичному URL.
  </Step>
</Steps>

<Note>
  Платежи, проверка личности, налоги, доставка и антифрод-проверки остаются на
  стороне Shopify. Полный чек-лист запуска и подробности о коннекторе
  см. в разделе [Подключение
  Shopify](/ru/features/integrations/shopify#go-live-on-shopify).
</Note>
