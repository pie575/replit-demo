export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "복사됨!";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        프롬프트 복사
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Replit에서 Build하기 ↗
      </button>
    </div>;
};

<div id="connect-shopify">
  ## Shopify 연결
</div>

Shopify는 모든 앱에서 모든 빌더가 사용할 수 있으며, 대화 안에서 바로 연결할 수 있습니다. 만들고 싶은 스토어를 설명하는 것부터 시작하세요.

<Steps>
  <Step title="스토어 설명하기">
    프롬프트를 복사해 [replit.com](https://replit.com)의 프롬프트 상자에 붙여넣거나, **Build on Replit**을 클릭해 프롬프트가 미리 입력된 상태로 Replit을 여세요.

    <Frame>
      <img src="/assets/mirror/describe-your-store-b5d724daee99.png" alt="Ember & Oak 캔들 스토어 프롬프트가 입력되어 빌드를 시작할 준비가 된 Replit 홈 프롬프트 상자" width="1440" height="900" data-path="images/replitai/shopify/describe-your-store.png" />
    </Frame>

    <PromptActions prompt={CANDLE_STORE_PROMPT} campaign="docs-ecommerce-store" />

    Agent는 실물 상품을 판매한다는 점을 인식하고, 스토어프론트의 백엔드로 Shopify를 추천합니다.

    <Frame>
      <img src="/assets/mirror/connect-shopify-store-7f36389f976f.png" alt="Ember & Oak 캔들 스토어를 위해 Shopify 스킬로 작업을 넘기는 Replit Agent의 화면으로, Connect Shopify 버튼이 있는 'Connect to Shopify' 카드가 표시되어 있습니다" width="3452" height="1984" data-path="images/replitai/shopify/connect-shopify-store.png" />
    </Frame>
  </Step>

  <Step title="Shopify 스토어 생성 승인하기">
    Agent가 앱에 사용할 Shopify 스토어를 만들어도 되는지 묻습니다. 승인하면 Agent가 Replit 소유의 개발용 스토어를 프로비저닝하고 연결합니다. 추가 설정은 필요하지 않습니다.
    나중에 스토어의 소유권을 가져올 수 있도록 이메일 주소가 Shopify와 공유됩니다.

    스토어는 처음에 비밀번호로 보호된 상태이며 정식 오픈 전까지는 실제 결제를 받을 수 없으므로, 부담 없이 자유롭게 빌드할 수 있습니다.
  </Step>
</Steps>

<Note>
  각 Shopify 스토어는 하나의 앱에만 적용됩니다. 나중에 두 번째 스토어를
  만들면 Agent가 해당 앱을 위한 새 Shopify 스토어를 프로비저닝합니다. 커넥터에 대한
  자세한 내용은 [Connect Shopify](/ko/features/integrations/shopify)를 참고하세요.
</Note>

  <Step title="Agent에게 프런트엔드 구축 맡기기">
    Agent는 여러분의 프롬프트를 바탕으로 hero, 제품 그리드, 장바구니, 부가 섹션까지 스토어프론트 전체를 생성합니다.     작업이 끝나면 오른쪽 pane이 **Preview**로 전환되어 페이지를 스크롤하고 디자인을 클릭해 볼 수 있습니다.

    <Frame>

</Tip>

<div id="iterate-on-the-design-and-checkout">
  ## 디자인과 checkout 다듬기
</div>

스토어프론트의 모든 요소는 대화만으로 편집할 수 있습니다. 먼저 외관을 다듬은 다음, 구매자 flow를 처음부터 끝까지 점검해 보세요.

<Steps>
  <Step title="디자인 다듬기">
    바꾸고 싶은 내용을 설명하면 Agent가 스토어프론트를 업데이트합니다.

    <AiPrompt>
      hero section을 좀 더 매거진 편집 디자인처럼 만들어 줘 — 헤드라인을 키우고, 여백을 넓히고, 버튼에는 부드러운 앰버 색상 포인트를 넣어 줘.
    </AiPrompt>
  </Step>

  <Step title="장바구니와 checkout 테스트하기">
    Preview에서 캔들을 장바구니에 담고 checkout을 시작해 보세요. 장바구니는 구매자를 Shopify가 호스팅하는 checkout으로 연결합니다. 개발 중에는 store가 비밀번호로 보호된 상태이므로 checkout 미리보기에 비밀번호 페이지가 나타날 수 있습니다. 정상적인 동작이며 문제가 생긴 것이 아닙니다.
  </Step>
</Steps>

<Tip>
  **한 단계 더:** Agent에게 상품 상세 페이지, 장바구니 drawer, 검색, 추천 products 섹션을 추가해 달라고 요청해 보세요.
  스토어프론트는 실제 Replit project이므로 블로그, 위시리스트, 맞춤형 landing page 등
  설명할 수 있는 것이라면 무엇이든 덧붙일 수 있고, checkout과 주문은 Shopify가 알아서 처리합니다.
</Tip>

<div id="ship-it">
  ## 출시하기
</div>

실제 출시는 Shopify에서 이루어집니다. Agent에게 Go Live 플로우를 시작해 달라고 요청한 다음, 판매자가 직접 진행해야 하는 단계를 완료하세요.

<Steps>
  <Step title="Shopify에서 스토어 소유권 확보하기">
    소유권 확보는 Shopify에서 진행되며, Replit 밖에서 이루어지는 유일한 단계입니다.
    스토어를 생성할 때 여러분의 이메일이 Shopify에 공유되었으므로,
    Replit이 제공하는 링크 또는 Shopify가 보낸 이메일에서 소유권을 확보할 수 있습니다.

    Shopify는 절차를 안내하는 **Store owner invitation** 이메일을 보냅니다.
    이메일을 열고 **Get started**를 선택하세요.

    <Frame>
      <img src="/assets/mirror/claim-store-email-a0d108f2191c.png" alt="스토어 소유자가 되도록 초대하는 Shopify 'Store owner invitation' 이메일과 'Get started' 버튼" width="856" height="1024" data-path="images/replitai/shopify/claim-store-email.png" />
    </Frame>

    기존 Shopify 계정으로 로그인하거나, 새 계정을 만들어 초대를 수락하세요.

    <Frame>
      <img src="/assets/mirror/claim-store-create-account-99d2e132da19.png" alt="초대를 수락하기 위한 Shopify 'Create an account' 화면으로, 이메일, Google, Apple, Facebook 로그인 옵션 제공" width="3452" height="1978" data-path="images/replitai/shopify/claim-store-create-account.png" />
    </Frame>

    **Claim store**를 선택해 스토어에서 Replit 앱을 승인하고 무료 체험을 시작하세요.
    이후 기본 소유자 자격으로 Shopify admin에 접속하게 됩니다.

    <Frame>
      <img src="/assets/mirror/claim-store-4facb7c8b10c.png" alt="Shopify 무료 체험을 시작하는 'Claim store' 버튼이 있는 Replit 'Claim your store' 화면" width="3452" height="1978" data-path="images/replitai/shopify/claim-store.png" />

  </Step>

  <Step title="플랜 선택 및 결제 활성화">
    Shopify 플랜을 선택하고 Shopify Payments 또는 다른 제공업체를 활성화하세요.
    대금 지급에 필요한 신원 및 은행 정보는 Shopify가 수집하며,
    이 과정은 Replit에서 진행되지 않습니다.
  </Step>

  <Step title="배송, 세금, 도메인 설정">
    Shopify에서 배송비를 설정하고 세금을 검토하세요. 필요하다면
    커스텀 도메인을 연결할 수도 있습니다.
  </Step>

  <Step title="출시 및 게시">
    Shopify에서 스토어프론트 비밀번호를 해제해 구매자에게 스토어를 공개하세요.
    다시 Replit으로 돌아와 **Publish**를 클릭하면 스토어프론트가 공개 URL로 배포됩니다.
  </Step>
</Steps>

<Note>
  결제, 신원 확인, 세금, 배송, 사기 검증은 모두 Shopify에서 처리됩니다.
  전체 출시 체크리스트와 커넥터 세부 정보는 [Shopify 연결하기](/ko/features/integrations/shopify#go-live-on-shopify)를 참고하세요.
</Note>
