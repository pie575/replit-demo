export const FreeModePlanAvailability = `режим Free доступен на тарифах Starter, Core и Pro. Лимиты режима Free зависят от вашего тарифа, а в Core и Pro предусмотрен лимит, который обновляется каждые ${FreeModeAllowanceRefreshPeriod}.`;

export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "Скопировано!";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        Скопировать промпт
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Создать в Replit ↗
      </button>
    </div>;
};

export const GATHER_PROMPT = `Создайте адаптивное приложение для поиска мастер-классов под названием Gather.

Посетители должны иметь возможность просматривать предстоящие мастер-классы, открывать информацию о мероприятиях и подтверждать участие, используя реалистичные демонстрационные данные.

Приложение считается готовым, если навигация работает, после подтверждения участия отображается понятное подтверждение, а интерфейс корректно работает на мобильных устройствах и компьютерах.

Пока не добавляйте вход в систему, базу данных, платежи или внешние интеграции.`;

⏰ *Примерное время: 10 минут*

<div id="section">
  ## Перед началом работы
</div>

Вам понадобится [аккаунт Replit](https://replit.com). В этом руководстве используется **режим Free** с Auto для выбора модели. {FreeModePlanAvailability}

Цель — попрактиковаться в повторяемом цикле работы с промптами:

1. Опишите один полезный результат.
2. Добавьте проверяемые критерии успеха.
3. Определите границы того, что пока не нужно создавать.
4. Протестируйте результат в Preview.
5. Попросите об одном точечном улучшении.

<div id="section-2">
  ## Создайте первую версию
</div>

<Steps>
  <Step title="Создайте новый Website">
    Начните новый чат и выберите **Website** в качестве артефакта. Его можно выбрать в меню **New** или в поле промпта: **Add attachment → Create something new → Website**.

    Откройте переключатель режимов и выберите **Free**.

    <Tip>
      Вы также можете просто попросить Replit создать веб-сайт на естественном языке. Если сначала выбрать Website, Replit точнее определит нужный артефакт и добавит к промпту плашку Website.
    </Tip>
  </Step>

  <Step title="Опишите результат и критерии успеха">
    Используйте следующий промпт:

    <PromptActions prompt={GATHER_PROMPT} campaign="docs-your-first-app" />

    <Frame>
      <img src="/assets/mirror/01-prompt-free-mode-f5c56ce766c8.jpg" alt="Промпт Gather готов к отправке: выбран артефакт Website и включён режим Free" width="1280" height="720" data-path="images/first-app-gather/01-prompt-free-mode.jpg" />
    </Frame>

    Этот промпт начинается с одного результата: приложения для поиска воркшопов. Затем в нём описано, что смогут делать посетители, определены проверяемые критерии успеха, а вход в систему, данные, платежи и интеграции отложены на потом.

    Эти рамки помогают сохранить фокус первого Build. Когда основной сценарий будет работать хорошо, вы сможете добавить функции для продакшена.
  </Step>

  <Step title="Поручите Agent выполнить Build в режиме Free">
    Отправьте промпт. Agent спланирует артефакт, создаст и запустит приложение, а затем проверит результат. Следите за ходом работы в чате и дождитесь, пока Agent представит артефакт Website и запустится Preview.

    Первый полный Build в режиме Free может занять несколько минут. Если Replit порекомендует более высокий режим для сложных запросов, вы сами решите, переключаться ли на него. Этот Build Gather будет завершён в режиме Free.

    <Frame>
      <img src="/assets/mirror/02-first-version-163a5fb49aac.jpg" alt="Первый работающий Website Gather в стандартной компоновке Build с чатом Agent и предпросмотром" width="1393" height="868" data-path="images/first-app-gather/02-first-version.jpg" />
    </Frame>
  </Step>

  <Step title="Проверьте результат в Preview">
    Попробуйте использовать приложение как посетитель. Убедитесь, что:

    * можно просматривать список воркшопов;
    * можно открыть страницу с подробностями воркшопа;
    * можно отправить RSVP;
    * после отправки отображается понятное подтверждение;
    * работает основная навигация;
    * Preview можно уменьшить до ширины экрана телефона.

    Сравните результат с критериями успеха из промпта. Если что-то не работает, опишите Agent симптомы, а не пытайтесь самостоятельно диагностировать код.
  </Step>
</Steps>

<div id="section-3">
  ## Улучшайте приложение с помощью точечных промптов
</div>

Когда основной сценарий заработал, запрашивайте по одному понятному улучшению за раз. В случае с Gather добавим представление отправленных RSVP, ограниченное текущей сессией:

> Добавьте представление My RSVPs и сделайте так, чтобы отправленные воркшопы отображались в нём в рамках текущей сессии. Сохраните существующий визуальный стиль и не добавляйте авторизацию или базу данных.

<Frame>
  <img src="/assets/mirror/03-refinement-prompt-94b96ae38acb.jpg" alt="Точечный уточняющий промпт с просьбой к Agent добавить ограниченное сессией представление My RSVPs в режиме Free" width="1393" height="868" data-path="images/first-app-gather/03-refinement-prompt.jpg" />
</Frame>

Снова протестируйте процесс RSVP и откройте **My RSVPs**. Отправленные вами воркшопы должны отображаться в текущей сессии браузера.

Вы прошли основной цикл работы в Replit: **промпт, сборка, тестирование, улучшение**.
