  const AGENT_INPUT_TYPEWRITER_SRC = "https://unpkg.com/typewriter-effect@2.18.2/dist/core.js";
  const AGENT_INPUT_TYPEWRITER_PROMPTS = ["Build a project tracker for my team", "Design a landing page for my coffee shop", "Add Stripe payments to my app"];

  const AGENT_INPUT_DELETE_SPEED = 20;
  const AGENT_INPUT_PAUSE_AFTER = 2000;
  const AGENT_INPUT_START_DELAY = 500;
  const getAgentInputTypewriterLoader = () => {
    if (typeof window === "undefined") {
      return null;
    }
    if (!window.__replitDocsAgentInputTypewriterLoader) {
      window.__replitDocsAgentInputTypewriterLoader = {};
    }
    return window.__replitDocsAgentInputTypewriterLoader;
  };
  const ensureAgentInputTypewriterCore = () => {
    if (typeof window === "undefined" || typeof document === "undefined") {
      return Promise.resolve(null);
    }
    if (window.Typewriter) {
      return Promise.resolve(window.Typewriter);
    }
    const loader = getAgentInputTypewriterLoader();
    if (loader.promise) {
      return loader.promise;
    }
    loader.promise = new Promise((resolve, reject) => {
      const handleLoad = () => {
        if (window.Typewriter) {
          resolve(window.Typewriter);
          return;
        }
        loader.promise = null;
        reject(new Error("Typewriter core loaded without a global Typewriter export."));
      };
      const handleError = () => {
        loader.promise = null;
        reject(new Error("Failed to load typewriter core."));
      };
      const existingScript = document.getElementById(AGENT_INPUT_TYPEWRITER_SCRIPT_ID);
      if (existingScript) {
        existingScript.addEventListener("load", handleLoad, {
          once: true
        });
        existingScript.addEventListener("error", handleError, {
          once: true
        });
        return;
      }
      const script = document.createElement("script");
      script.id = AGENT_INPUT_TYPEWRITER_SCRIPT_ID;
      script.src = AGENT_INPUT_TYPEWRITER_SRC;
      script.async = true;
      script.addEventListener("load", handleLoad, {
        once: true
      });
      script.addEventListener("error", handleError, {
        once: true
      });
      document.head.appendChild(script);
    });
    return loader.promise;
  };
  const getAgentInputTypewriterRegistry = () => {
    if (typeof window === "undefined") {
      return null;
    }
    if (!window.__replitDocsAgentInputTypewriters) {
      window.__replitDocsAgentInputTypewriters = new WeakMap();
    }
    return window.__replitDocsAgentInputTypewriters;
  };
  const findAgentInputContainer = node => node?.closest?.("[data-agent-input-container]") || null;
  const getAgentInputElements = container => ({
    textarea: container?.querySelector?.('[data-agent-input-role="textarea"]') || null,
    startButton: container?.querySelector?.('[data-agent-input-role="start-button"]') || null,
    pillsContainer: container?.querySelector?.('[data-agent-input-role="pills"]') || null
  });
  const getAgentInputPromptValue = container => {
    const {textarea} = getAgentInputElements(container);
    return textarea?.value?.trim() || "";
  };
  const updateAgentInputButtonState = container => {
    const {startButton} = getAgentInputElements(container);
    if (!startButton) {
      return;
    }
    const hasText = getAgentInputPromptValue(container).length > 0;
    startButton.style.opacity = hasText ? "1" : "0.4";
    startButton.style.cursor = hasText ? "pointer" : "not-allowed";
  };
  const stopAgentInputTypewriter = (textarea, {placeholder, clearDemo = false} = {}) => {
    if (typeof window === "undefined" || !textarea) {
      return;
    }
    const registry = getAgentInputTypewriterRegistry();
    const record = registry?.get(textarea);
    if (!record) {
      if (placeholder) {
        textarea.setAttribute("placeholder", placeholder);
      }
      return;
    }
    record.stopped = true;
    if (record.startTimer) {
      window.clearTimeout(record.startTimer);
    }
    if (record.writeFrame) {
      window.cancelAnimationFrame(record.writeFrame);
    }
    if (record.typewriter?.stop) {
      record.typewriter.stop();
    }
    textarea.removeEventListener("focus", record.handleFocus);
    textarea.removeEventListener("pointerdown", record.handlePointerDown);
    textarea.removeEventListener("input", record.handleInput);
    if (clearDemo && textarea.value === record.pendingValue) {
      textarea.value = "";
    }
    textarea.setAttribute("placeholder", placeholder || "");
    updateAgentInputButtonState(record.container);
    registry.delete(textarea);
  };

    if (typeof window === "undefined" || typeof document === "undefined" || !container) {
      return;
    }
    const {textarea} = getAgentInputElements(container);
    if (!textarea) {
      return;
    }
    if (!showTypewriter || defaultPrompt) {
      stopAgentInputTypewriter(textarea, {
        placeholder

      updateAgentInputButtonState(container);
      return;

    const registry = getAgentInputTypewriterRegistry();

      updateAgentInputButtonState(container);
      return;

      renderedValue: "",
      stopped: false,
      startTimer: null,

      typewriter: null,
      handleFocus: null,
      handlePointerDown: null,
      handleInput: null
    };

      record.pendingValue = nextValue;
      if (record.writeFrame) {
        return;
      }
      record.writeFrame = window.requestAnimationFrame(() => {
        record.writeFrame = null;
        if (record.stopped || registry.get(textarea) !== record || !textarea.isConnected) {
          return;
        }
        if (textarea.value !== record.pendingValue) {

        }

      });
    };

      stopAgentInputTypewriter(textarea, {
        placeholder,
        clearDemo: true
      });

    });

          cursorClassName: "",

            record.renderedValue += character;
            queueValue(record.renderedValue);
            return null;
          },
          onRemoveNode: () => {
            record.renderedValue = record.renderedValue.slice(0, -1);
            queueValue(record.renderedValue);
          }
        });

          typewriter.typeString(prompt).pauseFor(AGENT_INPUT_PAUSE_AFTER).deleteAll(AGENT_INPUT_DELETE_SPEED).pauseFor(400);
        });
        record.typewriter = typewriter;
        typewriter.start();
      }).catch(() => {
        stopAgentInputTypewriter(textarea, {
          placeholder
        });
      });

[Vibe coding 101](/ko/learn/foundations/vibe-coding-101)이 마인드셋을 설명하는 반면, 이 페이지는 Agent와 실제로 어떻게 작업하는지 설명합니다. [효과적인 프롬프팅](/ko/learn/effective-prompting)은 더 명확한 지시를 작성하는 방법을 설명합니다.

Agent를 필드의 쿼터백으로 생각하세요: 상황을 읽고, 경로를 선택하고, 작업을 앞으로 진행시킵니다. 여러분은 코치입니다: 전략을 세우고, 플레이를 정하고, 결과를 검토하고, 다음 움직임을 결정합니다.

Agent는 많은 것을 할 수 있지만, 여러분이 프로세스를 이끌 때 가장 잘 작동합니다: 구체적으로, 계획을 세우고, 컨텍스트를 추가하고, 검토 및 테스트하고, 필요할 때 체크포인트를 사용하세요.

## Agent와 빌드하기 위한 다섯 가지 팁

Agent와 작업할 때 이 습관들을 사용하세요:

1. 구체적으로.
2. 계획 세우기.
3. 컨텍스트 추가.
4. 검토 및 테스트.
5. 체크포인트 사용.

## 구체적으로

Agent는 복잡한 작업을 처리할 수 있지만, 명확한 방향이 필요합니다.

유용한 요청은 원하는 것, 중요한 것, 그리고 변경하지 말아야 할 것을 설명합니다. 모든 구현 세부사항을 설명할 필요는 없지만, Agent가 좋은 결정을 내릴 수 있을 만큼 충분한 방향을 제공해야 합니다.

다음 대신:

<AiPrompt>
  내 앱을 더 좋게 만들어줘.
</AiPrompt>

이렇게 시도하세요:

<AiPrompt>
  가입 페이지를 개선하여 방문자가 계정을 만들기 전에 제품이 무엇을 하는지 이해할 수 있게 해줘.<br />
  현재 색상 팔레트를 유지하고, 양식 필드는 그대로 두고, 기본 버튼을 더 눈에 띄게 만들어줘.
</AiPrompt>

두 번째 프롬프트는 Agent에게 목표, 집중 영역, 제약사항을 제공합니다.

요청에 여러 부분이 있을 때 가장 중요한 것을 말하세요.

<AiPrompt>
  케이터링 요청을 관리하기 위한 간단한 관리자 뷰를 추가해줘.<br />

</AiPrompt>

Agent가 그 작업을 단계로 나눌 수 있지만, 여러분의 방향이 성공이 어떤 것인지 알려줍니다.

더 구체적인 지시의 예시는 [효과적인 프롬프팅](/ko/learn/effective-prompting)을 참조하세요.

## 컨텍스트 추가

컨텍스트는 Agent가 무엇이 중요한지 이해하는 데 도움을 줍니다.

텍스트, 스크린샷, 스케치, 파일, 데이터, 오류, 또는 Canvas 주석으로 컨텍스트를 추가할 수 있습니다. 문제를 가장 잘 설명하는 형식을 사용하세요.

변경이 동작, 목표, 또는 제약사항에 관한 것일 때 텍스트 사용:

<AiPrompt>
  컨텍스트: 이 앱은 생일 케이크를 주문하는 부모를 위한 것입니다.<br />
  양식을 짧게 유지하고, 픽업 세부사항을 명확하게 하고, 기업적인 느낌의 것은 피하세요.
</AiPrompt>

레이아웃을 말로 설명하기 어려울 때 스크린샷이나 스케치 사용:

<AiPrompt>
  이 스크린샷을 요청 양식의 레이아웃 방향으로 사용해줘.<br />
  전체 구조는 맞추되 기존 필드와 제출 동작은 유지해줘.
</AiPrompt>

변경이 시각적이고 앱의 특정 부분을 가리키고 싶을 때 [Canvas](/ko/design/canvas) 주석 사용.

<Frame>
  <img src="https://mintcdn.com/replit/63zH0ZBPWnbQaqFI/images/fitstart/work-with-agent/agent-canvas-annotations.png?fit=max&auto=format&n=63zH0ZBPWnbQaqFI&q=85&s=db21c9df13df76c84d98d99cf7fc9730" alt="진행 카드를 가리키는 직사각형, 화살표, 텍스트 레이블이 있는 Canvas의 피트니스 앱 프레임, 양식 영역에 펜 스케치와 레이블" width="1440" height="900" data-path="images/fitstart/work-with-agent/agent-canvas-annotations.png" />
</Frame>

<AiPrompt>
  내 Canvas 주석을 사용하여 히어로 섹션을 업데이트해줘.<br />
  헤드라인, 이미지, 기본 버튼에만 메모를 적용해줘.<br />
  케이터링 요청 양식은 변경하지 마세요.
</AiPrompt>

좋은 컨텍스트에는 변경하지 말아야 할 것도 포함됩니다. 그러면 Agent가 의도치 않게 다른 부분을 다시 작성하지 않고 한 영역을 개선할 수 있습니다.

## 검토 및 테스트

Agent가 빌드 전에 계획을 제안하고, 이후에 변경사항을 요약할 수 있습니다. 둘 다 읽으세요.

작업 계획 카드에서 **보기**를 선택하여 계획을 엽니다. 확장된 계획은 Agent가 무엇을 빌드할 의도인지, 성공 기준, 범위에서 제외된 것, Agent가 따를 단계를 보여줍니다.

<Frame>
  <img src="https://mintcdn.com/replit/63zH0ZBPWnbQaqFI/images/fitstart/work-with-agent/agent-plan-content.png?fit=max&auto=format&n=63zH0ZBPWnbQaqFI&q=85&s=5106f59ceb7b41aa26e3bb8c22daa6bf" alt="What and Why, 완료 기준, 범위 외 항목, 번호가 매겨진 빌드 단계를 보여주는 확장된 작업 계획" width="1440" height="900" data-path="images/fitstart/work-with-agent/agent-plan-content.png" />
</Frame>

Agent가 빌드하기 전에 계획을 검토하세요:

* 이것이 올바른 문제를 해결하나요?
* 빠진 것이 있나요?
* 범위를 벗어난 것이 있나요?
* 먼저 빌드할 더 작은 버전이 있나요?
* 결과를 어떻게 테스트해야 하는지 설명하나요?

계획이 너무 광범위하면 Agent에게 줄이도록 요청하세요.

<AiPrompt>
  이 계획은 너무 광범위합니다.<br />
  오늘 테스트할 수 있는 가장 작은 버전으로 줄여줘.<br />
  나머지는 향후 개선 목록으로 옮겨줘.
</AiPrompt>

Agent가 빌드한 후, 앱을 직접 테스트하세요. Preview를 열고 그것이 대상인 사람처럼 사용하세요.

확인사항:

* 누군가 주요 작업을 완료할 수 있나요?
* 요청한 변경이 발생했나요?
* 모바일에서도 앱이 여전히 작동하나요?
* 중요한 동작이 깨졌나요?
* Agent가 관련 없는 것을 변경했나요?
* 결과가 계속 빌드할 만큼 충분히 좋은가요?

무언가 잘못되었다면 본 것을 설명하세요.

다음 대신:

<AiPrompt>
  양식을 수정해줘.
</AiPrompt>

이렇게 시도하세요:

<AiPrompt>
  케이터링 요청 양식을 제출하면 성공 메시지가 나타나지만 양식에 여전히 이전 텍스트가 표시됩니다.<br />
  제출 후 양식을 지우고 성공 메시지가 계속 보이도록 해줘.
</AiPrompt>

구체적인 피드백이 Agent가 올바른 이슈를 수정하는 데 도움을 줍니다.

## 체크포인트 사용

체크포인트는 변경이 잘못된 방향으로 갔을 때 복구하는 데 도움을 줍니다.

결과가 가깝다면 집중적인 피드백을 사용하세요:

<AiPrompt>
  가깝지만 양식이 너무 길어졌습니다.<br />
  새 시각적 스타일은 유지하고, 선택적 메모 필드를 제거하고, 픽업 날짜를 더 읽기 쉽게 만들어줘.
</AiPrompt>

앱이 이전보다 나빠졌거나, 중요한 동작이 깨졌거나, Agent가 예상보다 더 많이 변경했을 때 체크포인트를 사용하세요.

Agent 패널에서 **History**를 열어 체크포인트를 확인하세요. 각 체크포인트에는 변경사항과 함께 **여기서 롤백** 및 **변경사항** 컨트롤이 표시됩니다.

<Frame>
  <img src="https://mintcdn.com/replit/63zH0ZBPWnbQaqFI/images/fitstart/work-with-agent/agent-checkpoints-history.png?fit=max&auto=format&n=63zH0ZBPWnbQaqFI&q=85&s=fdc1aa8d372341fae140542f3adebad1" alt="'Plan 모드에서 Build 모드로 전환됨'이라는 레이블이 있는 체크포인트와 여기서 롤백 및 변경사항 버튼을 보여주는 Agent History 패널" width="720" height="450" data-path="images/fitstart/work-with-agent/agent-checkpoints-history.png" />
</Frame>

**여기서 롤백**을 선택하면 Agent가 롤백을 적용하기 전에 복원될 내용을 확인합니다.

<Frame>
  <img src="https://mintcdn.com/replit/63zH0ZBPWnbQaqFI/images/fitstart/work-with-agent/agent-checkpoints-rollback-dialog.png?fit=max&auto=format&n=63zH0ZBPWnbQaqFI&q=85&s=cb8fc415dac91353f209eea5f85aa06a" alt="대상 체크포인트와 파일, Agent 메모리, 작업, 데이터베이스 체크박스를 포함한 영향을 받을 내용을 나열하는 체크포인트 롤백 확인 대화 상자" width="900" height="880" data-path="images/fitstart/work-with-agent/agent-checkpoints-rollback-dialog.png" />
</Frame>

앞으로 수정할지 롤백할지 확실하지 않다면 Agent에게 비교를 도와달라고 요청하세요.

<AiPrompt>
  마지막 변경이 앱을 더 나쁘게 만들었습니다.<br />
  현재 버전을 이전 작동 버전과 비교해줘.<br />
  앞으로 수정하는 것이 더 안전한지 롤백하는 것이 더 안전한지, 그리고 이유를 알려줘.
</AiPrompt>

잘못된 방향으로 가는 것은 정상입니다. 체크포인트를 사용하면 더 안전한 버전으로 돌아갈 수 있는 능력을 잃지 않고 탐색할 수 있습니다.

더 많은 복구 가이드는 [체크포인트 및 롤백](/ko/features/version-control/checkpoints-and-rollbacks)을 참조하세요.

## 좋은 Agent 협업의 느낌

다음과 같을 때 올바른 방향에 있습니다:

* 요청이 목표와 제약사항을 설명합니다.
* Agent의 계획이 검토할 만큼 충분히 명확합니다.
* 말로 부족할 때 스크린샷, 파일, Canvas 메모를 추가합니다.
* 변경 후 중요한 흐름을 테스트합니다.
* 무언가 잘못되었을 때 집중적인 피드백을 제공합니다.
* 잘못된 변경을 영원히 풀려고 하는 대신 체크포인트를 사용합니다.
