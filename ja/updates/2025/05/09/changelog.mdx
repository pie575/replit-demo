export const YouTubeEmbed = ({videoId, title = "YouTube video", startAt}) => {
  if (!videoId) {
    return null;
  }
  let url = "https://www.youtube.com/embed/" + videoId;
  if (startAt) {
    url = url + "?start=" + startAt;
  }
  return <Frame>
      <iframe src={url} title={title} allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowFullScreen></iframe>
    </Frame>;
};

### Notion統合

<YouTubeEmbed videoId="ENYxveSU-pM" title="Notion統合" />

NotionのAgent統合がすべてのユーザーに利用可能になり、ReplitプロジェクトとNotionワークスペースのシームレスな統合が実現しました。

  <img src="/assets/mirror/notion-app-2632ca957cc8.png" width="5120" height="2880" data-path="images/changelog/2025-05-09/notion-app.png" />

Agentによるガイド付きセットアップで、Notionの強力なデータベースとドキュメント機能を活用したアプリを素早く構築できます。この統合により、接続されたアプリケーションの構築にNotion APIとの接続が可能になります。

つまり、Notionワークスペースとのデータのプッシュ・プルができるReplitアプリを構築できます！

### 壊れたUnsplash URLの画像処理の改善

プロジェクトの壊れたストック画像にさようならを言いましょう！

ご存知の通り、AgentはすでにプロジェクトにUnsplashからのストック画像を追加しています。しかしURLが壊れることがあります。その理由は？LLMはストック画像URLをハルシネーションするのが得意だからです。

壊れたストック画像を自動的に機能する代替画像に置き換える機能を追加し、アプリが常にプロフェッショナルに表示されるようにしました。
