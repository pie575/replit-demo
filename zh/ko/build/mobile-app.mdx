    if (typeof window === "undefined") {
      return null;
    }
    if (!window.__replitDocsAgentInputTypewriterLoader) {
      window.__replitDocsAgentInputTypewriterLoader = {};
    }
    return window.__replitDocsAgentInputTypewriterLoader;
  };
  const ensureAgentInputTypewriterCore = () => {
    if (typeof window === "undefined" || typeof document === "undefined") {
      return Promise.resolve(null);
    }
    if (window.Typewriter) {
      return Promise.resolve(window.Typewriter);
    }
    const loader = getAgentInputTypewriterLoader();
    if (loader.promise) {
      return loader.promise;
    }
    loader.promise = new Promise((resolve, reject) => {
      const handleLoad = () => {
        if (window.Typewriter) {
          resolve(window.Typewriter);
          return;
        }
        loader.promise = null;
        reject(new Error("Typewriter core loaded without a global Typewriter export."));
      };
      const handleError = () => {
        loader.promise = null;
        reject(new Error("Failed to load typewriter core."));
      };
      const existingScript = document.getElementById(AGENT_INPUT_TYPEWRITER_SCRIPT_ID);
      if (existingScript) {
        existingScript.addEventListener("load", handleLoad, {
          once: true

        existingScript.addEventListener("error", handleError, {
          once: true

        return;

      script.addEventListener("load", handleLoad, {
        once: true

        once: true

      document.head.appendChild(script);
    });
    return loader.promise;

    startButton.style.cursor = hasText ? "pointer" : "not-allowed";
  };
  const stopAgentInputTypewriter = (textarea, {placeholder, clearDemo = false} = {}) => {

      return;

    if (!record) {
      if (placeholder) {
        textarea.setAttribute("placeholder", placeholder);
      }
      return;

    if (record.startTimer) {
      window.clearTimeout(record.startTimer);
    }
    if (record.writeFrame) {
      window.cancelAnimationFrame(record.writeFrame);
    }

      record.typewriter.stop();

    textarea.removeEventListener("input", record.handleInput);

<div id="section-5">
  ## 定义移动端流程
</div>

在构建之前，先明确用户需要在手机上完成哪些操作。

以 Pace Mobile 为例：

* **目标用户**：希望快速记录跑步的跑者。
* **主要流程**：选择跑步类型 → 输入距离和时间 → 保存跑步记录 → 查看今日摘要。
* **移动端限制**：核心操作应能轻松用单拇指触及。
* **首个版本**：无需账户、没有社交动态、没有高级分析功能。

这样可以让首个移动端功能切片保持足够小，便于测试。

使用 Expo Go 在模拟器和手机上测试 Pace Mobile 后，您可以将同一款应用发布到 TestFlight 和 App Store。此流程需要拥有已启用双重认证的 **Apple Developer Program** 会员资格。有关注册、个人账户与组织账户、TestFlight 与 App Store 审核以及外部测试的背景信息，请参阅[构建并上线移动应用](/zh/ko/build/mobile-app)。

<Steps>
  <Step title="打开 Publishing 选项卡，开始发布到应用商店">
    当 Expo Go 预览版本上线后，打开 **Publishing** 标签页。在 **Production** 下方，您会看到 **Ready to publish Pace to the App Store?** 卡片。选择 **Start publishing to the App Store**。

    <Frame>
      <img src="/assets/mirror/app-store-02-start-publishing-190eb7841bcf.png" alt="Project Editor 的 Publishing 标签页，显示 Production 状态以及带有 Start publishing to the App Store 按钮的 Ready to publish Pace to the App Store? 卡片" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-02-start-publishing.png" />
    </Frame>
  </Step>

  <Step title="选择 Expo 项目">
    Replit 会打开 **Launch**，这是 Replit 内置的 Expo 流程。输入或确认 Expo 项目名称和 slug，然后选择 **Continue**。如果已有 Expo 项目，请选择 **Use existing**。

    <Frame>
      <img src="/assets/mirror/app-store-03-choose-expo-project-4707fafffafe.png" alt="Launch 对话框中“Choose Expo project”步骤已启用，项目名称和 slug 字段已预填为 Pace，包含 Continue 按钮和 Use existing 选项" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-03-choose-expo-project.png" />
    </Frame>
  </Step>

  <Step title="使用 Apple 登录">
    关联将拥有 App Store 上架信息的 Apple Developer 账户。该账户必须已加入 Apple Developer Program，且 Apple 要求进行双重认证——请准备好在受信任的设备上批准验证码。

    <Frame>
      <img src="/assets/mirror/app-store-04-sign-in-apple-459dfc9b84ba.png" alt="“使用 Apple 登录”步骤中的 Launch 对话框——包含 Apple Developer 电子邮箱或电话号码及密码字段，并提示密码不会离开设备，且账户必须已加入 Apple Developer Program" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-04-sign-in-apple.png" />
    </Frame>
  </Step>

  <Step title="选择或创建 Apple 应用">
    请输入将在 App Store 中显示的 **App name** 和 **Bundle ID** (例如 `app.replit.pace`) 。Bundle ID 在首次构建后无法更改，请谨慎选择。

    <Frame>
      <img src="/assets/mirror/app-store-05-choose-apple-app-3971aea62f14.png" alt="“选择 Apple App”步骤中的 Launch 对话框：App 名称设为 Pace，Bundle ID 为 app.replit.pace，并提示 Bundle ID 日后无法更改；“正在创建 App…”按钮" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-05-choose-apple-app.png" />
    </Frame>
  </Step>

  <Step title="在应用商店上线">
    Replit 会代表你配置 Apple 证书。当 **Ready for launch** 阶段可用时，选择 **Launch to App Store**。Replit 会在云端构建原生应用，并将构建产物提交到 App Store Connect，进行 TestFlight 测试版审核。

    <Frame>
      <img src="/assets/mirror/app-store-06-ready-for-launch-ab02659ee0bb.png" alt="Ready for launch 阶段的 Launch 对话框，其中 Choose Apple app 和 Configure Apple certificates 显示为已完成，并带有 Launch to App Store 按钮和可选的上线参数" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-06-ready-for-launch.png" />
    </Frame>

    Launch 完成后，会出现确认消息，提示你的应用已定向到应用商店，随后对话框关闭，返回 Project Editor。

    <Frame>
      <img src="/assets/mirror/app-store-targeting-f8797071c9e0.png" alt="Launch 对话框中 Targeting App Store 显示为已完成，并带有 Redirecting back to Replit 消息 —— 选择 Launch to App Store 后的确认状态" width="3452" height="1988" data-path="images/native-mobile-apps/app-store-targeting.png" />
    </Frame>
  </Step>
</Steps>

<Note>
  每个版本的首个构建通常需先通过 TestFlight Beta 审核，测试人员才能安装。获批后，后续构建通常能更快推送给测试人员。若要与外部测试人员共享，或提交至 App Store 上线，请在 **App Store Connect** 中完善应用信息。有关完整的 App Store Connect 清单，请参阅[构建并上线移动应用](/zh/ko/build/mobile-app)。
</Note>

<div id="section-11">
  ## 完成标准
</div>

* 应用能在 iOS Simulator 或 Android Emulator 中打开。
* 应用能通过 Expo Go 在手机上打开。
* 能够完成主要的移动端流程。
* 文字、按钮和导航在手机上用起来顺手自然。
* 已发布或分享了可供测试的移动版本。
* 至少有一位用户可以试用并给出反馈。
* (可选) 已通过 App Store 流程将应用提交到 TestFlight。

<div id="section-13">
  ## 还需要更多帮助吗？
</div>

* **如果 Agent 构建了 Web 应用：** 请重新开始，并在构建前选择 **移动 App** 作为应用类型。
* **如果 Expo Go 预览未更新：** 摇晃手机打开开发者菜单，然后选择 **刷新**。
* **如果触控目标太小：** 请 Agent 增大触控目标，并将默认操作安排在拇指易于触及的范围内。
* **如果导航过于复杂：** 请 Agent 将应用精简为完成主要流程所需的最少页面。
* **如果模拟器能运行，但手机无法运行：** 请在真实设备上测试硬件特有行为，并参阅[原生移动应用](/zh/ko/features/artifact-types/building-mobile-apps)。
* **如果发布失败：** 请参阅[移动应用故障排查](/zh/ko/features/troubleshooting/mobile-app)。
