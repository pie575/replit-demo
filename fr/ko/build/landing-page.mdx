export const PromptActions = ({prompt = "", campaign = "docs-prompt-actions"}) => {
  const LZ_SRC = "https://cdnjs.cloudflare.com/ajax/libs/lz-string/1.5.0/lz-string.min.js";
  const ensureLZString = () => {
    if (typeof document === "undefined") return;
    if (window.LZString) return;
    if (document.querySelector(`script[src="${LZ_SRC}"]`)) return;
    const s = document.createElement("script");
    s.src = LZ_SRC;
    s.async = true;
    document.head.appendChild(s);
  };
  const handleCopy = async e => {
    const btn = e.currentTarget;
    if (!prompt) return;
    try {
      await navigator.clipboard.writeText(prompt);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = prompt;
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand("copy");
      } catch {}
      document.body.removeChild(ta);
    }
    const original = btn.dataset.label || btn.innerText;
    btn.dataset.label = original;
    btn.innerText = "Copié !";
    setTimeout(() => {
      if (btn.isConnected) btn.innerText = original;
    }, 1500);
  };
  const handleBuild = () => {
    if (!prompt || typeof window === "undefined") return;
    const utm = `utm_source=replit-docs&utm_medium=docs&utm_campaign=${encodeURIComponent(campaign)}&utm_content=prompt-actions`;
    let url;
    if (window.LZString) {
      const encoded = window.LZString.compressToEncodedURIComponent(prompt);
      url = `https://replit.com/?prompt=${encoded}&referrer=replit-docs&${utm}`;
    } else {
      url = `https://replit.com/?prompt=${encodeURIComponent(prompt)}&referrer=replit-docs&${utm}`;
    }
    window.open(url, "_blank", "noopener,noreferrer");
  };
  ensureLZString();
  const baseButtonStyle = {
    display: "inline-flex",
    alignItems: "center",
    gap: "6px",
    padding: "8px 16px",
    borderRadius: "8px",
    fontSize: "14px",
    fontWeight: 500,
    cursor: "pointer",
    lineHeight: 1.2,
    fontFamily: "inherit",
    textDecoration: "none"
  };
  return <div style={{
    display: "flex",
    gap: "12px",
    margin: "16px 0",
    flexWrap: "wrap",
    alignItems: "center"
  }}>
      <button type="button" onClick={handleCopy} style={{
    ...baseButtonStyle,
    background: "transparent",
    color: "inherit",
    border: "1px solid rgba(127,127,127,0.4)"
  }}>
        Copier le prompt
      </button>
      <button type="button" onClick={handleBuild} style={{
    ...baseButtonStyle,
    background: "#F26207",
    color: "#FFFFFF",
    border: "1px solid #F26207"
  }}>
        Créer avec Replit ↗
      </button>
    </div>;
};

Les pages de destination offrent un espace dédié à une idée, une campagne, une liste d’attente, un événement ou une suggestion.

<div id="section">
  ## Ce que vous allez apprendre
</div>

Vous apprendrez à :

* transformer un public cible, des suggestions et un appel à l’action en page ;
* demander à Agent de créer le contenu, la mise en page et le comportement des formulaires de la page ;
* vérifier que la page est claire avant de la partager ;
* tester les formulaires depuis l’URL publiée ;
* améliorer la page en fonction de la qualité des réponses.

<div id="canvas">
  ## Affiner dans Canvas et explorer des alternatives
</div>

Une fois qu’Agent a terminé le premier build, ouvrez Canvas pour lui fournir des commentaires visuels et lui demander d’explorer d’autres pistes, sans avoir à réécrire votre prompt.

<Steps>
  <Step title="Créer une page de destination">
    Copiez le prompt et collez-le dans la zone de saisie de [replit.com](https://replit.com), ou cliquez sur **Build on Replit** pour ouvrir Replit avec le prompt déjà renseigné.

    <PromptActions prompt={RACHEL_JONES_PROMPT} />
  </Step>

  <Step title="Aperçu de l'application créée par Agent">
    Une fois l’Agent terminé, le panneau de droite passe à **Preview**, ce qui vous permet d’interagir avec la page de destination en temps réel. Faites défiler la section héro, cliquez sur les boutons d’appel à l’action et vérifiez que le texte et les images correspondent à votre demande avant de commencer à les peaufiner.

    <Frame>
      <img src="/assets/mirror/landing-preview-4855c4b25270.png" alt="Page de destination de la coach fitness Rachel Jones affichée dans la fenêtre Replit Preview à côté du chat Agent : une section héro éditoriale sombre avec le titre « Un corps plus fort. Un esprit plus confiant. Vous, inarrêtable. » sur un portrait aux teintes dorées, les boutons Rejoindre Rachel et Voir l’histoire de Rachel, des certifications à droite et une bande de cinq étoiles indiquant « Plus de 500 femmes dans le monde lui font confiance »" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-preview.png" />
    </Frame>
  </Step>

  <Step title="Ouvrir Canvas">
    Sélectionnez le bouton **Canvas** sous les onglets Preview et Tools pour remplacer Preview par Canvas dans le panneau de droite.

    <Frame>
      <img src="/assets/mirror/landing-canvas-button-494c5b0076b1.png" alt="Espace de travail Replit avec le bouton Canvas mis en évidence sous les onglets Preview et Tools" width="720" height="450" data-path="images/fitstart/launch-landing-page/landing-canvas-button.png" />
    </Frame>
  </Step>

  <Step title="Ajouter des annotations en dehors de l’aperçu">
    Utilisez les outils Canvas **Texte** et **Dessin** pour ajouter des notes et des flèches dans l’espace vide autour de votre maquette, puis indiquez à Agent les éléments précis à modifier. En dehors de l’aperçu, les annotations placées sur le Canvas restent liées à vos commentaires afin qu’Agent puisse les consulter avec le design.

    Limitez chaque annotation à une seule modification. Des notes courtes et précises (« Réduis la largeur de ce bouton ») sont plus efficaces que des commentaires vagues.

    <Frame>
      <img src="/assets/mirror/landing-canvas-annotate-text-8135bfe1e8e2.png" alt="Annotation Canvas « Réduis la largeur de ce bouton » placée dans la marge vide à droite du Canvas de la page de destination Rachel Jones, avec une flèche pointant vers le bouton doré « Book a Free Consultation » dans l’en-tête de la page" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-annotate-text.png" />
    </Frame>
  </Step>

  <Step title="Sélectionnez des annotations et un aperçu à envoyer à Agent">
    Sélectionnez par glissement les annotations avec le frame d’aperçu, puis saisissez un court prompt dans **Chat about selected items** et envoyez-le. Agent interprète les annotations et le design dans un même contexte ; il sait donc à quel élément chaque note fait référence.

    Gardez votre message de chat général — par exemple : « Applique les modifications indiquées dans les annotations. » Incluez les détails dans les annotations.

    <Frame>
      <img src="/assets/mirror/landing-canvas-send-24b771bf29bf.png" alt="Page de destination Rachel Jones sélectionnée dans Canvas avec deux annotations : « Réduis la largeur de ce bouton » pointant vers le bouton Book a Free Consultation et « Ajoute une animation de glissement à l’image de Rachel » pointant vers la photo principale — un prompt de chat est saisi dans le champ Chat about selected items" width="1726" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-send.png" />
    </Frame>
  </Step>

  <Step title="Explorer différentes pistes">
    Dans Canvas, cliquez sur votre design et sélectionnez une suggestion à côté, comme **Explore different approaches**, pour demander à Agent d’explorer d’autres pistes visuelles.

    <Frame>
      <img src="/assets/mirror/landing-canvas-reimagine-9e60c24f342f.png" alt="Trois pistes alternatives de pages de destination générées sous forme de frames Design dans Canvas, à côté de l’original" width="1726" height="994" data-path="images/fitstart/launch-landing-page/landing-canvas-reimagine.png" />
    </Frame>
  </Step>

  <Step title="Explorer les alternatives et itérer">
    Attendez qu’Agent ait terminé la génération. **Explore different approaches** met à jour le même artefact avec une nouvelle variation — utilisez le menu déroulant de l’artefact au-dessus de la maquette pour basculer entre elles. **Explore different vibes** crée une maquette visuelle distincte qui apparaît sur Canvas à côté de l’original.

    Lorsque les variations apparaissent, choisissez une piste et continuez à itérer. Ajoutez une autre annotation au frame sélectionné, sélectionnez par zone la maquette et les annotations, puis envoyez une nouvelle demande — par exemple : &quot;Applique les modifications demandées dans les annotations.&quot; Agent utilise le contexte des variations et les nouvelles annotations dans un même prompt.

    <Frame>
      <img src="/assets/mirror/landing-canvas-zoomed-4d1585d33fe6.png" alt="Page de destination de Rachel Jones dans Canvas après qu’Agent a généré trois variations d’ambiance (Editorial Warmth, Quiet Clinical, Vibrant Pop) — le frame original est sélectionné avec deux nouvelles annotations (« Réduis la largeur de ce bouton » et « Ajoute une animation de glissement à l’image de Rachel ») et le chat contient « Applique les modifications demandées dans les annotations »" width="3452" height="1988" data-path="images/fitstart/launch-landing-page/landing-canvas-zoomed.png" />
    </Frame>
  </Step>

  <Step title="Publier une tâche">
    Lorsque la page fonctionne dans Preview, cliquez sur **Publier** (ou sur **Republier** lors de déploiements ultérieurs) dans l’angle supérieur droit de l’espace de travail pour mettre en ligne la dernière version à l’URL publique. Avant de la partager, ouvrez l’URL publiée dans un nouvel onglet et soumettez une réponse de test pour vérifier que la page publiée collecte bien les réponses.

    <Frame>
      <img src="/assets/mirror/landing-publish-e3b901a79440.png" alt="Coin supérieur droit de l’espace de travail Replit agrandi 2 fois — le bouton gris Republier dans l’en-tête est mis en évidence par un contour orange, et une fenêtre contextuelle ouverte en dessous affiche « Raouf a publié il y a environ 18 heures », Visibilité : Public, une vérification de sécurité et le bouton d’action Republier" width="2142" height="1072" data-path="images/fitstart/launch-landing-page/landing-publish.png" />
    </Frame>
  </Step>
</Steps>
